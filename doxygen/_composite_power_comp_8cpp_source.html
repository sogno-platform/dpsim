23template <typename VarType>
24void CompositePowerComp<VarType>::addMNASubComponent(
25 typename SimPowerComp<VarType>::Ptr subc,
26 MNA_SUBCOMP_TASK_ORDER preStepOrder, MNA_SUBCOMP_TASK_ORDER postStepOrder,
27 Bool contributeToRightVector) {
28 this->mSubComponents.push_back(subc);
29 if (auto mnasubcomp =
30 std::dynamic_pointer_cast<MNASimPowerComp<VarType>>(subc)) {
31 this->mSubcomponentsMNA.push_back(mnasubcomp);
32
33 if (contributeToRightVector) {
34 this->mRightVectorStamps.push_back(mnasubcomp->mRightVector);
35 this->mRightVectorStampComponents.push_back(subc);
36 }
37
38 switch (preStepOrder) {
39 case MNA_SUBCOMP_TASK_ORDER::NO_TASK:
40 break;
41 case MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT: {
42 this->mSubcomponentsPreStepBeforeParent.push_back(mnasubcomp);
43 break;
44 }
45 case MNA_SUBCOMP_TASK_ORDER::TASK_AFTER_PARENT: {
46 this->mSubcomponentsPreStepAfterParent.push_back(mnasubcomp);
47 break;
48 }
49 }
50 switch (postStepOrder) {
51 case MNA_SUBCOMP_TASK_ORDER::NO_TASK:
52 break;
53 case MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT: {
54 this->mSubcomponentsPostStepBeforeParent.push_back(mnasubcomp);
55 break;
56 }
57 case MNA_SUBCOMP_TASK_ORDER::TASK_AFTER_PARENT: {
58 this->mSubcomponentsPostStepAfterParent.push_back(mnasubcomp);
59 break;
60 }
61 }
62 }
63}

65template <typename VarType>
66void CompositePowerComp<VarType>::mnaCompInitialize(
67 Real omega, Real timeStep, Attribute<Matrix>::Ptr leftVector) {
68 SimPowerComp<VarType>::updateMatrixNodeIndices();
69
70 for (auto subComp : mSubcomponentsMNA) {
71 subComp->mnaInitialize(omega, timeStep, leftVector);
72 }
73
74 **this->mRightVector = Matrix::Zero(leftVector->get().rows(), 1);
75
76 // RHS stamps are dense vectors for historical reasons, but a subcomponent
77 // can only write at its connected terminal and virtual-node indices. Cache
78 // those locations once so the composite can aggregate the stamps without
79 // clearing and adding full network-sized vectors on every step.
80 const UInt vectorSize = static_cast<UInt>(leftVector->get().rows());
81 const UInt baseRowLimit = [&]() {
82 if constexpr (std::is_same_v<VarType, Complex>)
83 return vectorSize / (2 * this->mNumFreqs);
84 else
85 return vectorSize;
86 }();
87 auto appendNodeIndices = [baseRowLimit](SimPowerComp<VarType> &component,
88 std::vector<UInt> &indices) {
89 for (UInt terminal = 0; terminal < component.terminalNumberConnected();
90 ++terminal) {
91 if (component.node(terminal)->isGround())
92 continue;
93 for (UInt index : component.matrixNodeIndices(terminal)) {
94 if (index < baseRowLimit)
95 indices.push_back(index);
96 }
97 }
98 for (UInt virtualNode = 0; virtualNode < component.virtualNodesNumber();
99 ++virtualNode) {
100 for (UInt index : component.virtualMatrixNodeIndices(virtualNode)) {
101 if (index < baseRowLimit)
102 indices.push_back(index);
103 }
104 }
105 // A nested composite may contribute at its children's internal virtual
106 // nodes as well as at its own terminals. Its MNA initialization has
107 // already cached those rows before this parent is initialized.
108 if (auto nested = dynamic_cast<CompositePowerComp<VarType> *>(&component))
109 indices.insert(indices.end(), nested->mRightVectorAffectedIndices.begin(),
110 nested->mRightVectorAffectedIndices.end());
111 std::sort(indices.begin(), indices.end());
112 indices.erase(std::unique(indices.begin(), indices.end()), indices.end());
113 };
114
115 mRightVectorStampIndices.clear();
116 mRightVectorStampIndices.reserve(mRightVectorStampComponents.size());
117 mRightVectorAffectedIndices.clear();
118 appendNodeIndices(*this, mRightVectorAffectedIndices);
119 for (const auto &component : mRightVectorStampComponents) {
120 std::vector<UInt> indices;
121 appendNodeIndices(*component, indices);
122 mRightVectorAffectedIndices.insert(mRightVectorAffectedIndices.end(),
123 indices.begin(), indices.end());
124 mRightVectorStampIndices.push_back(std::move(indices));
125 }
126 std::sort(mRightVectorAffectedIndices.begin(),
127 mRightVectorAffectedIndices.end());
128 mRightVectorAffectedIndices.erase(
129 std::unique(mRightVectorAffectedIndices.begin(),
130 mRightVectorAffectedIndices.end()),
131 mRightVectorAffectedIndices.end());
132 mRightVectorStampIndicesInitialized = true;
133
134 mnaParentInitialize(omega, timeStep, leftVector);
135}

146template <typename VarType>
147void CompositePowerComp<VarType>::mnaCompApplyRightSideVectorStamp(
148 Matrix &rightVector) {
149 if (!mRightVectorStampIndicesInitialized) {
150 rightVector.setZero();
151 for (const auto &stamp : mRightVectorStamps) {
152 if ((**stamp).size() != 0)
153 rightVector += **stamp;
154 }
155 } else {
156 if constexpr (std::is_same_v<VarType, Complex>) {
157 for (UInt freq = 0; freq < this->mNumFreqs; ++freq)
158 for (UInt index : mRightVectorAffectedIndices)
159 Math::setVectorElement(rightVector, index, Complex(0., 0.),
160 this->mNumFreqs, freq);
161 } else {
162 for (UInt index : mRightVectorAffectedIndices)
163 rightVector(index, 0) = 0.;
164 }
165 for (std::size_t stampIndex = 0; stampIndex < mRightVectorStamps.size();
166 ++stampIndex) {
167 const auto &stamp = mRightVectorStamps[stampIndex];
168 if ((**stamp).size() == 0)
169 continue;
170 if constexpr (std::is_same_v<VarType, Complex>) {
171 for (UInt freq = 0; freq < this->mNumFreqs; ++freq)
172 for (UInt index : mRightVectorStampIndices[stampIndex])
173 Math::addToVectorElement(rightVector, index,
174 Math::complexFromVectorElement(
175 **stamp, index, this->mNumFreqs, freq),
176 this->mNumFreqs, freq);
177 } else {
178 for (UInt index : mRightVectorStampIndices[stampIndex])
179 rightVector(index, 0) += (**stamp)(index, 0);
180 }
181 }
182 }
183 mnaParentApplyRightSideVectorStamp(rightVector);
184}

186template <typename VarType>
187void CompositePowerComp<VarType>::mnaAddCompactRightSideVectorStamp(
188 Matrix &rightVector) {
189 if (!mRightVectorStampIndicesInitialized) {
190 rightVector += **this->mRightVector;
191 return;
192 }
193
194 const Matrix &stamp = **this->mRightVector;
195 if constexpr (std::is_same_v<VarType, Complex>) {
196 for (UInt freq = 0; freq < this->mNumFreqs; ++freq)
197 for (UInt index : mRightVectorAffectedIndices)
198 Math::addToVectorElement(
199 rightVector, index,
200 Math::complexFromVectorElement(stamp, index, this->mNumFreqs, freq),
201 this->mNumFreqs, freq);
202 } else {
203 for (UInt index : mRightVectorAffectedIndices)
204 rightVector(index, 0) += stamp(index, 0);
205 }
206}
