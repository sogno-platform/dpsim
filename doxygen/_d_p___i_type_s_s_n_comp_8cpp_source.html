23void DP::ITypeSSNComp::initializeFromNodesAndTerminals(Real frequency) {
24 if (!mParametersSet)
25 throw std::logic_error("setParameters() must be called before "
26 "initializeFromNodesAndTerminals().");
27
28 Real omega = 2. * PI * frequency;
29 MatrixComp uInit = buildInitialInputFromNodes(frequency);
30 MatrixComp xInit = calculateSteadyStateStateFromInput(uInit, omega);
31 MatrixComp yInit = calculateSteadyStateOutputFromInput(xInit, uInit);
32
33 **mX = xInit;
34 **mIntfCurrent = uInit;
35 **mIntfVoltage = yInit;
36
37 SPDLOG_LOGGER_INFO(mSLog,
38 "\n--- Initialization from nodes and terminals ---"
39 "\nInput u: {:s}"
40 "\nOutput y: {:s}"
41 "\nState x: {:s}"
42 "\n--- Initialization finished ---",
43 Logger::matrixCompToString(**mIntfCurrent),
44 Logger::matrixCompToString(**mIntfVoltage),
45 Logger::matrixCompToString(**mX));
46}
