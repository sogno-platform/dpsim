21DP::Ph1::AvVoltageSourceInverterDQ::AvVoltageSourceInverterDQ(
22 String uid, String name, Bool useSubcomponents, Logger::Level logLevel,
23 Bool withTrafo)
24 : CompositePowerComp<Complex>(uid, name, true, true, logLevel),
25 mOmegaN(mAttributes->create<Real>("Omega_nom")),
26 mPref(mAttributes->create<Real>("P_ref")),
27 mQref(mAttributes->create<Real>("Q_ref")),
28 mVcd(mAttributes->create<Real>("Vc_d", 0)),
29 mVcq(mAttributes->create<Real>("Vc_q", 0)),
30 mIrcd(mAttributes->create<Real>("Irc_d", 0)),
31 mIrcq(mAttributes->create<Real>("Irc_q", 0)),
32 mVsref(mAttributes->create<MatrixComp>("Vsref", MatrixComp::Zero(1, 1))),
33 mVs(mAttributes->createDynamic<MatrixComp>("Vs")),
34 mDirectSourceVoltageAttr(mAttributes->create<MatrixComp>(
35 "direct_source_voltage", MatrixComp::Zero(1, 1))),
36 mPllOutput(mAttributes->createDynamic<Matrix>("pll_output")),
37 mPowerctrlInputs(mAttributes->createDynamic<Matrix>("powerctrl_inputs")),
38 mPowerctrlOutputs(
39 mAttributes->createDynamic<Matrix>("powerctrl_outputs")),
40 mPowerctrlStates(mAttributes->createDynamic<Matrix>("powerctrl_states")) {
41
42 mUseSubcomponents = useSubcomponents;
43 if (withTrafo) {
44 setVirtualNodeNumber(4);
45 mConnectionTransformer = DP::Ph1::Transformer::make(
46 **mName + "_trans", **mName + "_trans", mLogLevel, false);
47 addMNASubComponent(mConnectionTransformer,
48 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
49 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
50 } else {
51 setVirtualNodeNumber(mUseSubcomponents ? 3 : 4);
52 }
53 mWithConnectionTransformer = withTrafo;
54 setTerminalNumber(1);
55
56 SPDLOG_LOGGER_INFO(mSLog, "Create {} {}", type(), name);
57 **mIntfVoltage = MatrixComp::Zero(1, 1);
58 **mIntfCurrent = MatrixComp::Zero(1, 1);
59
60 if (mUseSubcomponents) {
61 // Create electrical sub components
62 mSubResistorF = DP::Ph1::Resistor::make(**mName + "_resF", mLogLevel);
63 mSubResistorC = DP::Ph1::Resistor::make(**mName + "_resC", mLogLevel);
64 mSubCapacitorF = DP::Ph1::Capacitor::make(**mName + "_capF", mLogLevel);
65 mSubInductorF = DP::Ph1::Inductor::make(**mName + "_indF", mLogLevel);
66 addMNASubComponent(mSubResistorF,
67 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
68 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
69 addMNASubComponent(mSubResistorC,
70 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
71 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
72 addMNASubComponent(mSubCapacitorF,
73 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
74 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
75 addMNASubComponent(mSubInductorF,
76 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
77 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
78 }
79
80 // Keep the controlled source as a component in transformer-enabled direct
81 // mode. Its branch unknown is distinct from the transformer's low-voltage
82 // node, while the R-L-C filter remains stamped directly by this parent.
83 if (!mUseSubcomponents && mWithConnectionTransformer) {
84 mSubCtrledVoltageSource =
85 DP::Ph1::VoltageSource::make(**mName + "_src", mLogLevel);
86 addMNASubComponent(mSubCtrledVoltageSource, MNA_SUBCOMP_TASK_ORDER::NO_TASK,
87 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
88 } else if (mUseSubcomponents) {
89 mSubCtrledVoltageSource =
90 DP::Ph1::VoltageSource::make(**mName + "_src", mLogLevel);
91 addMNASubComponent(mSubCtrledVoltageSource, MNA_SUBCOMP_TASK_ORDER::NO_TASK,
92 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
93 }
94
95 SPDLOG_LOGGER_INFO(mSLog, "Electrical subcomponents: ");
96 for (auto subcomp : mSubComponents)
97 SPDLOG_LOGGER_INFO(mSLog, "- {}", subcomp->name());
98
99 // Create control sub components
100 mPLL = Signal::PLL::make(**mName + "_PLL", mLogLevel);
101 mPowerControllerVSI = Signal::PowerControllerVSI::make(
102 **mName + "_PowerControllerVSI", mLogLevel);
103
104 // Sub voltage source
105 if (mUseSubcomponents || mWithConnectionTransformer)
106 mVs->setReference(mSubCtrledVoltageSource->mIntfVoltage);
107 else
108 mVs->setReference(mDirectSourceVoltageAttr);
109
110 // PLL
111 mPLL->mInputRef->setReference(mVcq);
112 mPllOutput->setReference(mPLL->mOutputCurr);
113
114 // Power controller
115 // input references
116 mPowerControllerVSI->mVc_d->setReference(mVcd);
117 mPowerControllerVSI->mVc_q->setReference(mVcq);
118 mPowerControllerVSI->mIrc_d->setReference(mIrcd);
119 mPowerControllerVSI->mIrc_q->setReference(mIrcq);
120 // input, state and output vector for logging
121 mPowerctrlInputs->setReference(mPowerControllerVSI->mInputCurr);
122 mPowerctrlStates->setReference(mPowerControllerVSI->mStateCurr);
123 mPowerctrlOutputs->setReference(mPowerControllerVSI->mOutputCurr);
124}

225void DP::Ph1::AvVoltageSourceInverterDQ::initializeParentFromNodesAndTerminals(
226 Real frequency) {
227
228 // set initial interface quantities
229 (**mIntfVoltage)(0, 0) = initialSingleVoltage(0);
230 (**mIntfCurrent)(0, 0) =
231 -std::conj(Complex(**mPref, **mQref) / (**mIntfVoltage)(0, 0));
232
233 Complex filterInterfaceInitialVoltage;
234 Complex filterInterfaceInitialCurrent;
235
236 if (mWithConnectionTransformer) {
237 // calculate quantities of low voltage side of transformer (being the interface quantities of the filter)
238 filterInterfaceInitialVoltage =
239 ((**mIntfVoltage)(0, 0) -
240 Complex(mTransformerResistance, mTransformerInductance * **mOmegaN) *
241 (**mIntfCurrent)(0, 0)) /
242 Complex(mTransformerRatioAbs, mTransformerRatioPhase);
243 filterInterfaceInitialCurrent =
244 (**mIntfCurrent)(0, 0) *
245 Complex(mTransformerRatioAbs, mTransformerRatioPhase);
246
247 // connect transformer
248 mVirtualNodes[3]->setInitialVoltage(filterInterfaceInitialVoltage);
249 mConnectionTransformer->connect({mTerminals[0]->node(), mVirtualNodes[3]});
250 } else {
251 // if no transformer used, filter interface equal to inverter interface
252 filterInterfaceInitialVoltage = (**mIntfVoltage)(0, 0);
253 filterInterfaceInitialCurrent = (**mIntfCurrent)(0, 0);
254 }
255
256 // derive initialization quantities of filter
257 Complex vcInit =
258 filterInterfaceInitialVoltage - filterInterfaceInitialCurrent * mRc;
259 Complex icfInit = vcInit * Complex(0., 2. * PI * frequency * mCf);
260 Complex vfInit = vcInit - (filterInterfaceInitialCurrent - icfInit) *
261 Complex(0., 2. * PI * frequency * mLf);
262 Complex vsInit =
263 vfInit - (filterInterfaceInitialCurrent - icfInit) * Complex(mRf, 0);
264 mVirtualNodes[0]->setInitialVoltage(vsInit);
265 mVirtualNodes[1]->setInitialVoltage(vfInit);
266 mVirtualNodes[2]->setInitialVoltage(vcInit);
267
268 (**mVsref)(0, 0) = mVirtualNodes[0]->initialSingleVoltage();
269 mDirectSourceValue = (**mVsref)(0, 0);
270 (**mDirectSourceVoltageAttr)(0, 0) = mDirectSourceValue;
271 if (mUseSubcomponents) {
272 mSubCtrledVoltageSource->setParameters((**mVsref)(0, 0));
273 mSubCtrledVoltageSource->connect({SimNode::GND, mVirtualNodes[0]});
274 mSubResistorF->connect({mVirtualNodes[0], mVirtualNodes[1]});
275 mSubInductorF->connect({mVirtualNodes[1], mVirtualNodes[2]});
276 mSubCapacitorF->connect({mVirtualNodes[2], SimNode::GND});
277 if (mWithConnectionTransformer)
278 mSubResistorC->connect({mVirtualNodes[2], mVirtualNodes[3]});
279 else
280 mSubResistorC->connect({mVirtualNodes[2], mTerminals[0]->node()});
281
282 for (auto subcomp : mSubComponents) {
283 subcomp->initialize(mFrequencies);
284 subcomp->initializeFromNodesAndTerminals(frequency);
285 }
286 } else {
287 mDirectGridCurrent = filterInterfaceInitialCurrent;
288 // DP inductor/capacitor component voltage is terminal-1 minus terminal-0.
289 mDirectInductorVoltage = vcInit - vfInit;
290 mDirectInductorCurrent = filterInterfaceInitialCurrent - icfInit;
291 mDirectCapacitorVoltage = -vcInit;
292 mDirectCapacitorCurrent = -icfInit;
293 if (mWithConnectionTransformer) {
294 mSubCtrledVoltageSource->setParameters((**mVsref)(0, 0));
295 mSubCtrledVoltageSource->connect({SimNode::GND, mVirtualNodes[0]});
296 }
297 }
298
299 // Initialize control subcomponents
300 // current and voltage inputs to PLL and power controller
301 Complex vcdq, ircdq;
302 // The filter-side virtual node is node 3 only when the optional connection
303 // transformer is present. Without it, use the filter interface quantities
304 // derived above; indexing virtual node 3 in that configuration is invalid.
305 const Complex pllVoltage = mWithConnectionTransformer
306 ? mVirtualNodes[3]->initialSingleVoltage()
307 : filterInterfaceInitialVoltage;
308 vcdq = Math::rotatingFrame2to1(pllVoltage, std::arg(pllVoltage), 0);
309 const Complex initialGridCurrent = mUseSubcomponents
310 ? (**mSubResistorC->mIntfCurrent)(0, 0)
311 : mDirectGridCurrent;
312 ircdq = Math::rotatingFrame2to1(-1. * initialGridCurrent,
313 std::arg(pllVoltage), 0);
314 **mVcd = vcdq.real();
315 **mVcq = vcdq.imag();
316 **mIrcd = ircdq.real();
317 **mIrcq = ircdq.imag();
318 const Complex vsdq = Math::rotatingFrame2to1(vsInit, std::arg(pllVoltage), 0);
319 mPowerControllerVSI->initializeSteadyState(**mPref, **mQref, **mIrcd, **mIrcq,
320 vsdq.real(), vsdq.imag());
321 // angle input
322 Matrix matrixStateInit = Matrix::Zero(2, 1);
323 matrixStateInit(0, 0) = std::arg(pllVoltage);
324 mPLL->setInitialValues(**mVcq, matrixStateInit, Matrix::Zero(2, 1));
325
326 SPDLOG_LOGGER_INFO(
327 mSLog,
328 "\n--- Initialization from powerflow ---"
329 "\nInterface voltage across: {:s}"
330 "\nInterface current: {:s}"
331 "\nTerminal 0 initial voltage: {:s}"
332 "\nTerminal 0 connected to {:s} = sim node {:d}"
333 "\nVirtual node 0 initial voltage: {:s}"
334 "\nVirtual node 1 initial voltage: {:s}"
335 "\nVirtual node 2 initial voltage: {:s}",
336 Logger::phasorToString((**mIntfVoltage)(0, 0)),
337 Logger::phasorToString((**mIntfCurrent)(0, 0)),
338 Logger::phasorToString(initialSingleVoltage(0)),
339 mTerminals[0]->node()->name(), mTerminals[0]->node()->matrixNodeIndex(),
340 Logger::phasorToString(mVirtualNodes[0]->initialSingleVoltage()),
341 Logger::phasorToString(mVirtualNodes[1]->initialSingleVoltage()),
342 Logger::phasorToString(mVirtualNodes[2]->initialSingleVoltage()));
343 if (mWithConnectionTransformer)
344 SPDLOG_LOGGER_INFO(
345 mSLog, "\nVirtual node 3 initial voltage: {:s}",
346 Logger::phasorToString(mVirtualNodes[3]->initialSingleVoltage()));
347 SPDLOG_LOGGER_INFO(mSLog, "\n--- Initialization from powerflow finished ---");
348}

350void DP::Ph1::AvVoltageSourceInverterDQ::mnaParentInitialize(
351 Real omega, Real timeStep, Attribute<Matrix>::Ptr leftVector) {
352 mTimeStep = timeStep;
353
354 if (!mUseSubcomponents) {
355 updateMatrixNodeIndices();
356 const Real b = timeStep * omega / 2.;
357 const Real a = timeStep / (2. * mLf);
358 const Real denominator = 1. + b * b;
359 mDirectInductorConductance = Complex(a / denominator, -a * b / denominator);
360 mDirectInductorPreviousCurrentFactor =
361 Complex((1. - b * b) / denominator, -2. * b / denominator);
362 mDirectCapacitorConductance = Complex(2. * mCf / timeStep, omega * mCf);
363 mDirectCapacitorPreviousVoltageFactor =
364 Complex(2. * mCf / timeStep, -omega * mCf);
365 mDirectInductorHistory =
366 mDirectInductorConductance * mDirectInductorVoltage +
367 mDirectInductorPreviousCurrentFactor * mDirectInductorCurrent;
368 mDirectCapacitorHistory =
369 -mDirectCapacitorCurrent -
370 mDirectCapacitorPreviousVoltageFactor * mDirectCapacitorVoltage;
371
372 if (mWithConnectionTransformer) {
373 const UInt harmonicOffset =
374 static_cast<UInt>(leftVector->get().rows()) / mNumFreqs;
375 const UInt complexOffset = harmonicOffset / 2;
376 const UInt vectorSize = static_cast<UInt>(leftVector->get().rows());
377 auto appendComponentRows = [&](SimPowerComp<Complex> &component) {
378 std::vector<UInt> baseIndices;
379 for (UInt terminal = 0; terminal < component.terminalNumberConnected();
380 ++terminal) {
381 if (component.node(terminal)->isGround())
382 continue;
383 const auto indices = component.matrixNodeIndices(terminal);
384 baseIndices.insert(baseIndices.end(), indices.begin(), indices.end());
385 }
386 for (UInt virtualNode = 0; virtualNode < component.virtualNodesNumber();
387 ++virtualNode) {
388 const auto indices = component.virtualMatrixNodeIndices(virtualNode);
389 baseIndices.insert(baseIndices.end(), indices.begin(), indices.end());
390 }
391 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
392 for (UInt index : baseIndices) {
393 const UInt realIndex = index + freq * harmonicOffset;
394 const UInt imagIndex = realIndex + complexOffset;
395 if (realIndex < vectorSize)
396 mDirectChildRightVectorIndices.push_back(realIndex);
397 if (imagIndex < vectorSize)
398 mDirectChildRightVectorIndices.push_back(imagIndex);
399 }
400 }
401 };
402 appendComponentRows(*mConnectionTransformer);
403 appendComponentRows(*mSubCtrledVoltageSource);
404 std::sort(mDirectChildRightVectorIndices.begin(),
405 mDirectChildRightVectorIndices.end());
406 mDirectChildRightVectorIndices.erase(
407 std::unique(mDirectChildRightVectorIndices.begin(),
408 mDirectChildRightVectorIndices.end()),
409 mDirectChildRightVectorIndices.end());
410 }
411 }
412
413 // initialize state space controller
414 mPowerControllerVSI->initializeStateSpaceModel(omega, timeStep, leftVector);
415 mPLL->setSimulationParameters(timeStep);
416
417 // TODO: these are actually no MNA tasks
418 mMnaTasks.push_back(std::make_shared<ControlPreStep>(*this));
419 mMnaTasks.push_back(std::make_shared<ControlStep>(*this));
420}

455void DP::Ph1::AvVoltageSourceInverterDQ::controlStep(Real time,
456 Int timeStepCount) {
457 // Transformation interface forward
458 const Complex pllVoltage = mWithConnectionTransformer
459 ? mVirtualNodes[3]->singleVoltage()
460 : (**mIntfVoltage)(0, 0);
461 Complex vcdq, ircdq;
462 vcdq = Math::rotatingFrame2to1(
463 pllVoltage, mPLL->attributeTyped<Matrix>("output_prev")->get()(0, 0),
464 mThetaN);
465 const Complex gridCurrent = mUseSubcomponents
466 ? (**mSubResistorC->mIntfCurrent)(0, 0)
467 : mDirectGridCurrent;
468 ircdq = Math::rotatingFrame2to1(
469 -1. * gridCurrent,
470 mPLL->attributeTyped<Matrix>("output_prev")->get()(0, 0), mThetaN);
471 **mVcd = vcdq.real();
472 **mVcq = vcdq.imag();
473 **mIrcd = ircdq.real();
474 **mIrcq = ircdq.imag();
475
476 // add step of subcomponents
477 mPLL->signalStep(time, timeStepCount);
478 mPowerControllerVSI->signalStep(time, timeStepCount);
479
480 // Transformation interface backward
481 (**mVsref)(0, 0) = Math::rotatingFrame2to1(
482 Complex(mPowerControllerVSI->attributeTyped<Matrix>("output_curr")
483 ->get()(0, 0),
484 mPowerControllerVSI->attributeTyped<Matrix>("output_curr")
485 ->get()(1, 0)),
486 mThetaN, mPLL->attributeTyped<Matrix>("output_prev")->get()(0, 0));
487
488 // Update nominal system angle
489 mThetaN = mThetaN + mTimeStep * **mOmegaN;
490}

492void DP::Ph1::AvVoltageSourceInverterDQ::mnaParentApplySystemMatrixStamp(
493 SparseMatrixRow &systemMatrix) {
494 if (mUseSubcomponents) {
495 return CompositePowerComp<Complex>::mnaParentApplySystemMatrixStamp(
496 systemMatrix);
497 }
498
499 const Int sourceNode = mVirtualNodes[0]->matrixNodeIndex();
500 const Int resistorNode = mVirtualNodes[1]->matrixNodeIndex();
501 const Int capacitorNode = mVirtualNodes[2]->matrixNodeIndex();
502 const Int sourceBranch = mVirtualNodes[3]->matrixNodeIndex();
503 const Int terminal = matrixNodeIndex(0);
504 const Int filterOutput = mWithConnectionTransformer
505 ? mVirtualNodes[3]->matrixNodeIndex()
506 : terminal;
507 const Complex rfConductance(1. / mRf, 0.);
508 const Complex rcConductance(1. / mRc, 0.);
509
510 MNAStampUtils::stampAdmittance(rfConductance, systemMatrix, sourceNode,
511 resistorNode, true, true, mSLog);
512 MNAStampUtils::stampAdmittance(mDirectInductorConductance, systemMatrix,
513 resistorNode, capacitorNode, true, true,
514 mSLog);
515 MNAStampUtils::stampAdmittance(mDirectCapacitorConductance, systemMatrix,
516 capacitorNode, 0, true, false, mSLog);
517 MNAStampUtils::stampAdmittance(
518 rcConductance, systemMatrix, capacitorNode, filterOutput, true,
519 mWithConnectionTransformer || terminalNotGrounded(0), mSLog);
520
521 if (!mWithConnectionTransformer) {
522 Math::setMatrixElement(systemMatrix, sourceBranch, sourceNode,
523 Complex(1., 0.));
524 Math::setMatrixElement(systemMatrix, sourceNode, sourceBranch,
525 Complex(1., 0.));
526 }

529void DP::Ph1::AvVoltageSourceInverterDQ::mnaParentApplyRightSideVectorStamp(
530 Matrix &rightVector) {
531 if (mUseSubcomponents)
532 return CompositePowerComp<Complex>::mnaParentApplyRightSideVectorStamp(
533 rightVector);
534
535 const Int sourceNode = mVirtualNodes[0]->matrixNodeIndex();
536 const Int resistorNode = mVirtualNodes[1]->matrixNodeIndex();
537 const Int capacitorNode = mVirtualNodes[2]->matrixNodeIndex();
538 const Complex inductorEquivalent =
539 mDirectInductorConductance * mDirectInductorVoltage +
540 mDirectInductorPreviousCurrentFactor * mDirectInductorCurrent;
541 const Complex capacitorEquivalent =
542 -mDirectCapacitorCurrent -
543 mDirectCapacitorPreviousVoltageFactor * mDirectCapacitorVoltage;
544 if (!mWithConnectionTransformer)
545 Math::setVectorElement(rightVector, mVirtualNodes[3]->matrixNodeIndex(),
546 mDirectSourceValue, mNumFreqs);
547 Math::setVectorElement(rightVector, resistorNode, inductorEquivalent,
548 mNumFreqs);
549 Math::setVectorElement(rightVector, capacitorNode,
550 capacitorEquivalent - inductorEquivalent, mNumFreqs);
551}

553void DP::Ph1::AvVoltageSourceInverterDQ::mnaAddCompactRightSideVectorStamp(
554 Matrix &rightVector) {
555 CompositePowerComp<Complex>::mnaAddCompactRightSideVectorStamp(rightVector);
556 if (mUseSubcomponents || !mWithConnectionTransformer)
557 return;
558
559 // The direct filter realization keeps only its external voltage source and
560 // connection transformer as MNA children. Unlike the legacy pre-step path,
561 // the direct parent does not fold those child histories into its own vector.
562 const Matrix &transformerStamp = **mConnectionTransformer->mRightVector;
563 const Matrix &sourceStamp = **mSubCtrledVoltageSource->mRightVector;
564 for (UInt index : mDirectChildRightVectorIndices) {
565 rightVector(index, 0) += transformerStamp(index, 0);
566 rightVector(index, 0) += sourceStamp(index, 0);
567 }
568}

626void DP::Ph1::AvVoltageSourceInverterDQ::mnaCompUpdateCurrent(
627 const Matrix &leftvector) {
628 if (!mUseSubcomponents) {
629 const Complex sourceVoltage = Math::complexFromVectorElement(
630 leftvector, mVirtualNodes[0]->matrixNodeIndex());
631 const Complex resistorVoltage = Math::complexFromVectorElement(
632 leftvector, mVirtualNodes[1]->matrixNodeIndex());
633 const Complex capacitorVoltage = Math::complexFromVectorElement(
634 leftvector, mVirtualNodes[2]->matrixNodeIndex());
635 const Complex terminalVoltage =
636 terminalNotGrounded(0)
637 ? Math::complexFromVectorElement(leftvector, matrixNodeIndex(0))
638 : Complex(0., 0.);
639 const Complex inductorVoltage = capacitorVoltage - resistorVoltage;
640 const Complex inductorEquivalent =
641 mDirectInductorConductance * mDirectInductorVoltage +
642 mDirectInductorPreviousCurrentFactor * mDirectInductorCurrent;
643 const Complex capacitorEquivalent =
644 -mDirectCapacitorCurrent -
645 mDirectCapacitorPreviousVoltageFactor * mDirectCapacitorVoltage;
646 mDirectInductorCurrent =
647 mDirectInductorConductance * inductorVoltage + inductorEquivalent;
648 mDirectInductorVoltage = inductorVoltage;
649 const Complex capacitorComponentVoltage = -capacitorVoltage;
650 mDirectCapacitorCurrent =
651 mDirectCapacitorConductance * capacitorComponentVoltage +
652 capacitorEquivalent;
653 mDirectCapacitorVoltage = capacitorComponentVoltage;
654 const Complex filterOutputVoltage =
655 mWithConnectionTransformer
656 ? Math::complexFromVectorElement(
657 leftvector, mVirtualNodes[3]->matrixNodeIndex())
658 : terminalVoltage;
659 mDirectGridCurrent = (filterOutputVoltage - capacitorVoltage) / mRc;
660 (**mIntfVoltage)(0, 0) = terminalVoltage;
661 (**mIntfCurrent)(0, 0) =
662 mWithConnectionTransformer
663 ? (**mConnectionTransformer->mIntfCurrent)(0, 0)
664 : mDirectGridCurrent;
665 return;
666 }
667 if (mWithConnectionTransformer)
668 **mIntfCurrent = **mConnectionTransformer->mIntfCurrent;
669 else
670 **mIntfCurrent = **mSubResistorC->mIntfCurrent;
671}

673void DP::Ph1::AvVoltageSourceInverterDQ::mnaCompUpdateVoltage(
674 const Matrix &leftVector) {
675 if (!mUseSubcomponents) {
676 for (auto virtualNode : mVirtualNodes)
677 virtualNode->mnaUpdateVoltage(leftVector);
678 (**mIntfVoltage)(0, 0) =
679 terminalNotGrounded(0)
680 ? Math::complexFromVectorElement(leftVector, matrixNodeIndex(0))
681 : Complex(0., 0.);
682 return;
683 }
684 for (auto virtualNode : mVirtualNodes)
685 virtualNode->mnaUpdateVoltage(leftVector);
686 (**mIntfVoltage)(0, 0) =
687 Math::complexFromVectorElement(leftVector, matrixNodeIndex(0));
688}
