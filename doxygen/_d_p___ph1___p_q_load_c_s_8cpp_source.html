58void DP::Ph1::PQLoadCS::createSubComponents() {
59 if (!mUseSubcomponents)
60 return;
61 if (mSubCompCreated)
62 return;
63 mSubCompCreated = true;
64
65 mSubCurrentSource =
66 std::make_shared<DP::Ph1::CurrentSource>(**mName + "_cs", mLogLevel);
67 // A positive power should result in a positive current to ground.
68 mSubCurrentSource->connect({mTerminals[0]->node(), SimNode::GND});
69 addMNASubComponent(mSubCurrentSource,
70 MNA_SUBCOMP_TASK_ORDER::TASK_AFTER_PARENT,
71 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
72}

74void DP::Ph1::PQLoadCS::initializeParentFromNodesAndTerminals(Real frequency) {
75 // Read power from terminals here, not in createSubComponents(): power-flow results aren't available that early.
76 if (**mActivePower == 0 && **mReactivePower == 0 && !mParametersSet) {
77 **mActivePower = mTerminals[0]->singleActivePower();
78 **mReactivePower = mTerminals[0]->singleReactivePower();
79 **mNomVoltage = std::abs(mTerminals[0]->initialSingleVoltage());
80 }
81 Complex power = Complex(**mActivePower, **mReactivePower);
82 Complex current =
83 (**mNomVoltage != 0) ? std::conj(power / **mNomVoltage) : Complex(0);
84 mDirectCurrentRef = current;
85 if (mUseSubcomponents) {
86 mSubCurrentSource->setParameters(current);
87 mSubCurrentSource->initializeFromNodesAndTerminals(frequency);
88 updateIntfValues();
89 } else {
90 (**mIntfVoltage)(0, 0) = initialSingleVoltage(0);
91 (**mIntfCurrent)(0, 0) = mDirectCurrentRef;
92 }
93
94 SPDLOG_LOGGER_INFO(mSLog,
95 "\n--- Initialization from powerflow ---"
96 "\nVoltage across: {:s}"
97 "\nCurrent: {:s}"
98 "\nTerminal 0 voltage: {:s}"
99 "\nCurrent set point: {:s}"
100 "\n--- Initialization from powerflow finished ---",
101 Logger::phasorToString((**mIntfVoltage)(0, 0)),
102 Logger::phasorToString((**mIntfCurrent)(0, 0)),
103 Logger::phasorToString(initialSingleVoltage(0)),
104 Logger::phasorToString(current));
105}

125void DP::Ph1::PQLoadCS::updateSetPoint() {
126 // Calculate new current set point.
127 Complex power = {**mActivePower, **mReactivePower};
128 Complex current = power / **mNomVoltage;
129 //Complex current = power / (**mIntfVoltage)(0,0);
130
131 mDirectCurrentRef = std::conj(current);
132 if (mUseSubcomponents)
133 **mSubCurrentSource->mCurrentRef = mDirectCurrentRef;
134 SPDLOG_LOGGER_DEBUG(mSLog,
135 "\n--- update set points ---"
136 "\npower: {:s}"
137 "\nCurrent: {:s}",
138 Logger::phasorToString(power),
139 Logger::phasorToString(std::conj(current)));
140}

177 Attribute<Matrix>::Ptr &leftVector) {
178 updateIntfValues();
179 if (!mUseSubcomponents) {
