35void DP::Ph1::PiLine::createSubComponents() {
36 if (!mUseSubcomponents)
37 return;
38 if (mSubCompCreated)
39 return;
40 mSubCompCreated = true;
41
42 // By default there is always a small conductance to ground to
43 // avoid problems with floating nodes.
44 Real defaultParallelCond = 1e-6;
45 **mParallelCond =
46 (**mParallelCond > 0) ? **mParallelCond : defaultParallelCond;
47
48 mSubSeriesResistor =
49 std::make_shared<DP::Ph1::Resistor>(**mName + "_res", mLogLevel);
50 mSubSeriesResistor->setParameters(**mSeriesRes);
51 mSubSeriesResistor->connect({mTerminals[0]->node(), mVirtualNodes[0]});
52 addMNASubComponent(mSubSeriesResistor,
53 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
54 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
55
56 mSubSeriesInductor =
57 std::make_shared<DP::Ph1::Inductor>(**mName + "_ind", mLogLevel);
58 mSubSeriesInductor->setParameters(**mSeriesInd);
59 mSubSeriesInductor->connect({mVirtualNodes[0], mTerminals[1]->node()});
60 addMNASubComponent(mSubSeriesInductor,
61 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
62 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
63
64 mSubParallelResistor0 =
65 std::make_shared<DP::Ph1::Resistor>(**mName + "_con0", mLogLevel);
66 mSubParallelResistor0->setParameters(2. / **mParallelCond);
67 mSubParallelResistor0->connect(
68 SimNode::List{SimNode::GND, mTerminals[0]->node()});
69 addMNASubComponent(mSubParallelResistor0,
70 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
71 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
72
73 mSubParallelResistor1 =
74 std::make_shared<DP::Ph1::Resistor>(**mName + "_con1", mLogLevel);
75 mSubParallelResistor1->setParameters(2. / **mParallelCond);
76 mSubParallelResistor1->connect(
77 SimNode::List{SimNode::GND, mTerminals[1]->node()});
78 addMNASubComponent(mSubParallelResistor1,
79 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
80 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
81
82 if (**mParallelCap > 0) {
83 mSubParallelCapacitor0 =
84 std::make_shared<DP::Ph1::Capacitor>(**mName + "_cap0", mLogLevel);
85 mSubParallelCapacitor0->setParameters(**mParallelCap / 2.);
86 mSubParallelCapacitor0->connect(
87 SimNode::List{SimNode::GND, mTerminals[0]->node()});
88 addMNASubComponent(mSubParallelCapacitor0,
89 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
90 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
91
92 mSubParallelCapacitor1 =
93 std::make_shared<DP::Ph1::Capacitor>(**mName + "_cap1", mLogLevel);
94 mSubParallelCapacitor1->setParameters(**mParallelCap / 2.);
95 mSubParallelCapacitor1->connect(
96 SimNode::List{SimNode::GND, mTerminals[1]->node()});
97 addMNASubComponent(mSubParallelCapacitor1,
98 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
99 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
100 }

103void DP::Ph1::PiLine::initializeParentFromNodesAndTerminals(Real frequency) {
104 if (**mParallelCond <= 0.)
105 **mParallelCond = 1e-6;
106 mDirectSeriesConductance = (**mSeriesRes != 0.) ? 1. / **mSeriesRes : 0.;
107 mDirectShuntConductance = **mParallelCond / 2.;
108 mDirectShuntCapacitance = std::max(0., **mParallelCap) / 2.;
109
110 // Static calculation
111 Real omega = 2. * PI * frequency;
112 Complex impedance = {**mSeriesRes, omega * **mSeriesInd};
113 (**mIntfVoltage)(0, 0) = initialSingleVoltage(1) - initialSingleVoltage(0);
114 (**mIntfCurrent)(0, 0) = (**mIntfVoltage)(0, 0) / impedance;
115
116 // Initialization of virtual node
117 if (mUseSubcomponents)
118 mVirtualNodes[0]->setInitialVoltage(initialSingleVoltage(0) +
119 (**mIntfCurrent)(0, 0) * **mSeriesRes);
120
121 SPDLOG_LOGGER_INFO(
122 mSLog,
123 "\n--- Initialization from powerflow ---"
124 "\nVoltage across: {:s}"
125 "\nCurrent: {:s}"
126 "\nTerminal 0 voltage: {:s}"
127 "\nTerminal 1 voltage: {:s}"
128 "\nVirtual Node 1 voltage: {:s}"
129 "\n--- Initialization from powerflow finished ---",
130 Logger::phasorToString((**mIntfVoltage)(0, 0)),
131 Logger::phasorToString((**mIntfCurrent)(0, 0)),
132 Logger::phasorToString(initialSingleVoltage(0)),
133 Logger::phasorToString(initialSingleVoltage(1)),
134 Logger::phasorToString(mUseSubcomponents
135 ? mVirtualNodes[0]->initialSingleVoltage()
136 : Complex(0., 0.)));
137}

139void DP::Ph1::PiLine::mnaParentInitialize(Real omega, Real timeStep,
140 Attribute<Matrix>::Ptr leftVector) {
141 if (mUseSubcomponents)
142 return;
143 mDirectSeriesCond = MatrixComp::Zero(mNumFreqs, 1);
144 mDirectInductorCond = MatrixComp::Zero(mNumFreqs, 1);
145 mDirectSeriesPrevCurrentFactor = MatrixComp::Zero(mNumFreqs, 1);
146 mDirectSeriesCurrent = MatrixComp::Zero(mNumFreqs, 1);
147 mDirectSeriesVoltage = MatrixComp::Zero(mNumFreqs, 1);
148 mDirectSeriesHistory = MatrixComp::Zero(mNumFreqs, 1);
149 mDirectCapCond = MatrixComp::Zero(mNumFreqs, 1);
150 mDirectCapPrevVoltageFactor = MatrixComp::Zero(mNumFreqs, 1);
151 mDirectCapCurrent0 = MatrixComp::Zero(mNumFreqs, 1);
152 mDirectCapCurrent1 = MatrixComp::Zero(mNumFreqs, 1);
153 mDirectCapVoltage0 = MatrixComp::Zero(mNumFreqs, 1);
154 mDirectCapVoltage1 = MatrixComp::Zero(mNumFreqs, 1);
155 mDirectCapHistory0 = MatrixComp::Zero(mNumFreqs, 1);
156 mDirectCapHistory1 = MatrixComp::Zero(mNumFreqs, 1);
157
158 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
159 const Real harmonicOmega = 2. * PI * mFrequencies(freq, 0);
160 const Real b = timeStep * harmonicOmega / 2.;
161 const Real a = timeStep / (2. * **mSeriesInd);
162 mDirectInductorCond(freq, 0) =
163 Complex(a / (1. + b * b), -a * b / (1. + b * b));
164 mDirectSeriesCond(freq, 0) = mDirectInductorCond(freq, 0);
165 mDirectSeriesPrevCurrentFactor(freq, 0) =
166 Complex((1. - b * b) / (1. + b * b), -2. * b / (1. + b * b));
167 const Real capG = 2. * mDirectShuntCapacitance / timeStep;
168 const Real capB = harmonicOmega * mDirectShuntCapacitance;
169 mDirectCapCond(freq, 0) = Complex(capG, capB);
170 mDirectCapPrevVoltageFactor(freq, 0) = Complex(capG, -capB);
171 if (freq == 0) {
172 const Complex voltage1 = initialSingleVoltage(1);
173 const Complex seriesCurrent = (**mIntfCurrent)(0, 0);
174 mDirectSeriesCurrent(freq, 0) = seriesCurrent;
175 mDirectSeriesVoltage(freq, 0) =
176 (**mIntfVoltage)(0, 0) - **mSeriesRes * seriesCurrent;
177 mDirectCapCurrent0(freq, 0) =
178 Complex(0., harmonicOmega * mDirectShuntCapacitance) *
179 initialSingleVoltage(0);
180 mDirectCapCurrent1(freq, 0) =
181 Complex(0., harmonicOmega * mDirectShuntCapacitance) * voltage1;
182 mDirectCapVoltage0(freq, 0) = initialSingleVoltage(0);
183 mDirectCapVoltage1(freq, 0) = voltage1;
184 }
185 }

188void DP::Ph1::PiLine::mnaParentApplySystemMatrixStamp(
189 SparseMatrixRow &systemMatrix) {
190 if (mUseSubcomponents)
191 return;
192 const UInt terminal0 = matrixNodeIndex(0);
193 const UInt terminal1 = matrixNodeIndex(1);
194 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
195 MNAStampUtils::stampAdmittance(
196 Complex(mDirectSeriesConductance, 0.), systemMatrix, terminal0,
197 virtualNode(0)->matrixNodeIndex(), terminalNotGrounded(0), true, mSLog,
198 mNumFreqs, freq);
199 MNAStampUtils::stampAdmittance(mDirectSeriesCond(freq, 0), systemMatrix,
200 virtualNode(0)->matrixNodeIndex(), terminal1,
201 true, terminalNotGrounded(1), mSLog,
202 mNumFreqs, freq);
203 MNAStampUtils::stampAdmittance(
204 Complex(mDirectShuntConductance, 0.) + mDirectCapCond(freq, 0),
205 systemMatrix, terminal0, 0, terminalNotGrounded(0), false, mSLog,
206 mNumFreqs, freq);
207 MNAStampUtils::stampAdmittance(
208 Complex(mDirectShuntConductance, 0.) + mDirectCapCond(freq, 0),
209 systemMatrix, terminal1, 0, terminalNotGrounded(1), false, mSLog,
210 mNumFreqs, freq);
211 }

214void DP::Ph1::PiLine::mnaParentApplyRightSideVectorStamp(Matrix &rightVector) {
215 if (mUseSubcomponents)
216 return;
217 const UInt terminal0 = matrixNodeIndex(0);
218 const UInt terminal1 = matrixNodeIndex(1);
219 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
220 mDirectSeriesHistory(freq, 0) =
221 mDirectInductorCond(freq, 0) * mDirectSeriesVoltage(freq, 0) +
222 mDirectSeriesPrevCurrentFactor(freq, 0) * mDirectSeriesCurrent(freq, 0);
223 mDirectCapHistory0(freq, 0) =
224 -mDirectCapCurrent0(freq, 0) -
225 mDirectCapPrevVoltageFactor(freq, 0) * mDirectCapVoltage0(freq, 0);
226 mDirectCapHistory1(freq, 0) =
227 -mDirectCapCurrent1(freq, 0) -
228 mDirectCapPrevVoltageFactor(freq, 0) * mDirectCapVoltage1(freq, 0);
229 Math::setVectorElement(rightVector, virtualNode(0)->matrixNodeIndex(),
230 mDirectSeriesHistory(freq, 0), mNumFreqs, freq);
231 if (terminalNotGrounded(1))
232 Math::setVectorElement(rightVector, terminal1,
233 -mDirectSeriesHistory(freq, 0) -
234 mDirectCapHistory1(freq, 0),
235 mNumFreqs, freq);
236 }

265void DP::Ph1::PiLine::mnaParentPostStep(Real time, Int timeStepCount,
266 Attribute<Matrix>::Ptr &leftVector) {
267 if (mUseSubcomponents) {
268 this->mnaUpdateVoltage(**leftVector);
269 this->mnaUpdateCurrent(**leftVector);
270 return;
271 }
272
273 const Matrix &solution = **leftVector;
274 const Bool terminal0Connected = terminalNotGrounded(0);
275 const Bool terminal1Connected = terminalNotGrounded(1);
276 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
277 const Complex voltage0 =
278 terminal0Connected ? Math::complexFromVectorElement(
279 solution, matrixNodeIndex(0), mNumFreqs, freq)
280 : Complex(0., 0.);
281 const Complex voltage1 =
282 terminal1Connected ? Math::complexFromVectorElement(
283 solution, matrixNodeIndex(1), mNumFreqs, freq)
284 : Complex(0., 0.);
285 const Complex internalVoltage = Math::complexFromVectorElement(
286 solution, virtualNode(0)->matrixNodeIndex(), mNumFreqs, freq);
287
288 (**mIntfVoltage)(0, freq) = voltage1 - voltage0;
289 mDirectSeriesVoltage(freq, 0) = voltage1 - internalVoltage;
290 mDirectSeriesCurrent(freq, 0) =
291 mDirectSeriesCond(freq, 0) * mDirectSeriesVoltage(freq, 0) +
292 mDirectSeriesHistory(freq, 0);
293 (**mIntfCurrent)(0, freq) = mDirectSeriesCurrent(freq, 0);
294
295 mDirectCapVoltage0(freq, 0) = voltage0;
296 mDirectCapVoltage1(freq, 0) = voltage1;
297 mDirectCapCurrent0(freq, 0) =
298 mDirectCapCond(freq, 0) * voltage0 + mDirectCapHistory0(freq, 0);
299 mDirectCapCurrent1(freq, 0) =
300 mDirectCapCond(freq, 0) * voltage1 + mDirectCapHistory1(freq, 0);
301 }
302}

304void DP::Ph1::PiLine::mnaCompUpdateVoltage(const Matrix &leftVector) {
305 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
306 (**mIntfVoltage)(0, freq) = 0.;
307 if (terminalNotGrounded(1))
308 (**mIntfVoltage)(0, freq) = Math::complexFromVectorElement(
309 leftVector, matrixNodeIndex(1), mNumFreqs, freq);
310 if (terminalNotGrounded(0))
311 (**mIntfVoltage)(0, freq) -= Math::complexFromVectorElement(
312 leftVector, matrixNodeIndex(0), mNumFreqs, freq);
313 }
314}

316void DP::Ph1::PiLine::mnaCompUpdateCurrent(const Matrix &leftVector) {
317 if (mUseSubcomponents) {
318 (**mIntfCurrent)(0, 0) = mSubSeriesInductor->intfCurrent()(0, 0);
319 return;
320 }
321 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
322 const Complex internalVoltage = Math::complexFromVectorElement(
323 leftVector, virtualNode(0)->matrixNodeIndex(), mNumFreqs, freq);
324 const Complex terminalVoltage1 =
325 terminalNotGrounded(1)
326 ? Math::complexFromVectorElement(leftVector, matrixNodeIndex(1),
327 mNumFreqs, freq)
328 : Complex(0., 0.);
329 mDirectSeriesVoltage(freq, 0) = terminalVoltage1 - internalVoltage;
330 mDirectSeriesCurrent(freq, 0) =
331 mDirectSeriesCond(freq, 0) * mDirectSeriesVoltage(freq, 0) +
332 mDirectSeriesHistory(freq, 0);
333 (**mIntfCurrent)(0, freq) = mDirectSeriesCurrent(freq, 0);
334 const Complex voltage0 =
335 terminalNotGrounded(0)
336 ? Math::complexFromVectorElement(leftVector, matrixNodeIndex(0),
337 mNumFreqs, freq)
338 : Complex(0., 0.);
339 const Complex voltageTerminal1 =
340 terminalNotGrounded(1)
341 ? Math::complexFromVectorElement(leftVector, matrixNodeIndex(1),
342 mNumFreqs, freq)
343 : Complex(0., 0.);
344 mDirectCapVoltage0(freq, 0) = voltage0;
345 mDirectCapVoltage1(freq, 0) = voltageTerminal1;
346 mDirectCapCurrent0(freq, 0) =
347 mDirectCapCond(freq, 0) * voltage0 + mDirectCapHistory0(freq, 0);
348 mDirectCapCurrent1(freq, 0) = mDirectCapCond(freq, 0) * voltageTerminal1 +
349 mDirectCapHistory1(freq, 0);
350 }
