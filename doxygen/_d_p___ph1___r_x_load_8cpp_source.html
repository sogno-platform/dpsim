52void DP::Ph1::RXLoad::initializeParentFromNodesAndTerminals(Real frequency) {
53 if (!mParametersSet) {
54 setParameters(mTerminals[0]->singleActivePower(),
55 mTerminals[0]->singleReactivePower(),
56 std::abs(mTerminals[0]->initialSingleVoltage()));
57 }
58
59 Real omega = 2. * PI * frequency;
60
61 if (**mActivePower != 0) {
62 mResistance = std::pow(**mNomVoltage, 2) / **mActivePower;
63 mDirectConductance = 1. / mResistance;
64 if (mUseSubcomponents) {
65 mSubResistor =
66 std::make_shared<DP::Ph1::Resistor>(**mName + "_res", mLogLevel);
67 mSubResistor->setParameters(mResistance);
68 mSubResistor->connect({SimNode::GND, mTerminals[0]->node()});
69 addMNASubComponent(mSubResistor,
70 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
71 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
72 }
73 } else {
74 mResistance = 0;
75 mDirectConductance = 0.;
76 }
77
78 mReactance = (**mReactivePower != 0)
79 ? std::pow(**mNomVoltage, 2) / **mReactivePower
80 : 0;
81
82 if (mReactance > 0) {
83 mInductance = mReactance / omega;
84 mDirectUseInductor = true;
85 mDirectUseCapacitor = false;
86 if (mUseSubcomponents) {
87 mSubInductor =
88 std::make_shared<DP::Ph1::Inductor>(**mName + "_ind", mLogLevel);
89 mSubInductor->setParameters(mInductance);
90 mSubInductor->connect({SimNode::GND, mTerminals[0]->node()});
91 addMNASubComponent(mSubInductor,
92 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
93 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
94 }
95 } else if (mReactance < 0) {
96 mCapacitance = -1. / omega / mReactance;
97 mDirectUseInductor = false;
98 mDirectUseCapacitor = true;
99 mDirectCapacitance = mCapacitance;
100 if (mUseSubcomponents) {
101 mSubCapacitor =
102 std::make_shared<DP::Ph1::Capacitor>(**mName + "_cap", mLogLevel);
103 mSubCapacitor->setParameters(mCapacitance);
104 mSubCapacitor->connect({SimNode::GND, mTerminals[0]->node()});
105 addMNASubComponent(mSubCapacitor,
106 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
107 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
108 }
109 } else {
110 mDirectUseInductor = false;
111 mDirectUseCapacitor = false;
112 }
113
114 (**mIntfVoltage)(0, 0) = mTerminals[0]->initialSingleVoltage();
115 (**mIntfCurrent)(0, 0) =
116 (std::abs((**mIntfVoltage)(0, 0)) > 0)
117 ? std::conj(Complex(**mActivePower, **mReactivePower) /
118 (**mIntfVoltage)(0, 0))
119 : Complex(0, 0);
120 if (!mUseSubcomponents && std::abs((**mIntfVoltage)(0, 0)) > 0.) {
121 if (mDirectUseInductor)
122 mDirectBranchCurrent = MatrixComp::Zero(mNumFreqs, 1);
123 else if (mDirectUseCapacitor)
124 mDirectBranchCurrent = MatrixComp::Zero(mNumFreqs, 1);
125 }
126
127 SPDLOG_LOGGER_INFO(mSLog,
128 "\n--- Initialization from powerflow ---"
129 "\nVoltage across: {:s}"
130 "\nCurrent: {:s}"
131 "\nTerminal 0 voltage: {:s}"
132 "\nResistance: {:f}"
133 "\nReactance: {:f}"
134 "\n--- Initialization from powerflow finished ---",
135 Logger::phasorToString((**mIntfVoltage)(0, 0)),
136 Logger::phasorToString((**mIntfCurrent)(0, 0)),
137 Logger::phasorToString(initialSingleVoltage(0)),
138 mResistance, mReactance);
139}

141void DP::Ph1::RXLoad::mnaParentInitialize(Real omega, Real timeStep,
142 Attribute<Matrix>::Ptr leftVector) {
143 if (mUseSubcomponents)
144 return;
145 mDirectEquivCond = MatrixComp::Zero(mNumFreqs, 1);
146 mDirectHistory = MatrixComp::Zero(mNumFreqs, 1);
147 mDirectPrevCurrentFactor = MatrixComp::Zero(mNumFreqs, 1);
148 mDirectPrevVoltageFactor = MatrixComp::Zero(mNumFreqs, 1);
149 mDirectBranchCurrent = MatrixComp::Zero(mNumFreqs, 1);
150 if (!mDirectUseInductor && !mDirectUseCapacitor)
151 return;
152
153 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
154 const Real harmonicOmega = 2. * PI * mFrequencies(freq, 0);
155 if (mDirectUseInductor) {
156 const Real b = timeStep * harmonicOmega / 2.;
157 const Real a = timeStep / (2. * mInductance);
158 mDirectEquivCond(freq, 0) =
159 Complex(a / (1. + b * b), -a * b / (1. + b * b));
160 mDirectPrevCurrentFactor(freq, 0) =
161 Complex((1. - b * b) / (1. + b * b), -2. * b / (1. + b * b));
162 if (freq == 0 && harmonicOmega != 0.)
163 mDirectBranchCurrent(freq, 0) =
164 (**mIntfVoltage)(0, 0) / Complex(0., harmonicOmega * mInductance);
165 } else {
166 const Real capConductance = 2. * mDirectCapacitance / timeStep;
167 const Real capSusceptance = harmonicOmega * mDirectCapacitance;
168 mDirectEquivCond(freq, 0) = Complex(capConductance, capSusceptance);
169 mDirectPrevVoltageFactor(freq, 0) =
170 Complex(capConductance, -capSusceptance);
171 if (freq == 0)
172 mDirectBranchCurrent(freq, 0) =
173 Complex(0., harmonicOmega * mDirectCapacitance) *
174 (**mIntfVoltage)(0, 0);
175 }
176 }
177}

179void DP::Ph1::RXLoad::mnaParentApplySystemMatrixStamp(
180 SparseMatrixRow &systemMatrix) {
181 if (mUseSubcomponents)
182 return;
183 for (UInt freq = 0; freq < mNumFreqs; ++freq)
184 MNAStampUtils::stampAdmittance(
185 Complex(mDirectConductance, 0.) + mDirectEquivCond(freq, 0),
186 systemMatrix, matrixNodeIndex(0), 0, terminalNotGrounded(0), false,
187 mSLog, mNumFreqs, freq);
188}

190void DP::Ph1::RXLoad::mnaParentApplyRightSideVectorStamp(Matrix &rightVector) {
191 if (mUseSubcomponents || (!mDirectUseInductor && !mDirectUseCapacitor))
192 return;
193 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
194 if (mDirectUseInductor)
195 mDirectHistory(freq, 0) =
196 mDirectEquivCond(freq, 0) * (**mIntfVoltage)(0, freq) +
197 mDirectPrevCurrentFactor(freq, 0) * mDirectBranchCurrent(freq, 0);
198 else
199 mDirectHistory(freq, 0) =
200 -mDirectBranchCurrent(freq, 0) -
201 mDirectPrevVoltageFactor(freq, 0) * (**mIntfVoltage)(0, freq);
202 if (terminalNotGrounded(0))
203 Math::setVectorElement(rightVector, matrixNodeIndex(0),
204 -mDirectHistory(freq, 0), mNumFreqs, freq);
205 }

226void DP::Ph1::RXLoad::mnaCompUpdateCurrent(const Matrix &leftVector) {
227 if (!mUseSubcomponents) {
228 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
229 const Complex voltage = (**mIntfVoltage)(0, freq);
230 const Complex branchCurrent =
231 mDirectEquivCond(freq, 0) * voltage + mDirectHistory(freq, 0);
232 if (mDirectUseInductor || mDirectUseCapacitor)
233 mDirectBranchCurrent(freq, 0) = branchCurrent;
234 (**mIntfCurrent)(0, freq) =
235 Complex(mDirectConductance, 0.) * voltage +
236 ((mDirectUseInductor || mDirectUseCapacitor) ? branchCurrent
237 : Complex(0., 0.));
238 }
239 return;
240 }
241 (**mIntfCurrent)(0, 0) = 0;
242
243 for (auto subComp : mSubComponents) {
244 (**mIntfCurrent)(0, 0) += subComp->intfCurrent()(0, 0);
245 }
246}
