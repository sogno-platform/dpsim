34void DP::Ph1::SVC::createSubComponents() {
35 if (!mUseSubcomponents)
36 return;
37 if (mSubCompCreated)
38 return;
39 mSubCompCreated = true;
40
41 // Inductor/capacitor values depend on omega computed from the simulation
42 // frequency in initializeParentFromNodesAndTerminals(). They are created
43 // here (topology) and parametrized there (values).
44
45 // Inductor with Switch
46 mSubInductor =
47 std::make_shared<DP::Ph1::Inductor>(**mName + "_ind", mLogLevel);
48 mSubInductor->connect({SimNode::GND, mVirtualNodes[0]});
49 addMNASubComponent(mSubInductor, MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,

53 std::make_shared<DP::Ph1::Switch>(**mName + "_Lswitch", mLogLevel);
54 mSubInductorSwitch->setParameters(mSwitchROpen, mSwitchRClosed, false);
55 mSubInductorSwitch->connect({mVirtualNodes[0], mTerminals[0]->node()});
56 addMNASubComponent(mSubInductorSwitch,
57 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
58 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
59
60 // Capacitor with Switch
61 mSubCapacitor =
62 std::make_shared<DP::Ph1::Capacitor>(**mName + "_cap", mLogLevel);
63 mSubCapacitor->connect({SimNode::GND, mVirtualNodes[1]});
64 addMNASubComponent(mSubCapacitor, MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
65 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
66
67 mSubCapacitorSwitch =
68 std::make_shared<DP::Ph1::Switch>(**mName + "_Cswitch", mLogLevel);
69 mSubCapacitorSwitch->setParameters(mSwitchROpen, mSwitchRClosed, false);
70 mSubCapacitorSwitch->connect({mVirtualNodes[1], mTerminals[0]->node()});
71 addMNASubComponent(mSubCapacitorSwitch,
72 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
73 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
74}

76void DP::Ph1::SVC::initializeParentFromNodesAndTerminals(Real frequency) {
77 // initial state is both switches are open
78 Real omega = 2. * PI * frequency;
79 // init L and C with small/high values (both have high impedance)
80 Real LInit = 1e6 / omega;
81 Real CInit = 1e-6 / omega;
82 mLPrev = LInit;
83 mCPrev = CInit;
84
85 // impedances of both branches
86 Complex LImpedance = {mSwitchROpen, omega * LInit};
87 Complex CImpedance = {mSwitchROpen, -1 / (omega * CInit)};
88 Complex impedance = LImpedance * CImpedance / (LImpedance + CImpedance);
89
90 (**mIntfVoltage)(0, 0) = initialSingleVoltage(0);
91 (**mIntfCurrent)(0, 0) = (**mIntfVoltage)(0, 0) / impedance;
92
93 **mBPrev = 0;
94 mPrevVoltage = (**mIntfVoltage)(0, 0).real();
95 **mVmeasPrev = mPrevVoltage;
96
97 if (mMechMode) {
98 SPDLOG_LOGGER_INFO(mSLog, "Using Mechanical Model");
99 }
100
101 SPDLOG_LOGGER_INFO(mSLog,
102 "\n --- Parameters ---"
103 "\n Controller: T = {} K = {}"
104 "\n Reference Voltage {} [kV]"
105 "\n Qmax = {} [var] -> BN = {} [S]"
106 "\n Bmax = {} Bmin = {} [p.u.]"
107 "\n Initial B: {}",
108 mTr, mKr, mRefVolt, mQN, mBN, mBMax, mBMin, **mBPrev);
109
110 // set voltages at virtual nodes
111 Complex VLSwitch =
112 (**mIntfVoltage)(0, 0) - LImpedance * (**mIntfCurrent)(0, 0);
113 mVirtualNodes[0]->setInitialVoltage(VLSwitch);
114 Complex VCSwitch =
115 (**mIntfVoltage)(0, 0) - CImpedance * (**mIntfCurrent)(0, 0);
116 mVirtualNodes[1]->setInitialVoltage(VCSwitch);
117
118 if (mUseSubcomponents) {
119 mSubInductor->setParameters(LInit);
120 mSubCapacitor->setParameters(CInit);
121 } else {
122 mDirectInductance = LInit;
123 mDirectCapacitance = CInit;
124 }
125
126 SPDLOG_LOGGER_INFO(mSLog,
127 "\n--- Initialization from powerflow ---"
128 "\nImpedance: {}"
129 "\nVoltage across: {:s}"
130 "\nCurrent: {:s}"
131 "\nTerminal 0 voltage: {:s}"
132 "\n--- Initialization from powerflow finished ---",
133 impedance, Logger::phasorToString((**mIntfVoltage)(0, 0)),
134 Logger::phasorToString((**mIntfCurrent)(0, 0)),
135 Logger::phasorToString(initialSingleVoltage(0)));
136}

138void DP::Ph1::SVC::mnaParentInitialize(Real omega, Real timeStep,
139 Attribute<Matrix>::Ptr leftVector) {
140 if (mUseSubcomponents)
141 return;
142 mDirectTimeStep = timeStep;
143 mDirectInductorAdmittance = MatrixComp::Zero(mNumFreqs, 1);
144 mDirectInductorPrevCurrentFactor = MatrixComp::Zero(mNumFreqs, 1);
145 mDirectInductorCurrent = MatrixComp::Zero(mNumFreqs, 1);
146 mDirectInductorVoltage = MatrixComp::Zero(mNumFreqs, 1);
147 mDirectInductorHistory = MatrixComp::Zero(mNumFreqs, 1);
148 mDirectCapacitorAdmittance = MatrixComp::Zero(mNumFreqs, 1);
149 mDirectCapacitorPrevVoltageFactor = MatrixComp::Zero(mNumFreqs, 1);
150 mDirectCapacitorCurrent = MatrixComp::Zero(mNumFreqs, 1);
151 mDirectCapacitorVoltage = MatrixComp::Zero(mNumFreqs, 1);
152 mDirectCapacitorHistory = MatrixComp::Zero(mNumFreqs, 1);
153 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
154 const Real harmonicOmega = 2. * PI * mFrequencies(freq, 0);
155 mDirectInductorVoltage(freq, 0) = mVirtualNodes[0]->initialSingleVoltage();
156 mDirectCapacitorVoltage(freq, 0) = mVirtualNodes[1]->initialSingleVoltage();
157 mDirectInductorCurrent(freq, 0) =
158 harmonicOmega != 0. ? mDirectInductorVoltage(freq, 0) /
159 Complex(0., harmonicOmega * mDirectInductance)
160 : Complex(0., 0.);
161 mDirectCapacitorCurrent(freq, 0) =
162 Complex(0., harmonicOmega * mDirectCapacitance) *
163 mDirectCapacitorVoltage(freq, 0);
164 }
165 updateDirectCompanions();
166 // Match the state transition performed by DP::Inductor::mnaCompInitialize
167 // and DP::Capacitor::mnaCompInitialize. Those components convert the
168 // sinusoidal steady-state initial currents into their trapezoidal companion
169 // state before the first pre-step; retaining only the physical phasor here
170 // produces a different first history source in direct mode.
171 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
172 mDirectInductorHistory(freq, 0) =
173 mDirectInductorAdmittance(freq, 0) * mDirectInductorVoltage(freq, 0) +
174 mDirectInductorPrevCurrentFactor(freq, 0) *
175 mDirectInductorCurrent(freq, 0);
176 mDirectInductorCurrent(freq, 0) =
177 mDirectInductorAdmittance(freq, 0) * mDirectInductorVoltage(freq, 0) +
178 mDirectInductorHistory(freq, 0);
179
180 mDirectCapacitorHistory(freq, 0) =
181 -mDirectCapacitorCurrent(freq, 0) -
182 mDirectCapacitorPrevVoltageFactor(freq, 0) *
183 mDirectCapacitorVoltage(freq, 0);
184 mDirectCapacitorCurrent(freq, 0) =
185 mDirectCapacitorAdmittance(freq, 0) * mDirectCapacitorVoltage(freq, 0) +
186 mDirectCapacitorHistory(freq, 0);
187 }
188}

190void DP::Ph1::SVC::updateDirectCompanions() {
191 if (mUseSubcomponents || mDirectTimeStep <= 0.)
192 return;
193 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
194 const Real harmonicOmega = 2. * PI * mFrequencies(freq, 0);
195 const Real b = mDirectTimeStep * harmonicOmega / 2.;
196 const Real a = mDirectTimeStep / (2. * mDirectInductance);
197 mDirectInductorAdmittance(freq, 0) =
198 Complex(a / (1. + b * b), -a * b / (1. + b * b));
199 mDirectInductorPrevCurrentFactor(freq, 0) =
200 Complex((1. - b * b) / (1. + b * b), -2. * b / (1. + b * b));
201 const Real capG = 2. * mDirectCapacitance / mDirectTimeStep;
202 const Real capB = harmonicOmega * mDirectCapacitance;
203 mDirectCapacitorAdmittance(freq, 0) = Complex(capG, capB);
204 mDirectCapacitorPrevVoltageFactor(freq, 0) = Complex(capG, -capB);
205 }
206}

208void DP::Ph1::SVC::mnaParentApplySystemMatrixStamp(
209 SparseMatrixRow &systemMatrix) {
210 if (mUseSubcomponents)
211 return;
212 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
213 MNAStampUtils::stampAdmittance(
214 mDirectInductorAdmittance(freq, 0), systemMatrix,
215 SimNode::GND->matrixNodeIndex(), virtualNode(0)->matrixNodeIndex(),
216 false, true, mSLog, mNumFreqs, freq);
217 MNAStampUtils::stampAdmittance(
218 mDirectCapacitorAdmittance(freq, 0), systemMatrix,
219 SimNode::GND->matrixNodeIndex(), virtualNode(1)->matrixNodeIndex(),
220 false, true, mSLog, mNumFreqs, freq);
221 const Real gInductor =
222 mDirectInductorClosed ? 1. / mSwitchRClosed : 1. / mSwitchROpen;
223 const Real gCapacitor =
224 mDirectCapacitorClosed ? 1. / mSwitchRClosed : 1. / mSwitchROpen;
225 MNAStampUtils::stampAdmittance(
226 Complex(gInductor, 0.), systemMatrix, matrixNodeIndex(0),
227 virtualNode(0)->matrixNodeIndex(), terminalNotGrounded(0), true, mSLog,
228 mNumFreqs, freq);
229 MNAStampUtils::stampAdmittance(
230 Complex(gCapacitor, 0.), systemMatrix, matrixNodeIndex(0),
231 virtualNode(1)->matrixNodeIndex(), terminalNotGrounded(0), true, mSLog,
232 mNumFreqs, freq);
233 }

236void DP::Ph1::SVC::mnaParentApplyRightSideVectorStamp(Matrix &rightVector) {
237 if (mUseSubcomponents)
238 return;
239 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
240 mDirectInductorHistory(freq, 0) =
241 mDirectInductorAdmittance(freq, 0) * mDirectInductorVoltage(freq, 0) +
242 mDirectInductorPrevCurrentFactor(freq, 0) *
243 mDirectInductorCurrent(freq, 0);
244 mDirectCapacitorHistory(freq, 0) =
245 -mDirectCapacitorCurrent(freq, 0) -
246 mDirectCapacitorPrevVoltageFactor(freq, 0) *
247 mDirectCapacitorVoltage(freq, 0);
248 Math::setVectorElement(rightVector, virtualNode(0)->matrixNodeIndex(),
249 -mDirectInductorHistory(freq, 0), mNumFreqs, freq);
250 Math::setVectorElement(rightVector, virtualNode(1)->matrixNodeIndex(),
251 -mDirectCapacitorHistory(freq, 0), mNumFreqs, freq);
252 }

264void DP::Ph1::SVC::mnaParentPreStep(Real time, Int timeStepCount) {
265 if (mUseSubcomponents)
266 mnaCompApplyRightSideVectorStamp(**mRightVector);
267
268 if (time > 0.1 && !mDisconnect) {
269 if (mMechMode) {
270 mechanicalModelUpdateSusceptance(time);
271 } else {
272 updateSusceptance();
273 }
274 checkProtection(time);
275 }
276 if (!mUseSubcomponents) {
277 updateDirectCompanions();
278 mnaCompApplyRightSideVectorStamp(**mRightVector);
279 }
280}

316void DP::Ph1::SVC::mnaCompUpdateCurrent(const Matrix &leftVector) {
317 if (!mUseSubcomponents) {
318 mDirectInductorCurrent(0, 0) =
319 mDirectInductorAdmittance(0, 0) * mDirectInductorVoltage(0, 0) +
320 mDirectInductorHistory(0, 0);
321 mDirectCapacitorCurrent(0, 0) =
322 mDirectCapacitorAdmittance(0, 0) * mDirectCapacitorVoltage(0, 0) +
323 mDirectCapacitorHistory(0, 0);
324 (**mIntfCurrent)(0, 0) =
325 mDirectInductorCurrent(0, 0) + mDirectCapacitorCurrent(0, 0);
326 return;
327 }
328 (**mIntfCurrent)(0, 0) = 0;
329 (**mIntfCurrent)(0, 0) += mSubInductor->intfCurrent()(0, 0);
330 (**mIntfCurrent)(0, 0) += mSubCapacitor->intfCurrent()(0, 0);
331}

333void DP::Ph1::SVC::checkProtection(Real time) {
334 // check states for violation of protection values
335 // get inverse protection curve value (time delay value)
336
337 Real Vpu = **mVmeasPrev / mNomVolt;
338 if (Vpu > 1.4) {
339 mProtCount1 = mProtCount1 + mDeltaT;
340 if (mProtCount1 > 0.1) {
341 mDisconnect = true;
342 }
343 } else {
344 mProtCount1 = 0;
345 }
346 if (Vpu > 1.25) {
347 mProtCount2 = mProtCount2 + mDeltaT;
348 if (mProtCount2 > 1) {
349 mDisconnect = true;
350 }
351 } else {
352 mProtCount2 = 0;
353 }
354 if (Vpu > 1.15) {
355 mProtCount3 = mProtCount3 + mDeltaT;
356 if (mProtCount3 > 5) {
357 mDisconnect = true;
358 }
359 } else {
360 mProtCount3 = 0;
361 }
362
363 if (mDisconnect) {
364 SPDLOG_LOGGER_INFO(mSLog, "Disconnect SVC because of overvoltage at {}",
365 time);
366 if (mUseSubcomponents) {
367 mSubCapacitorSwitch->open();
368 mSubInductorSwitch->open();
369 } else {
370 mDirectCapacitorClosed = false;
371 mDirectInductorClosed = false;
372 }
373 mValueChange = true;
374 }

377void DP::Ph1::SVC::updateSusceptance() {
378 // calculate new B value
379 // summarize some constants
380 Real Fac1 = mDeltaT / (2 * mTr);
381 Real Fac2 = mDeltaT * mKr / (2 * mTr);
382
383 Real V = Math::abs((**mIntfVoltage)(0, 0).real());
384
385 // Pt1 with trapez rule for voltage measurement
386 Real Fac3 = mDeltaT / (2 * mTm);
387 Real Vmeas = (1 / (1 + Fac3)) * (V + mPrevVoltage - **mVmeasPrev);
388
389 **mDeltaV = (Vmeas - mRefVolt) / mNomVolt;
390 Real deltaVPrev = (**mVmeasPrev - mRefVolt) / mNomVolt;
391
392 // calc new B with trapezoidal rule
393 Real B = (1 / (1 + Fac1)) *
394 (Fac2 * (**mDeltaV + deltaVPrev) + (1 - Fac1) * **mBPrev);
395
396 // check bounds
397 if (B > mBMax) {
398 B = mBMax;
399 } else if (B < mBMin) {
400 B = mBMin;
401 }
402
403 // set new B if it has a new value and difference is big enough
404 if (B != **mBPrev) {
405 Real omega = 2 * M_PI * mFrequencies(0, 0);
406
407 if (B > 0) {
408 // model inductive behaviour (decrease voltage)
409 Real inductance = 1 / (omega * B * mBN);
410 //check if change in reactance is sufficient to trigger a change
411 if (Math::abs(1 - inductance / mLPrev) > 0.01) {
412 mInductiveMode = true;
413 if (mUseSubcomponents)
414 mSubInductor->updateInductance(inductance, mDeltaT);
415 else
416 mDirectInductance = inductance;
417 mLPrev = inductance;
418
419 mValueChange = true;
420 mBSetCounter = 0;
421 }
422 } else {
423 // model capacitive behaviour (increase voltage)
424 Real capacitance = B * mBN / (-omega);
425 //check if change in reactance is sufficient to trigger a change
426 if (Math::abs(1 - capacitance / mCPrev) > 0.01) {
427 mInductiveMode = false;
428 if (mUseSubcomponents)
429 mSubCapacitor->updateCapacitance(capacitance, mDeltaT);
430 else
431 mDirectCapacitance = capacitance;
432 mCPrev = capacitance;
433
434 mValueChange = true;
435 mBSetCounter = 0;
436 }
437 }
438
439 // update inductance model
440 setSwitchState();
441 } else {
442 mBSetCounter = mBSetCounter + mDeltaT;
443 }
444
445 // save values
446 **mBPrev = B;
447 mPrevVoltage = V;
448 **mVmeasPrev = Vmeas;
449}

451// model SVC with a mechanical component and discrete
452void DP::Ph1::SVC::mechanicalModelUpdateSusceptance(Real time) {
453 // current voltage
454 Real V = Math::abs((**mIntfVoltage)(0, 0).real());
455 Real omega = 2 * M_PI * mFrequencies(0, 0);
456
457 // Pt1 with trapez rule for voltage measurement
458 Real Fac3 = mDeltaT / (2 * mTm);
459 Real Vmeas = (1 / (1 + Fac3)) * (V + mPrevVoltage - **mVmeasPrev);
460
461 // V diff in pu
462 Real deltaV = (mRefVolt - Vmeas) / mRefVolt;
463
464 if (Math::abs(deltaV) > mDeadband) {
465 if (**mViolationCounter > mMechSwitchDelay) {
466 // change suszeptance one step
467 if (deltaV > 0 && (mTapPos > mMinPos)) {
468 // undervoltage
469
470 mTapPos = mTapPos - 1;
471 mTapPos = (mTapPos < mMinPos) ? mMinPos : mTapPos;
472 **mViolationCounter = 0;
473 SPDLOG_LOGGER_INFO(mSLog,
474 "Time: {}"
475 "\nDecreasing Tap. Reason: Undervoltage"
476 "\nNew Tap Position: {}",
477 time, mTapPos);
478 } else if (deltaV < 0 && (mTapPos < mMaxPos)) {
479 // overvoltage
480 mTapPos = mTapPos + 1;
481 mTapPos = (mTapPos > mMaxPos) ? mMaxPos : mTapPos;
482 **mViolationCounter = 0;
483 SPDLOG_LOGGER_INFO(mSLog,
484 "Time: {}"
485 "\nIncreasing Tap. Reason: Overvoltag"
486 "\nNew Tap Position: {}",
487 time, mTapPos);
488 }
489
490 if (**mViolationCounter == 0) {
491 // new value for suszeptance
492 if (mTapPos > 0) {
493 // inductor is active
494 mInductiveMode = true;
495 Real inductance = 1 / ((mTapPos / mMaxPos) * mBN * omega);
496 SPDLOG_LOGGER_INFO(mSLog, "New inductance: {}", inductance);
497 if (mUseSubcomponents)
498 mSubInductor->updateInductance(inductance, mDeltaT);
499 else
500 mDirectInductance = inductance;
501 mValueChange = true;
502 setSwitchState();
503 } else if (mTapPos < 0) {
504 // capacitor is active
505 mInductiveMode = false;
506 Real capacitance = ((mTapPos / mMinPos) * mBN) / omega;
507 SPDLOG_LOGGER_INFO(mSLog, "New capacitance: {}", capacitance);
508 if (mUseSubcomponents)
509 mSubCapacitor->updateCapacitance(capacitance, mDeltaT);
510 else
511 mDirectCapacitance = capacitance;
512 mValueChange = true;
513 setSwitchState();
514
515 } else if (mTapPos = 0) {
516 // open both
517 SPDLOG_LOGGER_INFO(mSLog,
518 "Time: {}"
519 "Tap Position: 0. Open both elements",
520 time);
521 if (mUseSubcomponents) {
522 mSubInductorSwitch->open();
523 mSubCapacitorSwitch->open();
524 } else {
525 mDirectInductorClosed = false;
526 mDirectCapacitorClosed = false;
527 mValueChange = true;
528 }
529 }
530 }
531 } else {
532 // increase counter
533 **mViolationCounter = **mViolationCounter + mDeltaT;
534 }
535 } else {
536 // reset counter
537 **mViolationCounter = 0;
538 }
539
540 // save states
541 mPrevVoltage = V;
542 **mVmeasPrev = Vmeas;
543}

545void DP::Ph1::SVC::setSwitchState() {
546 if (!mUseSubcomponents) {
547 const Bool newInductorClosed = mInductiveMode;
548 const Bool newCapacitorClosed = !mInductiveMode;
549 if (mDirectInductorClosed != newInductorClosed ||
550 mDirectCapacitorClosed != newCapacitorClosed)
551 mValueChange = true;
552 mDirectInductorClosed = newInductorClosed;
553 mDirectCapacitorClosed = newCapacitorClosed;
554 return;
555 }
556 // set switches according to current mode of svc
557 if (mInductiveMode) {
558 if (!mSubInductorSwitch->mnaIsClosed()) {
559 SPDLOG_LOGGER_INFO(mSLog, "Inductive Mode: Closed Inductor Switch");
560 mSubInductorSwitch->close();
561 }
562 if (mSubCapacitorSwitch->mnaIsClosed()) {
563 mSubCapacitorSwitch->open();
564 SPDLOG_LOGGER_INFO(mSLog, "Inductive Mode: Opened Capacitor Switch");
565 }
566 } else {
567 if (mSubInductorSwitch->mnaIsClosed()) {
568 mSubInductorSwitch->open();
569 SPDLOG_LOGGER_INFO(mSLog, "Capacitive Mode: Openend Inductor Switch");
570 }
571 if (!mSubCapacitorSwitch->mnaIsClosed()) {
572 mSubCapacitorSwitch->close();
573 SPDLOG_LOGGER_INFO(mSLog, "Capacitive Mode: Closed Capcitor Switch");
574 }
575 }
