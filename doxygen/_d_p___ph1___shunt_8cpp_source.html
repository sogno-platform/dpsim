37void DP::Ph1::Shunt::initializeParentFromNodesAndTerminals(Real frequency) {
38
39 // Static calculation
40 Real omega = 2. * PI * frequency;
41 Complex admittance = Complex(**mConductance, **mSusceptance);
42 (**mIntfVoltage)(0, 0) = initialSingleVoltage(0);
43 (**mIntfCurrent)(0, 0) = (**mIntfVoltage)(0, 0) * admittance;
44
45 if (mUseSubcomponents && **mConductance > 0) {
46 mSubResistor =
47 std::make_shared<DP::Ph1::Resistor>(**mName + "_Res", mLogLevel);
48 mSubResistor->connect(SimNode::List{SimNode::GND, mTerminals[0]->node()});
49 mSubResistor->setParameters(1. / **mConductance);
50 mSubResistor->initialize(mFrequencies);
51 mSubResistor->initializeFromNodesAndTerminals(frequency);
52 addMNASubComponent(mSubResistor, MNA_SUBCOMP_TASK_ORDER::NO_TASK,
53 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
54 }
55
56 if (**mSusceptance > 0) {
57 mDirectCapacitance = **mSusceptance / omega;
58 if (mUseSubcomponents) {
59 mSubCapacitor =
60 std::make_shared<DP::Ph1::Capacitor>(**mName + "_cap", mLogLevel);
61 mSubCapacitor->setParameters(mDirectCapacitance);
62 mSubCapacitor->connect(
63 SimNode::List{SimNode::GND, mTerminals[0]->node()});
64 mSubCapacitor->initialize(mFrequencies);
65 mSubCapacitor->initializeFromNodesAndTerminals(frequency);
66 addMNASubComponent(mSubCapacitor,
67 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
68 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
69 }
70 }
71
72 SPDLOG_LOGGER_INFO(mSLog,
73 "\n--- Initialization from powerflow ---"
74 "\nVoltage across: {:s}"
75 "\nCurrent: {:s}"
76 "\nTerminal voltage: {:s}"
77 "\n--- Initialization from powerflow finished ---",
78 Logger::phasorToString((**mIntfVoltage)(0, 0)),
79 Logger::phasorToString((**mIntfCurrent)(0, 0)),
80 Logger::phasorToString(initialSingleVoltage(0)));
81}

83void DP::Ph1::Shunt::mnaParentInitialize(Real omega, Real timeStep,
84 Attribute<Matrix>::Ptr leftVector) {
85 mDirectEquivCond = MatrixComp::Zero(mNumFreqs, 1);
86 mDirectPrevVoltCoeff = MatrixComp::Zero(mNumFreqs, 1);
87 mDirectCapCurrent = MatrixComp::Zero(mNumFreqs, 1);
88 mDirectHistory = MatrixComp::Zero(mNumFreqs, 1);
89 if (mUseSubcomponents || mDirectCapacitance <= 0.)
90 return;
91
92 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
93 const Real capacitiveConductance = 2. * mDirectCapacitance / timeStep;
94 const Real harmonicSusceptance =
95 2. * PI * mFrequencies(freq, 0) * mDirectCapacitance;
96 mDirectEquivCond(freq, 0) =
97 Complex(capacitiveConductance, harmonicSusceptance);
98 mDirectPrevVoltCoeff(freq, 0) =
99 Complex(capacitiveConductance, -harmonicSusceptance);
100 if (freq == 0)
101 mDirectCapCurrent(freq, 0) =
102 Complex(0., 2. * PI * mFrequencies(freq, 0) * mDirectCapacitance) *
103 (**mIntfVoltage)(0, 0);
104 }
105}

107void DP::Ph1::Shunt::mnaParentApplySystemMatrixStamp(
108 SparseMatrixRow &systemMatrix) {
109 if (mUseSubcomponents)
110 return;
111 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
112 const Complex admittance =
113 Complex(**mConductance, 0.) + mDirectEquivCond(freq, 0);
114 MNAStampUtils::stampAdmittance(admittance, systemMatrix, matrixNodeIndex(0),
115 0, terminalNotGrounded(0), false, mSLog,
116 mNumFreqs, freq);
117 }
118}

120void DP::Ph1::Shunt::mnaParentApplyRightSideVectorStamp(Matrix &rightVector) {
121 if (mUseSubcomponents || mDirectCapacitance <= 0.)
122 return;
123 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
124 const Complex history =
125 -mDirectCapCurrent(freq, 0) -
126 mDirectPrevVoltCoeff(freq, 0) * (**mIntfVoltage)(0, freq);
127 mDirectHistory(freq, 0) = history;
128 if (terminalNotGrounded(0))
129 Math::setVectorElement(rightVector, matrixNodeIndex(0), -history,
130 mNumFreqs, freq);
131 }
132}

171void DP::Ph1::Shunt::mnaCompUpdateCurrent(const Matrix &leftVector) {
172 if (!mUseSubcomponents) {
173 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
174 const Complex voltage = (**mIntfVoltage)(0, freq);
175 const Complex capCurrent =
176 mDirectCapacitance > 0.
177 ? mDirectEquivCond(freq, 0) * voltage + mDirectHistory(freq, 0)
178 : Complex(0., 0.);
179 mDirectCapCurrent(freq, 0) = capCurrent;
180 (**mIntfCurrent)(0, freq) =
181 Complex(**mConductance, 0.) * voltage + capCurrent;
182 }
183 return;
184 }
185
186 (**mIntfCurrent)(0, 0) = 0;
187 if (**mConductance > 0)
188 (**mIntfCurrent)(0, 0) += mSubResistor->intfCurrent()(0, 0);
189 if (**mSusceptance > 0)
190 (**mIntfCurrent)(0, 0) += mSubCapacitor->intfCurrent()(0, 0);
191}
