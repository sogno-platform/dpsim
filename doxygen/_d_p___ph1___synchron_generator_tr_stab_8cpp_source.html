43void DP::Ph1::SynchronGeneratorTrStab::setFundamentalParametersPU(
44 Real nomPower, Real nomVolt, Real nomFreq, Real Ll, Real Lmd, Real Llfd,
45 Real inertia, Real D) {
46 setBaseParameters(nomPower, nomVolt, nomFreq);
47 SPDLOG_LOGGER_INFO(mSLog,
48 "\n--- Base Parameters ---"
49 "\nnomPower: {:f}"
50 "\nnomVolt: {:f}"
51 "\nnomFreq: {:f}",
52 mNomPower, mNomVolt, mNomFreq);
53
54 // Input is in per unit but all values are converted to absolute values.
55 mParameterType = ParameterType::statorReferred;
56 mStateType = StateType::statorReferred;
57
58 **mLl = Ll;
59 mLmd = Lmd;
60 **mLd = **mLl + mLmd;
61 mLlfd = Llfd;
62 mLfd = mLlfd + mLmd;
63 // M = 2*H where H = inertia
64 **mInertia = inertia;
65 // X'd in absolute values
66 mXpd = mNomOmega * (**mLd - mLmd * mLmd / mLfd) * mBase_L;
67 mLpd = (**mLd - mLmd * mLmd / mLfd) * mBase_L;
68
69 //The units of D are per unit power divided by per unit speed deviation.
70 // D is transformed to an absolute value to obtain Kd, which will be used in the swing equation
71 mKd = D * mNomPower / mNomOmega;
72
73 SPDLOG_LOGGER_INFO(mSLog,
74 "\n--- Parameters ---"
75 "\nimpedance: {:f}"
76 "\ninductance: {:f}"
77 "\ninertia: {:f}"
78 "\ndamping: {:f}",
79 mXpd, mLpd, **mInertia, mKd);
80}

163void DP::Ph1::SynchronGeneratorTrStab::createSubComponents() {
164 if (!mUseSubcomponents)
165 return;
166 if (mSubCompCreated)
167 return;
168 mSubCompCreated = true;
169
170 // Create the inner voltage source (Ep). Parameters (mEp) depend on the
171 // power-flow solution and are not yet known here; they are applied in
172 // initializeFromNodesAndTerminals after the static calculation.
173 mSubVoltageSource = DP::Ph1::VoltageSource::make(**mName + "_src", mLogLevel);
174 mSubVoltageSource->connect({SimNode::GND, mVirtualNodes[0]});
175 // Alias the sub-source's virtual node to parent VN1 so the KCL row in the
176 // system matrix uses the same node index as the parent's VN1.
177 mSubVoltageSource->setVirtualNodeAt(mVirtualNodes[1], 0);
178 addMNASubComponent(mSubVoltageSource,
179 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
180 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
181
182 // Create the inner inductor representing X'd.
183 mSubInductor = DP::Ph1::Inductor::make(**mName + "_ind", mLogLevel);
184 mSubInductor->setParameters(mLpd);
185 mSubInductor->connect({mVirtualNodes[0], terminal(0)->node()});
186 addMNASubComponent(mSubInductor, MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
187 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
188}

190void DP::Ph1::SynchronGeneratorTrStab::initializeParentFromNodesAndTerminals(
191 Real frequency) {
192 // Initialize omega mech with nominal system frequency
193 **mOmMech = mNomOmega;
194
195 // Static calculation based on load flow
196 (**mIntfVoltage)(0, 0) = initialSingleVoltage(0);
197 mInitElecPower = (mInitElecPower == Complex(0, 0))
198 ? -terminal(0)->singlePower()
199 : mInitElecPower;
200 mInitMechPower = (mInitElecPower == Complex(0, 0)) ? mInitElecPower.real()
201 : mInitMechPower;
202
203 //I_intf is the current which is flowing into the Component, while mInitElecPower is flowing out of it
204 (**mIntfCurrent)(0, 0) = std::conj(-mInitElecPower / (**mIntfVoltage)(0, 0));
205
206 mImpedance = Complex(**mRs, mXpd);
207
208 // Calculate initial emf behind reactance from power flow results
209 **mEp = (**mIntfVoltage)(0, 0) - mImpedance * (**mIntfCurrent)(0, 0);
210
211 // The absolute value of Ep is constant, only delta_p changes every step
212 **mEp_abs = Math::abs(**mEp);
213 // Delta_p is the angular position of mEp with respect to the synchronously rotating reference
214 **mDelta_p = Math::phase(**mEp);
215
216 // Update active electrical power that is compared with the mechanical power
217 **mElecActivePower =
218 ((**mIntfVoltage)(0, 0) * std::conj(-(**mIntfCurrent)(0, 0))).real();
219
220 // Start in steady state so that electrical and mech. power are the same
221 // because of the initial condition mOmMech = mNomOmega the damping factor is not considered at the initialisation
222 **mMechPower = **mElecActivePower;
223
224 // Initialize node between X'd and Ep
225 mVirtualNodes[0]->setInitialVoltage(**mEp);
226
227 // Set emf on the already-created voltage source; the framework's generic
228 // sub-init loop will initialize it after this hook returns.
229 if (mUseSubcomponents)
230 mSubVoltageSource->setParameters(**mEp);
231
232 SPDLOG_LOGGER_INFO(mSLog,
233 "\n--- Initialize according to powerflow ---"
234 "\nTerminal 0 voltage: {:e}<{:e}"
235 "\nVoltage behind reactance: {:e}<{:e}"
236 "\ninitial electrical power: {:e}+j{:e}"
237 "\nactive electrical power: {:e}"
238 "\nmechanical power: {:e}"
239 "\n--- End of powerflow initialization ---",
240 Math::abs((**mIntfVoltage)(0, 0)),
241 Math::phaseDeg((**mIntfVoltage)(0, 0)), Math::abs(**mEp),
242 Math::phaseDeg(**mEp), mInitElecPower.real(),
243 mInitElecPower.imag(), **mElecActivePower, **mMechPower);
244}

246void DP::Ph1::SynchronGeneratorTrStab::step(Real time) {
247
248 // #### Calculations based on values from time step k ####
249 // Electrical power at time step k
250 **mElecActivePower =
251 ((**mIntfVoltage)(0, 0) * std::conj(-(**mIntfCurrent)(0, 0))).real();
252
253 // Mechanical speed derivative at time step k
254 // convert torque to power with actual rotor angular velocity or nominal omega
255 Real dOmMech;
256 if (mConvertWithOmegaMech)
257 dOmMech =
258 mNomOmega * mNomOmega / (2. * **mInertia * mNomPower * **mOmMech) *
259 (**mMechPower - **mElecActivePower - mKd * (**mOmMech - mNomOmega));
260 else
261 dOmMech =
262 mNomOmega / (2. * **mInertia * mNomPower) *
263 (**mMechPower - **mElecActivePower - mKd * (**mOmMech - mNomOmega));
264
265 // #### Calculate states for time step k+1 applying semi-implicit Euler ####
266 // Mechanical speed at time step k+1 applying Euler forward
267 if (mBehaviour == Behaviour::MNASimulation)
268 **mOmMech = **mOmMech + mTimeStep * dOmMech;
269
270 // Derivative of rotor angle at time step k + 1
271 // if reference omega is set, calculate delta with respect to reference
272 Real dDelta_p = **mOmMech - (mUseOmegaRef ? **mRefOmega : mNomOmega);
273
274 // Rotor angle at time step k + 1 applying Euler backward
275 // Update emf - only phase changes
276 if (mBehaviour == Behaviour::MNASimulation) {
277 **mDelta_p = **mDelta_p + mTimeStep * dDelta_p;
278 **mEp = Complex(**mEp_abs * cos(**mDelta_p), **mEp_abs * sin(**mDelta_p));
279 }
280
281 mStates << Math::abs(**mEp), Math::phaseDeg(**mEp), **mElecActivePower,
282 **mMechPower, **mDelta_p, **mOmMech, dOmMech, dDelta_p,
283 (**mIntfVoltage)(0, 0).real(), (**mIntfVoltage)(0, 0).imag();
284 SPDLOG_LOGGER_DEBUG(mSLog, "\nStates, time {:f}: \n{:s}", time,
285 Logger::matrixToString(mStates));
286}

288void DP::Ph1::SynchronGeneratorTrStab::mnaParentInitialize(
289 Real omega, Real timeStep, Attribute<Matrix>::Ptr leftVector) {
290 mTimeStep = timeStep;
291 if (!mUseSubcomponents) {
292 mDirectAdmittance = MatrixComp::Zero(mNumFreqs, 1);
293 mDirectPreviousCurrentFactor = MatrixComp::Zero(mNumFreqs, 1);
294 mDirectHistory = MatrixComp::Zero(mNumFreqs, 1);
295 mDirectCurrent = MatrixComp::Zero(mNumFreqs, 1);
296 mDirectVoltage = MatrixComp::Zero(mNumFreqs, 1);
297 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
298 const Real b = timeStep * 2. * PI * mFrequencies(freq, 0) / 2.;
299 const Real a = timeStep / (2. * mLpd);
300 mDirectAdmittance(freq, 0) =
301 Complex(a / (1. + b * b), -a * b / (1. + b * b));
302 mDirectPreviousCurrentFactor(freq, 0) =
303 Complex((1. - b * b) / (1. + b * b), -2. * b / (1. + b * b));
304 mDirectVoltage(freq, 0) = (**mIntfVoltage)(0, freq) - **mEp;
305 const Real harmonicOmega = 2. * PI * mFrequencies(freq, 0);
306 mDirectCurrent(freq, 0) =
307 harmonicOmega != 0.
308 ? mDirectVoltage(freq, 0) / Complex(0., harmonicOmega * mLpd)
309 : (**mIntfCurrent)(0, freq);
310 }
311 } else {
312 mMnaTasks.push_back(std::make_shared<AddBStep>(*this));
313 }
314}

316void DP::Ph1::SynchronGeneratorTrStab::mnaParentApplySystemMatrixStamp(
317 SparseMatrixRow &systemMatrix) {
318 if (mUseSubcomponents)
319 return;
320 const UInt emfNode = virtualNode(0)->matrixNodeIndex();
321 const UInt sourceCurrent = virtualNode(1)->matrixNodeIndex();
322 const UInt terminal = matrixNodeIndex(0);
323 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
324 MNAStampUtils::stampAdmittance(
325 mDirectAdmittance(freq, 0), systemMatrix, emfNode, terminal, true,
326 terminalNotGrounded(0), mSLog, mNumFreqs, freq);
327 Math::setMatrixElement(systemMatrix, sourceCurrent, emfNode,
328 Complex(1., 0.), mNumFreqs, freq);
329 Math::setMatrixElement(systemMatrix, emfNode, sourceCurrent,
330 Complex(1., 0.), mNumFreqs, freq);
331 }

334void DP::Ph1::SynchronGeneratorTrStab::mnaParentApplyRightSideVectorStamp(
335 Matrix &rightVector) {
336 if (mUseSubcomponents)
337 return;
338 const UInt emfNode = virtualNode(0)->matrixNodeIndex();
339 const UInt sourceCurrent = virtualNode(1)->matrixNodeIndex();
340 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
341 mDirectHistory(freq, 0) =
342 mDirectAdmittance(freq, 0) * mDirectVoltage(freq, 0) +
343 mDirectPreviousCurrentFactor(freq, 0) * mDirectCurrent(freq, 0);
344 if (terminalNotGrounded(0))
345 Math::setVectorElement(rightVector, matrixNodeIndex(0),
346 -mDirectHistory(freq, 0), mNumFreqs, freq);
347 Math::setVectorElement(rightVector, emfNode, mDirectHistory(freq, 0),
348 mNumFreqs, freq);
349 Math::setVectorElement(rightVector, sourceCurrent, **mEp, mNumFreqs, freq);
350 }
