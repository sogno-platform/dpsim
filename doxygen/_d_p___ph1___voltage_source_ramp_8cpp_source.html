65void DP::Ph1::VoltageSourceRamp::updateState(Real time) {
66 (**mIntfVoltage)(0, 0) = **mVoltageRef;
67
68 if (time >= mSwitchTime && time < mSwitchTime + mRampTime) {
69 Real voltageAbs = Math::abs(**mVoltageRef +
70 (time - mSwitchTime) / mRampTime * mAddVoltage);
71 Real voltagePhase = Math::phase(
72 **mVoltageRef + (time - mSwitchTime) / mRampTime * mAddVoltage);
73 Real fadeInOut =
74 0.5 + 0.5 * sin((time - mSwitchTime) / mRampTime * PI + -PI / 2);
75 (**mIntfVoltage)(0, 0) =
76 Math::polar(voltageAbs, voltagePhase + fadeInOut * mAddSrcFreq * time);
77 } else if (time >= mSwitchTime + mRampTime) {
78 Real voltageAbs = Math::abs(**mVoltageRef + mAddVoltage);
79 Real voltagePhase = Math::phase(**mVoltageRef + mAddVoltage);
80 (**mIntfVoltage)(0, 0) =
81 Math::polar(voltageAbs, voltagePhase + mAddSrcFreq * time);
82 }
83}
