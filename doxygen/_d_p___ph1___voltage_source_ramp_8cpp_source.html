53void DP::Ph1::VoltageSourceRamp::initialize(Matrix frequencies) {
54 SimPowerComp<Complex>::initialize(frequencies);
55
56 if (**mVoltageRef == Complex(0, 0))
57 **mVoltageRef = initialSingleVoltage(1) - initialSingleVoltage(0);
58
59 if (mUseSubcomponents) {
60 mSubVoltageSource = VoltageSource::make(**mName + "_src", mLogLevel);

77void DP::Ph1::VoltageSourceRamp::mnaParentApplySystemMatrixStamp(
78 SparseMatrixRow &systemMatrix) {
79 if (mUseSubcomponents)
80 return;
81 const UInt sourceCurrent = virtualNode(0)->matrixNodeIndex();
82 for (UInt freq = 0; freq < mNumFreqs; ++freq) {
83 if (terminalNotGrounded(0)) {
84 Math::setMatrixElement(systemMatrix, sourceCurrent, matrixNodeIndex(0),
85 Complex(-1., 0.), mNumFreqs, freq);
86 Math::setMatrixElement(systemMatrix, matrixNodeIndex(0), sourceCurrent,
87 Complex(-1., 0.), mNumFreqs, freq);
88 }
89 if (terminalNotGrounded(1)) {
90 Math::setMatrixElement(systemMatrix, sourceCurrent, matrixNodeIndex(1),
91 Complex(1., 0.), mNumFreqs, freq);
92 Math::setMatrixElement(systemMatrix, matrixNodeIndex(1), sourceCurrent,
93 Complex(1., 0.), mNumFreqs, freq);
94 }
95 }

126void DP::Ph1::VoltageSourceRamp::updateState(Real time) {
127 (**mIntfVoltage)(0, 0) = **mVoltageRef;
128
129 if (time >= mSwitchTime && time < mSwitchTime + mRampTime) {
130 Real voltageAbs = Math::abs(**mVoltageRef +
131 (time - mSwitchTime) / mRampTime * mAddVoltage);
132 Real voltagePhase = Math::phase(
133 **mVoltageRef + (time - mSwitchTime) / mRampTime * mAddVoltage);
134 Real fadeInOut =
135 0.5 + 0.5 * sin((time - mSwitchTime) / mRampTime * PI + -PI / 2);
136 (**mIntfVoltage)(0, 0) =
137 Math::polar(voltageAbs, voltagePhase + fadeInOut * mAddSrcFreq * time);
138 } else if (time >= mSwitchTime + mRampTime) {
139 Real voltageAbs = Math::abs(**mVoltageRef + mAddVoltage);
140 Real voltagePhase = Math::phase(**mVoltageRef + mAddVoltage);
141 (**mIntfVoltage)(0, 0) =
142 Math::polar(voltageAbs, voltagePhase + mAddSrcFreq * time);
143 }
144}

146void DP::Ph1::VoltageSourceRamp::mnaParentPreStep(Real time,
