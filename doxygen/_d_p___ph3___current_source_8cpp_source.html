27void DP::Ph3::CurrentSource::setParameters(MatrixComp currentRef,

59void DP::Ph3::CurrentSource::mnaCompApplyRightSideVectorStamp(
60 Matrix &rightVector) {
61 if (terminalNotGrounded(1)) {
62 Math::setVectorElement(rightVector, matrixNodeIndex(1, 0),
63 -(**mIntfCurrent)(0, 0));
64 Math::setVectorElement(rightVector, matrixNodeIndex(1, 1),
65 -(**mIntfCurrent)(1, 0));
66 Math::setVectorElement(rightVector, matrixNodeIndex(1, 2),
67 -(**mIntfCurrent)(2, 0));
68 }
69 if (terminalNotGrounded(0)) {
70 Math::setVectorElement(rightVector, matrixNodeIndex(0, 0),
71 (**mIntfCurrent)(0, 0));
72 Math::setVectorElement(rightVector, matrixNodeIndex(0, 1),
73 (**mIntfCurrent)(1, 0));
74 Math::setVectorElement(rightVector, matrixNodeIndex(0, 2),
75 (**mIntfCurrent)(2, 0));
76 }
77}

79void DP::Ph3::CurrentSource::updateCurrent(Real time) {
80 if (mSrcSig != nullptr) {
81 mSrcSig->step(time);
82 **mIntfCurrent = **mCurrentRef * mSrcSig->getSignal();
83 } else {
84 **mIntfCurrent = **mCurrentRef;
85 }
86}

116void DP::Ph3::CurrentSource::mnaCompUpdateVoltage(const Matrix &leftVector) {
117 **mIntfVoltage = MatrixComp::Zero(3, 1);
118 if (terminalNotGrounded(1)) {
119 (**mIntfVoltage)(0, 0) =
120 Math::complexFromVectorElement(leftVector, matrixNodeIndex(1, 0));
121 (**mIntfVoltage)(1, 0) =
122 Math::complexFromVectorElement(leftVector, matrixNodeIndex(1, 1));
123 (**mIntfVoltage)(2, 0) =
124 Math::complexFromVectorElement(leftVector, matrixNodeIndex(1, 2));
125 }
126 if (terminalNotGrounded(0)) {
127 (**mIntfVoltage)(0, 0) =
128 (**mIntfVoltage)(0, 0) -
129 Math::complexFromVectorElement(leftVector, matrixNodeIndex(0, 0));
130 (**mIntfVoltage)(1, 0) =
131 (**mIntfVoltage)(1, 0) -
132 Math::complexFromVectorElement(leftVector, matrixNodeIndex(0, 1));
133 (**mIntfVoltage)(2, 0) =
134 (**mIntfVoltage)(2, 0) -
135 Math::complexFromVectorElement(leftVector, matrixNodeIndex(0, 2));
136 }
