12DP::Ph3::NetworkInjection::NetworkInjection(String uid, String name,
13 Bool useSubcomponents,
14 Logger::Level logLevel)
15 : CompositePowerComp<Complex>(uid, name, true, true, logLevel),
16 mUseSubcomponents(useSubcomponents),
17 mVoltageRef(mAttributes->createDynamic<MatrixComp>("V_ref")),
18 mSrcFreq(mAttributes->createDynamic<Real>("f_src")) {
19 mPhaseType = PhaseType::ABC;
20 setVirtualNodeNumber(mUseSubcomponents ? 0 : 1);
21 setTerminalNumber(1);
22 **mIntfVoltage = MatrixComp::Zero(3, 1);
23 **mIntfCurrent = MatrixComp::Zero(3, 1);
24
25 SPDLOG_LOGGER_INFO(mSLog, "Create {} {}", this->type(), name);
26
27 // Create electrical sub components
28 if (mUseSubcomponents) {
29 mSubVoltageSource =
30 std::make_shared<DP::Ph3::VoltageSource>(**mName + "_vs", mLogLevel);
31 addMNASubComponent(mSubVoltageSource,
32 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
33 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
34 mSubVoltageSource->mVoltageRef->setReference(mVoltageRef);
35 mSubVoltageSource->mSrcFreq->setReference(mSrcFreq);
36 }
37}

45void DP::Ph3::NetworkInjection::setParameters(MatrixComp voltageRef,
46 Real srcFreq) {
47 mParametersSet = true;
48
49 if (mUseSubcomponents)
50 mSubVoltageSource->setParameters(voltageRef, srcFreq);
51 else {
52 **mVoltageRef = voltageRef;
53 **mSrcFreq = srcFreq;
54 auto signal = Signal::SineWaveGenerator::make(**mName + "_sw");
55 signal->setParameters(Complex(1., 0.), srcFreq);
56 mDirectSignal = signal;
57 }
58
59 SPDLOG_LOGGER_INFO(mSLog,
60 "\nVoltage Ref={:s} [V]"
61 "\nFrequency={:s} [Hz]",
62 Logger::matrixCompToString(voltageRef),
63 Logger::realToString(srcFreq));
64}

66void DP::Ph3::NetworkInjection::initializeParentFromNodesAndTerminals(
67 Real frequency) {
68 // Connect electrical subcomponents
69 if (mUseSubcomponents) {
70 mSubVoltageSource->connect({SimNode::GND, node(0)});
71 return;
72 }
73 if ((**mVoltageRef).isZero())
74 **mVoltageRef = Math::singlePhaseVariableToThreePhase(
75 RMS3PH_TO_PEAK1PH * initialSingleVoltage(0));
76 if (!mDirectSignal) {
77 auto signal = Signal::SineWaveGenerator::make(**mName + "_sw");
78 signal->setParameters(Complex(1., 0.), **mSrcFreq);
79 mDirectSignal = signal;
80 }
81}

90void DP::Ph3::NetworkInjection::mnaParentApplySystemMatrixStamp(
91 SparseMatrixRow &systemMatrix) {
92 if (mUseSubcomponents)
93 return;
94 const PhaseType phases[] = {PhaseType::A, PhaseType::B, PhaseType::C};
95 for (UInt phase = 0; phase < 3; ++phase) {
96 const UInt sourceCurrent = virtualNode(0)->matrixNodeIndex(phases[phase]);
97 if (terminalNotGrounded(0)) {
98 Math::setMatrixElement(systemMatrix, matrixNodeIndex(0, phase),
99 sourceCurrent, Complex(1., 0.));
100 Math::setMatrixElement(systemMatrix, sourceCurrent,
101 matrixNodeIndex(0, phase), Complex(1., 0.));
102 }
103 }
104}

117 Logger::matrixToString(rightVector));

158void DP::Ph3::NetworkInjection::mnaCompUpdateCurrent(const Matrix &leftVector) {
159 if (mUseSubcomponents)
160 **mIntfCurrent = **mSubVoltageSource->mIntfCurrent;
161 else
162 for (UInt phase = 0; phase < 3; ++phase) {
163 const PhaseType phases[] = {PhaseType::A, PhaseType::B, PhaseType::C};
164 (**mIntfCurrent)(phase, 0) = Math::complexFromVectorElement(
165 leftVector, virtualNode(0)->matrixNodeIndex(phases[phase]));
166 }
