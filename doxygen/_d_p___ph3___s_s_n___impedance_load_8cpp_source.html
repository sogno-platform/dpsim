42void DP::Ph3::SSN::ImpedanceLoad::setParameters(Real activePower,
43 Real reactivePower,
44 Real nominalVoltage,
45 Real shuntConductance,
46 Real shuntSusceptance) {
47 if (!std::isfinite(activePower) || !std::isfinite(reactivePower) ||
48 !std::isfinite(nominalVoltage) || !std::isfinite(shuntConductance) ||
49 !std::isfinite(shuntSusceptance) ||
50 ((activePower != 0. || reactivePower != 0.) && nominalVoltage <= 0.))
51 throw std::invalid_argument("Invalid impedance-load parameters.");
52 mActivePower = activePower;
53 mReactivePower = reactivePower;
54 mNominalVoltage = nominalVoltage;
55 mShuntConductance = shuntConductance;
56 mShuntSusceptance = shuntSusceptance;
57 mLoadConductance = nominalVoltage > 0.
58 ? activePower / (nominalVoltage * nominalVoltage)
59 : 0.;
60 DP::SSNComp::setParameters(Matrix::Zero(12, 12), Matrix::Zero(12, 3),
61 Matrix::Zero(3, 12), Matrix::Zero(3, 3));
62}

64void DP::Ph3::SSN::ImpedanceLoad::initializeFromNodesAndTerminals(
65 Real frequency) {
66 const Real omega = 2. * PI * frequency;
67 const Real loadSusceptance =
68 mNominalVoltage > 0.
69 ? mReactivePower / (mNominalVoltage * mNominalVoltage)
70 : 0.;
71 const Real inductiveSusceptance =
72 std::max(loadSusceptance, 0.) + std::max(-mShuntSusceptance, 0.);
73 const Real capacitiveSusceptance =
74 std::max(-loadSusceptance, 0.) + std::max(mShuntSusceptance, 0.);
75 mLoadInductance =
76 inductiveSusceptance > 0. ? 1. / (omega * inductiveSusceptance) : 0.;
77 mLoadCapacitance = capacitiveSusceptance / omega;
78 MatrixComp voltage = MatrixComp::Zero(3, 1);
79 voltage(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
80 voltage(1, 0) = voltage(0, 0) * SHIFT_TO_PHASE_B;
81 voltage(2, 0) = voltage(0, 0) * SHIFT_TO_PHASE_C;
82 MatrixComp inductorCurrent = MatrixComp::Zero(3, 1);
83 MatrixComp capacitorCurrent = MatrixComp::Zero(3, 1);
84 if (mLoadInductance > 0.)
85 inductorCurrent = Complex(0., -1. / (omega * mLoadInductance)) * voltage;
86 if (mLoadCapacitance > 0.)
87 capacitorCurrent = Complex(0., omega * mLoadCapacitance) * voltage;
88
89 **mIntfVoltage = voltage;
90 (**mX).setZero();
91 if (mLoadInductance > 0.) {
92 (**mX).block<3, 1>(kLoadInductorVoltage, 0) = voltage;
93 (**mX).block<3, 1>(kLoadInductorCurrent, 0) = inductorCurrent;
94 }
95 if (mLoadCapacitance > 0.) {
96 (**mX).block<3, 1>(kLoadCapacitorVoltage, 0) = voltage;
97 (**mX).block<3, 1>(kLoadCapacitorCurrent, 0) = capacitorCurrent;
98 }
99 **mIntfCurrent = (mLoadConductance + mShuntConductance) * voltage +
100 inductorCurrent + capacitorCurrent;
101}

112void DP::Ph3::SSN::ImpedanceLoad::recomputeDiscreteModel(Real omega) {
113 const Real timeStep = mTimeStep;
114 mNodalAdmittance =
115 (mLoadConductance + mShuntConductance) * MatrixComp::Identity(3, 3);
116 const Real inductiveSusceptance =
117 (mNominalVoltage > 0.
118 ? std::max(mReactivePower / (mNominalVoltage * mNominalVoltage), 0.)
119 : 0.) +
120 std::max(-mShuntSusceptance, 0.);
121 const Real capacitiveSusceptance =
122 (mNominalVoltage > 0.
123 ? std::max(-mReactivePower / (mNominalVoltage * mNominalVoltage), 0.)
124 : 0.) +
125 std::max(mShuntSusceptance, 0.);
126 mLoadInductance =
127 inductiveSusceptance > 0. ? 1. / (omega * inductiveSusceptance) : 0.;
128 mLoadCapacitance = capacitiveSusceptance / omega;
129 if (mLoadInductance > 0.) {
130 const Real b = timeStep * omega / 2.;
131 const Complex y(timeStep / (2. * mLoadInductance) / (1. + b * b),
132 -timeStep / (2. * mLoadInductance) * b / (1. + b * b));
133 mInductorAdmittance = y * MatrixComp::Identity(3, 3);
134 mInductorPreviousCurrent =
135 Complex((1. - b * b) / (1. + b * b), -2. * b / (1. + b * b));
136 mNodalAdmittance += mInductorAdmittance;
137 } else {
138 mInductorAdmittance.setZero();
139 }
140 if (mLoadCapacitance > 0.) {
141 const Real capReal = 2. * mLoadCapacitance / timeStep;
142 mCapacitorAdmittance =
143 Complex(capReal, omega * mLoadCapacitance) * MatrixComp::Identity(3, 3);
144 mCapacitorPreviousVoltage = Complex(capReal, -omega * mLoadCapacitance) *
145 MatrixComp::Identity(3, 3);
146 mNodalAdmittance += mCapacitorAdmittance;
147 } else {
148 mCapacitorAdmittance.setZero();
149 mCapacitorPreviousVoltage.setZero();
150 }
151 mW = mNodalAdmittance;
152 mDiscreteA.setZero(12, 12);
153 mDiscreteB.setZero(12, 3);
154}

180MatrixComp DP::Ph3::SSN::ImpedanceLoad::calculateHistoryVector() const {
181 mHistoryLoad.setZero();
182 mHistoryShunt.setZero();
183 if (mLoadInductance > 0.)
184 mHistoryLoad =
185 mInductorAdmittance * (**mX).block<3, 1>(kLoadInductorVoltage, 0) +
186 mInductorPreviousCurrent * (**mX).block<3, 1>(kLoadInductorCurrent, 0);
187 if (mLoadCapacitance > 0.)
188 mHistoryShunt = -(**mX).block<3, 1>(kLoadCapacitorCurrent, 0) -
189 mCapacitorPreviousVoltage *
190 (**mX).block<3, 1>(kLoadCapacitorVoltage, 0);
191 return mHistoryLoad + mHistoryShunt;
192}

198void DP::Ph3::SSN::ImpedanceLoad::updateState(const MatrixComp &,
199 const MatrixComp &voltage) {
200 if (mLoadInductance > 0.) {
201 const MatrixComp current = mInductorAdmittance * voltage + mHistoryLoad;
202 (**mX).block<3, 1>(kLoadInductorVoltage, 0) = voltage;
203 (**mX).block<3, 1>(kLoadInductorCurrent, 0) = current;
204 }
205 if (mLoadCapacitance > 0.) {
206 const MatrixComp current = mCapacitorAdmittance * voltage + mHistoryShunt;
207 (**mX).block<3, 1>(kLoadCapacitorVoltage, 0) = voltage;
208 (**mX).block<3, 1>(kLoadCapacitorCurrent, 0) = current;
209 }
