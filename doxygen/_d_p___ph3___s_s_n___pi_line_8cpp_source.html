20void DP::Ph3::SSN::PiLine::setParameters(const Matrix &seriesResistance,
21 const Matrix &seriesInductance,
22 const Matrix &parallelCapacitance,
23 const Matrix &parallelConductance) {
24 for (const Matrix *parameter : {&seriesResistance, &seriesInductance,
25 &parallelCapacitance, &parallelConductance}) {
26 if (parameter->rows() != 3 || parameter->cols() != 3)
27 throw std::invalid_argument("PiLine parameter matrices must be 3 x 3.");
28 }
29 Base::Ph3::PiLine::setParameters(seriesResistance, seriesInductance,
30 parallelCapacitance, parallelConductance);
31
32 const Matrix inverseInductance = seriesInductance.inverse();
33 const Matrix identity = Matrix::Identity(3, 3);
34 Matrix a = -inverseInductance * seriesResistance;
35 Matrix b = Matrix::Zero(3, 6);
36 b.block(0, 0, 3, 3) = -inverseInductance;
37 b.block(0, 3, 3, 3) = inverseInductance;
38 Matrix c = Matrix::Zero(6, 3);
39 c.block(0, 0, 3, 3) = -identity;
40 c.block(3, 0, 3, 3) = identity;
41 Matrix d = Matrix::Zero(6, 6);
42 d.block(0, 0, 3, 3) = parallelConductance / 2.;
43 d.block(3, 3, 3, 3) = parallelConductance / 2.;
44 Matrix d1 = Matrix::Zero(6, 6);
45 d1.block(0, 0, 3, 3) = parallelCapacitance / 2.;
46 d1.block(3, 3, 3, 3) = parallelCapacitance / 2.;
47
48 SSNComp::setParameters(a, b, c, d, d1);
49}
