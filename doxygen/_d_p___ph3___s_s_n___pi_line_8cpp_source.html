54void DP::Ph3::SSN::PiLine::initializeFromNodesAndTerminals(Real frequency) {
55 if ((**mParallelCond)(0, 0) <= 0.) {
56 **mParallelCond = Matrix::Zero(3, 3);
57 (**mParallelCond).diagonal().setConstant(1e-6);
58 }
59
60 const Real omega = 2. * PI * frequency;
61 MatrixComp v0 = MatrixComp::Zero(3, 1);
62 MatrixComp v1 = MatrixComp::Zero(3, 1);
63 v0(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
64 v1(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(1);
65 v0(1, 0) = v0(0, 0) * SHIFT_TO_PHASE_B;
66 v0(2, 0) = v0(0, 0) * SHIFT_TO_PHASE_C;
67 v1(1, 0) = v1(0, 0) * SHIFT_TO_PHASE_B;
68 v1(2, 0) = v1(0, 0) * SHIFT_TO_PHASE_C;
69
70 const MatrixComp voltage = v1 - v0;
71 MatrixComp impedance = (**mSeriesRes).cast<Complex>() +
72 Complex(0., omega) * (**mSeriesInd).cast<Complex>();
73 const MatrixComp current = impedance.inverse() * voltage;
74 const MatrixComp inductorVoltage = voltage - **mSeriesRes * current;
75 const MatrixComp cap0Current =
76 Complex(0., omega) * ((**mParallelCap) / 2.).cast<Complex>() * v0;
77 const MatrixComp cap1Current =
78 Complex(0., omega) * ((**mParallelCap) / 2.).cast<Complex>() * v1;
79
80 **mIntfVoltage = MatrixComp::Zero(6, 1);
81 (**mIntfVoltage).block(0, 0, 3, 1) = v0;
82 (**mIntfVoltage).block(3, 0, 3, 1) = v1;
83 **mIntfCurrent = MatrixComp::Zero(6, 1);
84 (**mIntfCurrent).block(0, 0, 3, 1) = -current + cap0Current;
85 (**mIntfCurrent).block(3, 0, 3, 1) = current + cap1Current;
86 (**mX).setZero();
87 (**mX).block<3, 1>(kSeriesCurrent, 0) = current;
88 (**mX).block<3, 1>(kSeriesInductorVoltage, 0) = inductorVoltage;
89 (**mX).block<3, 1>(kCapVoltage0, 0) = v0;
90 (**mX).block<3, 1>(kCapCurrent0, 0) = cap0Current;
91 (**mX).block<3, 1>(kCapVoltage1, 0) = v1;
92 (**mX).block<3, 1>(kCapCurrent1, 0) = cap1Current;
93}

95void DP::Ph3::SSN::PiLine::recomputeDiscreteModel(Real omega) {
96 const Real timeStep = mTimeStep;
97 const Real b = timeStep * omega / 2.;
98 const Matrix inverseL = (**mSeriesInd).inverse();
99 const MatrixComp a = (timeStep / 2.) * inverseL.cast<Complex>();
100 mInductorAdmittance = a / Complex(1., b);
101 mInductorPreviousCurrent =
102 Complex((1. - b * b) / (1. + b * b), -2. * b / (1. + b * b));
103 mSeriesSystemInverse = (MatrixComp::Identity(3, 3) +
104 mInductorAdmittance * (**mSeriesRes).cast<Complex>())
105 .inverse();
106 mSeriesAdmittance = mSeriesSystemInverse * mInductorAdmittance;
107 const Matrix capHalf = **mParallelCap / 2.;
108 const MatrixComp capReal = (2. / timeStep) * capHalf.cast<Complex>();
109 mCapAdmittance = capReal + Complex(0., omega) * capHalf.cast<Complex>();
110 mCapPreviousVoltage = capReal - Complex(0., omega) * capHalf.cast<Complex>();
111 const MatrixComp shunt =
112 mCapAdmittance + ((**mParallelCond) / 2.).cast<Complex>();
113 mW = MatrixComp::Zero(6, 6);
114 mW.block<3, 3>(0, 0) = mSeriesAdmittance + shunt;
115 mW.block<3, 3>(0, 3) = -mSeriesAdmittance;
116 mW.block<3, 3>(3, 0) = -mSeriesAdmittance;
117 mW.block<3, 3>(3, 3) = mSeriesAdmittance + shunt;
118 mDiscreteA.setZero(18, 18);
119 mDiscreteB.setZero(18, 6);
120}

122MatrixComp DP::Ph3::SSN::PiLine::calculateHistoryVector() const {
123 mInductorHistory.noalias() =
124 mInductorAdmittance * (**mX).block<3, 1>(kSeriesInductorVoltage, 0);
125 mInductorHistory +=
126 mInductorPreviousCurrent * (**mX).block<3, 1>(kSeriesCurrent, 0);
127 mSeriesHistory.noalias() = mSeriesSystemInverse * mInductorHistory;
128 mCapHistory0 = -(mCapPreviousVoltage * (**mX).block<3, 1>(kCapVoltage0, 0) +
129 (**mX).block<3, 1>(kCapCurrent0, 0));
130 mCapHistory1 = -(mCapPreviousVoltage * (**mX).block<3, 1>(kCapVoltage1, 0) +
131 (**mX).block<3, 1>(kCapCurrent1, 0));
132 MatrixComp history = MatrixComp::Zero(6, 1);
133 history.block<3, 1>(0, 0) = -mSeriesHistory + mCapHistory0;
134 history.block<3, 1>(3, 0) = mSeriesHistory + mCapHistory1;
135 return history;
136}

142void DP::Ph3::SSN::PiLine::updateState(const MatrixComp &,
143 const MatrixComp &input) {
144 const TerminalVoltage v0 = input.block<3, 1>(0, 0);
145 const TerminalVoltage v1 = input.block<3, 1>(3, 0);
146 const TerminalVoltage voltage = v1 - v0;
147 const TerminalVoltage current = mSeriesAdmittance * voltage + mSeriesHistory;
148 const TerminalVoltage inductorVoltage = voltage - **mSeriesRes * current;
149 const TerminalVoltage cap0Current = mCapAdmittance * v0 + mCapHistory0;
150 const TerminalVoltage cap1Current = mCapAdmittance * v1 + mCapHistory1;
151 (**mX).block<3, 1>(kSeriesCurrent, 0) = current;
152 (**mX).block<3, 1>(kSeriesInductorVoltage, 0) = inductorVoltage;
153 (**mX).block<3, 1>(kCapVoltage0, 0) = v0;
154 (**mX).block<3, 1>(kCapCurrent0, 0) = cap0Current;
155 (**mX).block<3, 1>(kCapVoltage1, 0) = v1;
156 (**mX).block<3, 1>(kCapCurrent1, 0) = cap1Current;
157}
