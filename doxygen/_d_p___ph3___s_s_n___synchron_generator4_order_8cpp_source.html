14MatrixComp unpack3(const Matrix &u) {
15 MatrixComp v(3, 1);
16 for (Int p = 0; p < 3; ++p)
17 v(p, 0) = Complex(u(2 * p, 0), u(2 * p + 1, 0));
18 return v;
19}
20Matrix pack3(const MatrixComp &y) {
21 Matrix out(6, 1);
22 for (Int p = 0; p < 3; ++p) {
23 out(2 * p, 0) = y(p, 0).real();
24 out(2 * p + 1, 0) = y(p, 0).imag();
25 }
26 return out;
27}

75void DP::Ph3::SSN::SynchronGenerator4Order::evaluate(const Matrix &x,
76 const Matrix &u, Matrix &f,
77 Matrix &y) const {
78 f.setZero(4, 1);
79 y.resize(6, 1);
80 const Complex v =
81 (Complex(u(0, 0), u(1, 0)) + alpha * Complex(u(2, 0), u(3, 0)) +
82 std::conj(alpha) * Complex(u(4, 0), u(5, 0))) /
83 (3. * mBaseV);
84 const Complex vdq = v * std::polar(1., -x(kTheta, 0));
85 const Real vd = vdq.real(), vq = vdq.imag();
86 const Real dd = vd - x(kEd, 0), dq = vq - x(kEq, 0),
87 det = mRa * mRa + mXdp * mXqp;
88 const Real id = (mRa * dd + mXqp * dq) / det,
89 iq = (-mXdp * dd + mRa * dq) / det;
90 const Complex idq(id, iq), iph = mBaseI * idq * std::polar(1., x(kTheta, 0));
91 const std::array<Complex, 3> currents = {iph, iph * std::conj(alpha),
92 iph * alpha};
93 for (Int phase = 0; phase < 3; ++phase) {
94 y(2 * phase, 0) = currents[phase].real();
95 y(2 * phase + 1, 0) = currents[phase].imag();
96 }
97 f(kEd, 0) = (-(x(kEd, 0) - mEd0) - (mXq - mXqp) * (iq - mIq0)) / mTq0p;
98 f(kEq, 0) = (-(x(kEq, 0) - mEq0) + (mXd - mXdp) * (id - mId0)) / mTd0p;
99 const Real pe = vd * id + vq * iq;
100 f(kOmega, 0) = (mPm + pe - mDamping * (x(kOmega, 0) - 1.)) / (2. * mH);
101 f(kTheta, 0) = mOmega * (x(kOmega, 0) - 1.);
102}

104void DP::Ph3::SSN::SynchronGenerator4Order::linearize(const Matrix &x,
105 const Matrix &u,
106 Matrix &A, Matrix &B,
107 Matrix &C,
108 Matrix &D) const {
109 A.setZero(4, 4);
110 B.setZero(4, 6);
111 C.setZero(6, 4);
112 D.setZero(6, 6);
113
114 // Positive-sequence projection and its exact derivatives with respect to
115 // the packed real/imaginary phase-envelope inputs.
116 const std::array<Complex, 3> sequenceCoeff = {Complex(1., 0.), alpha,
117 std::conj(alpha)};
118 Complex positiveSequence = 0.;
119 for (Int phase = 0; phase < 3; ++phase)
120 positiveSequence +=
121 sequenceCoeff[phase] * Complex(u(2 * phase, 0), u(2 * phase + 1, 0));
122 positiveSequence /= 3. * mBaseV;
123
124 const Real theta = x(kTheta, 0);
125 const Complex dqVoltage = positiveSequence * std::polar(1., -theta);
126 const Real vd = dqVoltage.real();
127 const Real vq = dqVoltage.imag();
128
129 std::array<Real, 6> dVdDu{};
130 std::array<Real, 6> dVqDu{};
131 const Complex rotateToDQ = std::polar(1., -theta) / (3. * mBaseV);
132 for (Int phase = 0; phase < 3; ++phase) {
133 const Complex dVdRe = sequenceCoeff[phase] * rotateToDQ;
134 const Complex dVdIm = Complex(0., 1.) * dVdRe;
135 dVdDu[2 * phase] = dVdRe.real();
136 dVqDu[2 * phase] = dVdRe.imag();
137 dVdDu[2 * phase + 1] = dVdIm.real();
138 dVqDu[2 * phase + 1] = dVdIm.imag();
139 }
140
141 const Real det = mRa * mRa + mXdp * mXqp;
142 const Real idVd = mRa / det;
143 const Real idVq = mXqp / det;
144 const Real iqVd = -mXdp / det;
145 const Real iqVq = mRa / det;
146 const Real idEd = -idVd;
147 const Real idEq = -idVq;
148 const Real iqEd = -iqVd;
149 const Real iqEq = -iqVq;
150
151 const Real dd = vd - x(kEd, 0);
152 const Real dq = vq - x(kEq, 0);
153 const Real id = (mRa * dd + mXqp * dq) / det;
154 const Real iq = (-mXdp * dd + mRa * dq) / det;
155
156 // d(vd,vq)/d(theta) = (vq,-vd).
157 const Real vdTheta = vq;
158 const Real vqTheta = -vd;
159 const Real idTheta = idVd * vdTheta + idVq * vqTheta;
160 const Real iqTheta = iqVd * vdTheta + iqVq * vqTheta;
161
162 const Real edScale = -(mXq - mXqp) / mTq0p;
163 const Real eqScale = (mXd - mXdp) / mTd0p;
164 const Real powerScale = 1. / (2. * mH);
165
166 A(0, 0) = -1. / mTq0p + edScale * iqEd;
167 A(0, 1) = edScale * iqEq;
168 A(0, 3) = edScale * iqTheta;
169 A(1, 0) = eqScale * idEd;
170 A(1, 1) = -1. / mTd0p + eqScale * idEq;
171 A(1, 3) = eqScale * idTheta;
172 A(2, 0) = powerScale * (vd * idEd + vq * iqEd);
173 A(2, 1) = powerScale * (vd * idEq + vq * iqEq);
174 A(2, 2) = -mDamping * powerScale;
175 A(2, 3) =
176 powerScale * (vdTheta * id + vd * idTheta + vqTheta * iq + vq * iqTheta);
177 A(3, 2) = mOmega;
178
179 for (Int input = 0; input < 6; ++input) {
180 const Real idDu = idVd * dVdDu[input] + idVq * dVqDu[input];
181 const Real iqDu = iqVd * dVdDu[input] + iqVq * dVqDu[input];
182 B(0, input) = edScale * iqDu;
183 B(1, input) = eqScale * idDu;
184 B(2, input) = powerScale * (vd * idDu + vq * iqDu);
185 }
186
187 const Complex idq(id, iq);
188 const Complex phaseCurrentGain = mBaseI * std::polar(1., theta);
189 const std::array<Complex, 3> phaseCoeff = {Complex(1., 0.), std::conj(alpha),
190 alpha};
191 for (Int phase = 0; phase < 3; ++phase) {
192 const Complex gain = phaseCurrentGain * phaseCoeff[phase];
193 const Int reRow = 2 * phase;
194 const Int imRow = reRow + 1;
195
196 const Complex dIdqDEd(idEd, iqEd);
197 const Complex dIdqDEq(idEq, iqEq);
198 const Complex dIdqDTheta(idTheta, iqTheta);
199 const Complex dITheta = gain * (Complex(0., 1.) * idq + dIdqDTheta);
200 const Complex dIEd = gain * dIdqDEd;
201 const Complex dIEq = gain * dIdqDEq;
202 C(reRow, 0) = dIEd.real();
203 C(imRow, 0) = dIEd.imag();
204 C(reRow, 1) = dIEq.real();
205 C(imRow, 1) = dIEq.imag();
206 C(reRow, 3) = dITheta.real();
207 C(imRow, 3) = dITheta.imag();
208
209 for (Int input = 0; input < 6; ++input) {
210 const Real idDu = idVd * dVdDu[input] + idVq * dVqDu[input];
211 const Real iqDu = iqVd * dVdDu[input] + iqVq * dVqDu[input];
212 const Complex dI = gain * Complex(idDu, iqDu);
213 D(reRow, input) = dI.real();
214 D(imRow, input) = dI.imag();
215 }
216 }
217}

219Bool DP::Ph3::SSN::SynchronGenerator4Order::updateComponentParameters() {
220 for (Int phase = 0; phase < 3; ++phase) {
221 const Complex voltage = (**mIntfVoltage)(phase, 0);
222 mPackedVoltage(2 * phase, 0) = voltage.real();
223 mPackedVoltage(2 * phase + 1, 0) = voltage.imag();
224 }
225
226 linearize(**mX, mPackedVoltage, mA, mB, mC, mD);
227 evaluate(**mX, mPackedVoltage, mEvaluationDerivative, mEvaluationOutput);
228
229 mE = mEvaluationDerivative;
230 mE.noalias() -= mA * (**mX);
231 mE.noalias() -= mB * mPackedVoltage;
232 mF = mEvaluationOutput;
233 mF.noalias() -= mC * (**mX);
234 mF.noalias() -= mD * mPackedVoltage;
235 return true;
236}

238void DP::Ph3::SSN::SynchronGenerator4Order::initializeState(
239 const MatrixComp &phaseVoltage) {
240 if (!mConfigured || !mInitialValuesConfigured)
241 throw std::logic_error(
242 "Configure SSN generator parameters and PF values first.");
243 const Complex v = mVpu * std::polar(1., mVangle);
244 const Complex i = -std::conj(Complex(mPm, mQm) / v);
245 const Complex e = v - Complex(mRa, .5 * (mXdp + mXqp)) * i;
246 const Real theta = std::arg(e) - PI / 2.;
247 const Complex rot = std::polar(1., -theta);
248 const Complex vdq = v * rot, idq = i * rot;
249 (**mX)(kTheta, 0) = theta;
250 (**mX)(kOmega, 0) = 1.;
251 (**mX)(kEd, 0) = vdq.real() - mRa * idq.real() + mXqp * idq.imag();
252 (**mX)(kEq, 0) = vdq.imag() - mXdp * idq.real() - mRa * idq.imag();
253 mEd0 = (**mX)(kEd, 0);
254 mEq0 = (**mX)(kEq, 0);
255 mId0 = idq.real();
256 mIq0 = idq.imag();
257 (void)phaseVoltage;
258}

260void DP::Ph3::SSN::SynchronGenerator4Order::initializeFromNodesAndTerminals(
261 Real) {
262 if (!mConfigured)
263 throw std::logic_error("SSN generator parameters are unset.");
264 MatrixComp voltage = MatrixComp::Zero(3, 1);
265 voltage(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(1);
266 voltage(1, 0) = voltage(0, 0) * SHIFT_TO_PHASE_B;
267 voltage(2, 0) = voltage(0, 0) * SHIFT_TO_PHASE_C;
268 // SystemTopology transfers solved SG power onto terminal 0 for the matching
269 // power-flow component name, including the slack-bus active-power result.
270 const Complex solvedPower = -terminal(0)->singlePower();
271 if (std::isfinite(solvedPower.real()) && std::isfinite(solvedPower.imag()) &&
272 std::abs(solvedPower) > 1.) {
273 mPm = solvedPower.real() / mNominalPower;
274 mQm = solvedPower.imag() / mNominalPower;
275 }
276 mVpu = std::abs(voltage(0, 0)) / mBaseV;
277 mVangle = std::arg(voltage(0, 0));
278 **mIntfVoltage = voltage;
279 initializeState(voltage);
280 updateComponentParameters();
281 Matrix f, y;
282 evaluate(**mX, pack3(voltage), f, y);
283 **mIntfCurrent = unpack3(y);
284}
