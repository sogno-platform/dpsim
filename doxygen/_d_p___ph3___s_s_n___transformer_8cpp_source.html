67void DP::Ph3::SSN::Transformer::resolveWindingRoles() {
68 switch (mReferenceWinding) {
69 case WindingReference::Primary:
70 mReferenceTerminal = 0;
71 break;
72 case WindingReference::Secondary:
73 mReferenceTerminal = 1;
74 break;
75 case WindingReference::Tertiary:
76 throw std::invalid_argument("Three-winding transformers are unsupported.");
77 case WindingReference::Auto:
78 mReferenceTerminal =
79 (mNominalVoltagePrimary >= mNominalVoltageSecondary) ? 0 : 1;
80 break;
81 }
82 mRatioFromReference =
83 (mReferenceTerminal == 0) ? effectiveRatio() : 1. / effectiveRatio();
84}

86void DP::Ph3::SSN::Transformer::initializeFromNodesAndTerminals(
87 Real frequency) {
88 if (frequency <= 0.)
89 throw std::invalid_argument("Frequency must be positive.");
90 resolveWindingRoles();
91 const Real omega = 2. * PI * frequency;
92 const Real magnetizingSusceptance =
93 std::sqrt(mNoLoadCurrent * mNoLoadCurrent - mNoLoadLoss * mNoLoadLoss) *
94 mRatedPower / std::pow(nominalVoltageAt(mReferenceTerminal), 2);
95 const Real magnetizingInductance = 1. / (omega * magnetizingSusceptance);
96 const Real magnetizingResistance =
97 std::pow(nominalVoltageAt(mReferenceTerminal), 2) /
98 (mNoLoadLoss * mRatedPower);
99 mMagnetizingInductance =
100 Math::singlePhaseParameterToThreePhase(magnetizingInductance);
101 mMagnetizingResistance =
102 Math::singlePhaseParameterToThreePhase(magnetizingResistance);
103
104 mSeriesInductanceHalf = mInductance / 2.;
105 mSeriesResistanceHalf.setZero();
106 if (mWithResistiveLosses)
107 mSeriesResistanceHalf = mResistance / 2.;
108
109 // State x = [i_ref; i_other], with the magnetizing current eliminated by
110 // midpoint KCL: i_mag = i_ref + i_other.
111 Matrix mass = Matrix::Zero(6, 6);
112 Matrix resistance = Matrix::Zero(6, 6);
113 mass.block<3, 3>(0, 0) = mSeriesInductanceHalf + mMagnetizingInductance;
114 mass.block<3, 3>(0, 3) = mMagnetizingInductance;
115 mass.block<3, 3>(3, 0) = mMagnetizingInductance;
116 mass.block<3, 3>(3, 3) = mSeriesInductanceHalf + mMagnetizingInductance;
117 resistance.block<3, 3>(0, 0) = mSeriesResistanceHalf + mMagnetizingResistance;
118 resistance.block<3, 3>(0, 3) = mMagnetizingResistance;
119 resistance.block<3, 3>(3, 0) = mMagnetizingResistance;
120 resistance.block<3, 3>(3, 3) = mSeriesResistanceHalf + mMagnetizingResistance;
121
122 const Matrix inverseMass = mass.inverse();
123 const Matrix identity = Matrix::Identity(3, 3);
124 const MatrixComp identityComp = identity.cast<Complex>();
125 mInputTransform.setZero();
126 mOutputTransform.setZero();
127 if (mReferenceTerminal == 0) {
128 mInputTransform.block<3, 3>(0, 0) = identityComp;
129 mInputTransform.block<3, 3>(3, 3) = mRatioFromReference * identityComp;
130 mOutputTransform.block<3, 3>(0, 0) = identityComp;
131 mOutputTransform.block<3, 3>(3, 3) =
132 std::conj(mRatioFromReference) * identityComp;
133 } else {
134 mInputTransform.block<3, 3>(0, 3) = identityComp;
135 mInputTransform.block<3, 3>(3, 0) = mRatioFromReference * identityComp;
136 mOutputTransform.block<3, 3>(0, 3) =
137 std::conj(mRatioFromReference) * identityComp;
138 mOutputTransform.block<3, 3>(3, 0) = identityComp;
139 }
140
141 SSNComp::setParameters(-inverseMass * resistance, inverseMass,
142 Matrix::Identity(6, 6), Matrix::Zero(6, 6));
143 MultiPortVTypeSSNComp::initializeFromNodesAndTerminals(frequency);
144}

146void DP::Ph3::SSN::Transformer::recomputeDiscreteModel(Real omega) {
147 SSNComp::recomputeDiscreteModel(omega);
148 mDiscreteB *= mInputTransform;
149 mW =
150 mOutputTransform * (mC.cast<Complex>() * mDiscreteB + mD.cast<Complex>());

153MatrixComp DP::Ph3::SSN::Transformer::calculateHistoryVector() const {
154 MatrixComp stateAndInput =
155 mDiscreteA * (**mX) + mDiscreteB * (**inputAttribute());
156 return mOutputTransform * mC.cast<Complex>() * stateAndInput;
157}

159void DP::Ph3::SSN::Transformer::updateHistoryVector(MatrixComp &history) {
160 mHistoryState.noalias() = mDiscreteA * (**mX);
161 mHistoryInput.noalias() = mDiscreteB * (**inputAttribute());
162 mHistoryState += mHistoryInput;
163 history.noalias() = mOutputTransform * mComplexC * mHistoryState;
164}
