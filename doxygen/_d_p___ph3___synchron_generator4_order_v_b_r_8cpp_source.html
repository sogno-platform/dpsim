36void DP::Ph3::SynchronGenerator4OrderVBR::mnaCompInitialize(
37 Real omega, Real timeStep, Attribute<Matrix>::Ptr leftVector) {
38 updateMatrixNodeIndices();
39 mTimeStep = timeStep;
40
41 // Reuse the established power-flow initialization for the rotor angle,
42 // speed and d/q terminal quantities. The winding-level state-space model
43 // is not used by this four-state VBR implementation.
44 initializeMatrixAndStates();
45 const Real xTransient = 0.5 * (**mLd_t + **mLq_t);
46 mEdTransient = mVdq0(0, 0) - mIdq0(1, 0) * **mLq_t;
47 mEqTransient = mVdq0(1, 0) + mIdq0(0, 0) * **mLd_t;
48 mInitialEdTransient = mEdTransient;
49 mInitialEqTransient = mEqTransient;
50 mCurrentD = mIdq0(0, 0);
51 mCurrentQ = mIdq0(1, 0);
52 mInitialCurrentD = mCurrentD;
53 mInitialCurrentQ = mCurrentQ;
54 // Use the initialized terminal power as the mechanical reference. The
55 // MATPOWER value is specified on the system base while the machine states
56 // are on the machine base, so directly dividing it by mNomPower can create
57 // a spurious initial acceleration when those bases differ.
58 mMechanicalPower =
59 -1.5 * (mVdq0(0, 0) * mIdq0(0, 0) + mVdq0(1, 0) * mIdq0(1, 0));
60
61 const Complex phaseImpedance = mBase_Z * Complex(**mRs, xTransient);
62 const Complex phaseAdmittance = Complex(1, 0) / phaseImpedance;
63 mAdmittance.setZero();
64 mAdmittance.diagonal().setConstant(phaseAdmittance);
65 const MatrixComp nortonVoltage =
66 **mIntfVoltage - phaseImpedance * **mIntfCurrent;
67 MatrixComp nortonVoltageCopy = nortonVoltage;
68 const Matrix nortonDq =
69 abcToDq0Transform(mThetaMech, nortonVoltageCopy) / mBase_V;
70 mNortonEd = nortonDq(0, 0);
71 mNortonEq = nortonDq(1, 0);
72 mTimeStep = timeStep;
73 mnaCompApplyRightSideVectorStamp(**mRightVector);
74}

83void DP::Ph3::SynchronGenerator4OrderVBR::mnaCompApplyRightSideVectorStamp(
84 Matrix &rightVector) {
85 MatrixFixedSize<3, 1> dqInternal;
86 dqInternal << mNortonEd + mEdTransient - mInitialEdTransient,
87 mNortonEq + mEqTransient - mInitialEqTransient, 0;
88 mInternalVoltage = mBase_V * dq0ToAbcTransform(mThetaMech, dqInternal);
89 // The phase-domain Norton admittance is diagonal with the same value in
90 // all three phases. Avoid dispatching a dense 3x3-by-3x1 product at every
91 // RHS refresh.
92 mNortonCurrent.noalias() = mAdmittance(0, 0) * mInternalVoltage;
93 if (terminalNotGrounded(0)) {
94 for (UInt phase = 0; phase < 3; ++phase)
95 Math::setVectorElement(rightVector, matrixNodeIndex(0, phase),
96 mNortonCurrent(phase, 0));
97 }
98}

100void DP::Ph3::SynchronGenerator4OrderVBR::mnaAddCompactRightSideVectorStamp(
101 Matrix &rightVector) {
102 if (!terminalNotGrounded(0))
103 return;
104 const Matrix::Index imaginaryOffset = rightVector.rows() / 2;
105 for (UInt phase = 0; phase < 3; ++phase) {
106 const Matrix::Index row = matrixNodeIndex(0, phase);
107 rightVector(row, 0) += mNortonCurrent(phase, 0).real();
108 rightVector(row + imaginaryOffset, 0) += mNortonCurrent(phase, 0).imag();
109 }
110}

126void DP::Ph3::SynchronGenerator4OrderVBR::mnaCompPostStep(
127 Real time, Int timeStepCount, Attribute<Matrix>::Ptr &leftVector) {
128 SynchronGeneratorDQ::mnaCompPostStep(time, timeStepCount, leftVector);
129 if (!mInitialMechanicalPowerSet) {
130 // Calibrate mechanical input from the solved initial network port. This
131 // makes the first dynamic step torque-balanced even when the machine and
132 // MATPOWER use different power bases or the PF voltage has rounding.
133 mVoltageDifference = **mIntfVoltage - mInternalVoltage;
134 **mIntfCurrent = mAdmittance(0, 0) * mVoltageDifference;
135 MatrixFixedSize<3, 1> dqVoltage =
136 abcToDq0Transform(mThetaMech, **mIntfVoltage) / mBase_V;
137 MatrixFixedSize<3, 1> dqCurrent =
138 abcToDq0Transform(mThetaMech, **mIntfCurrent) / mBase_I;
139 mMechanicalPower = -1.5 * (dqVoltage(0, 0) * dqCurrent(0, 0) +
140 dqVoltage(1, 0) * dqCurrent(1, 0));
141 // The solved PF port is authoritative. Anchor the transient current
142 // states to it rather than to the analytic machine-base reconstruction;
143 // this removes the small PF/phasor conversion residual at t=0.
144 mInitialCurrentD = dqCurrent(0, 0);
145 mInitialCurrentQ = dqCurrent(1, 0);
146 mCurrentD = mInitialCurrentD;
147 mCurrentQ = mInitialCurrentQ;
148 mInitialMechanicalPowerSet = true;
149 }

152void DP::Ph3::SynchronGenerator4OrderVBR::updateCurrentAndStates(Real time) {
153 // Reconstruct the component current from the implicit Norton equivalent and
154 // transform it into the rotor frame for the machine equations.
155 mVoltageDifference = **mIntfVoltage - mInternalVoltage;
156 **mIntfCurrent = mAdmittance(0, 0) * mVoltageDifference;
157 MatrixFixedSize<3, 1> dqVoltage =
158 abcToDq0Transform(mThetaMech, **mIntfVoltage) / mBase_V;
159 MatrixFixedSize<3, 1> dqCurrent =
160 abcToDq0Transform(mThetaMech, **mIntfCurrent) / mBase_I;
161 mCurrentD = dqCurrent(0, 0);
162 mCurrentQ = dqCurrent(1, 0);
163
164 const Real electricalPower =
165 -1.5 * (dqVoltage(0, 0) * mCurrentD + dqVoltage(1, 0) * mCurrentQ);
166 const Real omega = **mOmMech;
167 if (time > 0) {
168 // DP component current is positive into the machine; the classical
169 // generator equations use current positive out of the machine.
170 const Real dEd = (-(mEdTransient - mInitialEdTransient) -
171 (**mLq - **mLq_t) * (mCurrentQ - mInitialCurrentQ)) /
172 std::max(**mTq0_t, 1e-6);
173 const Real dEq = (-(mEqTransient - mInitialEqTransient) +
174 (**mLd - **mLd_t) * (mCurrentD - mInitialCurrentD)) /
175 std::max(**mTd0_t, 1e-6);
176 const Real dOmega =
177 (mMechanicalPower - electricalPower - mDamping * (omega - 1.0)) /
178 (2.0 * std::max(**mInertia, 1e-6));
179 mEdTransient += mTimeStep * dEd;
180 mEqTransient += mTimeStep * dEq;
181 **mOmMech = omega + mTimeStep * dOmega;
182 mThetaMech += mTimeStep * (omega - 1.0) * mBase_OmMech;
183 }
