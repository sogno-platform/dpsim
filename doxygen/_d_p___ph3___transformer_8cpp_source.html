45void DP::Ph3::Transformer::resolveWindingRoles() {
46 switch (mReferenceWinding) {
47 case WindingReference::Primary:
48 mReferenceTerminal = 0;
49 break;
50 case WindingReference::Secondary:
51 mReferenceTerminal = 1;
52 break;
53 case WindingReference::Tertiary:
54 SPDLOG_LOGGER_ERROR(mSLog,
55 "Transformer {}: three-winding transformers are "
56 "not implemented",
57 this->name());
58 throw InvalidArgumentException();
59 case WindingReference::Auto:
60 mReferenceTerminal =
61 (mNominalVoltagePrimary >= mNominalVoltageSecondary) ? 0 : 1;
62 break;
63 }
64
65 mRatioFromReference = (mReferenceTerminal == 0) ? **mRatio : 1. / **mRatio;
66 mOrientationSign = (mReferenceTerminal == 0) ? 1. : -1.;
67
68 SPDLOG_LOGGER_INFO(
69 mSLog,
70 "Impedance referred to winding {} ({} [V]), remaining winding "
71 "at terminal {} ({} [V]), ratio = {}",
72 mReferenceTerminal + 1, nominalVoltageAt(mReferenceTerminal),
73 nonReferenceTerminal(), nominalVoltageAt(nonReferenceTerminal()),
74 Logger::complexToString(mRatioFromReference));
75}

77void DP::Ph3::Transformer::createSubComponents() {
78 if (mSubCompCreated)
79 return;
80 mSubCompCreated = true;
81
82 resolveWindingRoles();
83
84 auto midpoint = mVirtualNodes[2];
85
86 mSubInductor =
87 std::make_shared<DP::Ph3::Inductor>(**mName + "_ind", mLogLevel);
88 mSubInductor->setParameters(mInductance / 2.);
89 addMNASubComponent(mSubInductor, MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
90 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
91
92 mSubInductor2 =
93 std::make_shared<DP::Ph3::Inductor>(**mName + "_ind2", mLogLevel);
94 mSubInductor2->setParameters(mInductance / 2.);
95 addMNASubComponent(mSubInductor2, MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
96 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
97
98 if (mNumVirtualNodes == 5) {
99 mSubResistor =
100 std::make_shared<DP::Ph3::Resistor>(**mName + "_res", mLogLevel);
101 mSubResistor->setParameters(mResistance / 2.);
102 mSubResistor->connect({node(mReferenceTerminal), mVirtualNodes[3]});
103 mSubInductor->connect({mVirtualNodes[3], midpoint});
104 addMNASubComponent(mSubResistor, MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
105 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
106
107 mSubResistor2 =
108 std::make_shared<DP::Ph3::Resistor>(**mName + "_res2", mLogLevel);
109 mSubResistor2->setParameters(mResistance / 2.);
110 mSubResistor2->connect({midpoint, mVirtualNodes[4]});
111 mSubInductor2->connect({mVirtualNodes[4], mVirtualNodes[0]});
112 addMNASubComponent(mSubResistor2,
113 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
114 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
115 } else {
116 mSubInductor->connect({node(mReferenceTerminal), midpoint});
117 mSubInductor2->connect({midpoint, mVirtualNodes[0]});
118 }
119
120 if (mRatedPower <= 0) {
121 SPDLOG_LOGGER_WARN(mSLog,
122 "Transformer {}: rated power is {} [VA], so the "
123 "magnetizing branch cannot be sized and is omitted",
124 this->name(), mRatedPower);
125 return;
126 }
127
128 if (mNoLoadCurrent <= mNoLoadLoss) {
129 SPDLOG_LOGGER_ERROR(mSLog,
130 "Transformer {}: no-load current {} must exceed the "
131 "no-load loss {}",
132 this->name(), mNoLoadCurrent, mNoLoadLoss);
133 throw InvalidArgumentException();
134 }
135
136 Real magnetizingResistance =
137 std::pow(nominalVoltageAt(mReferenceTerminal), 2) /
138 (mNoLoadLoss * mRatedPower);
139 mMagnetizingResistance =
140 Math::singlePhaseParameterToThreePhase(magnetizingResistance);
141 mSubMagnetizingResistor =
142 std::make_shared<DP::Ph3::Resistor>(**mName + "_mag_res", mLogLevel);
143 mSubMagnetizingResistor->setParameters(mMagnetizingResistance);
144 mSubMagnetizingResistor->connect({midpoint, DP::SimNode::GND});
145 addMNASubComponent(mSubMagnetizingResistor,
146 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
147 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
148
149 mSubMagnetizingInductor =
150 std::make_shared<DP::Ph3::Inductor>(**mName + "_mag_ind", mLogLevel);
151 mSubMagnetizingInductor->connect({midpoint, DP::SimNode::GND});
152 addMNASubComponent(mSubMagnetizingInductor,
153 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
154 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
155}

157void DP::Ph3::Transformer::initializeParentFromNodesAndTerminals(
158 Real frequency) {
159 Real omega = 2. * PI * frequency;
160
161 if (mSubMagnetizingInductor) {
162 Real magnetizingSusceptance =
163 std::sqrt(std::pow(mNoLoadCurrent, 2) - std::pow(mNoLoadLoss, 2)) *
164 mRatedPower / std::pow(nominalVoltageAt(mReferenceTerminal), 2);
165 Real magnetizingInductance = 1. / (omega * magnetizingSusceptance);
166 mMagnetizingInductance =
167 Math::singlePhaseParameterToThreePhase(magnetizingInductance);
168 mSubMagnetizingInductor->setParameters(mMagnetizingInductance);
169 }
170
171 MatrixComp secondaryVoltage = initialVoltage(nonReferenceTerminal());
172 mVirtualNodes[0]->setInitialVoltage(secondaryVoltage * mRatioFromReference);
173
174 Complex halfImpedance =
175 Complex(mResistance(0, 0) / 2., omega * mInductance(0, 0) / 2.);
176 Complex magnetizingAdmittance = Complex(0, 0);
177 if (mSubMagnetizingInductor)
178 magnetizingAdmittance =
179 1. / mMagnetizingResistance(0, 0) +
180 1. / Complex(0, omega * mMagnetizingInductance(0, 0));
181
182 MatrixComp referenceVoltage = initialVoltage(mReferenceTerminal);
183 MatrixComp midpointVoltage =
184 (referenceVoltage + mVirtualNodes[0]->initialVoltage()) /
185 (2. + halfImpedance * magnetizingAdmittance);
186 mVirtualNodes[2]->setInitialVoltage(midpointVoltage);
187
188 if (mNumVirtualNodes == 5) {
189 MatrixComp referenceCurrent =
190 (referenceVoltage - midpointVoltage) / halfImpedance;
191 MatrixComp nonReferenceCurrent =
192 (midpointVoltage - mVirtualNodes[0]->initialVoltage()) / halfImpedance;
193 mVirtualNodes[3]->setInitialVoltage(
194 referenceVoltage - referenceCurrent * mResistance(0, 0) / 2.);
195 mVirtualNodes[4]->setInitialVoltage(
196 midpointVoltage - nonReferenceCurrent * mResistance(0, 0) / 2.);
197 }
198
199 **mIntfVoltage = initialVoltage(1) - initialVoltage(0);
200
201 MatrixComp impedance = MatrixComp::Zero(3, 3);
202 for (UInt row = 0; row < 3; row++)
203 for (UInt col = 0; col < 3; col++)
204 impedance(row, col) =
205 Complex(mResistance(row, col), omega * mInductance(row, col));
206
207 MatrixComp impedanceVoltage =
208 mOrientationSign *
209 (mVirtualNodes[0]->initialVoltage() - initialVoltage(mReferenceTerminal));
210 **mIntfCurrent = impedance.inverse() * impedanceVoltage;
211}

213void DP::Ph3::Transformer::mnaParentApplySystemMatrixStamp(
214 SparseMatrixRow &systemMatrix) {
215 if (terminalNotGrounded(mReferenceTerminal)) {
216 for (UInt phase = 0; phase < 3; phase++) {
217 Math::setMatrixElement(
218 systemMatrix,
219 mVirtualNodes[0]->matrixNodeIndex(static_cast<PhaseType>(phase)),
220 mVirtualNodes[1]->matrixNodeIndex(static_cast<PhaseType>(phase)),
221 Complex(-1.0, 0));
222 Math::setMatrixElement(
223 systemMatrix,
224 mVirtualNodes[1]->matrixNodeIndex(static_cast<PhaseType>(phase)),
225 mVirtualNodes[0]->matrixNodeIndex(static_cast<PhaseType>(phase)),
226 Complex(1.0, 0));
227 }
228 }
229 if (terminalNotGrounded(nonReferenceTerminal())) {
230 for (UInt phase = 0; phase < 3; phase++) {
231 Math::setMatrixElement(
232 systemMatrix, matrixNodeIndex(nonReferenceTerminal(), phase),
233 mVirtualNodes[1]->matrixNodeIndex(static_cast<PhaseType>(phase)),
234 std::conj(mRatioFromReference));
235 Math::setMatrixElement(
236 systemMatrix,
237 mVirtualNodes[1]->matrixNodeIndex(static_cast<PhaseType>(phase)),
238 matrixNodeIndex(nonReferenceTerminal(), phase), -mRatioFromReference);
239 }
240 }

272void DP::Ph3::Transformer::mnaCompUpdateVoltage(const Matrix &leftVector) {
273 **mIntfVoltage = MatrixComp::Zero(3, 1);
274 for (UInt phase = 0; phase < 3; phase++) {
275 if (terminalNotGrounded(1))
276 (**mIntfVoltage)(phase, 0) =
277 Math::complexFromVectorElement(leftVector, matrixNodeIndex(1, phase));
278 if (terminalNotGrounded(0))
279 (**mIntfVoltage)(phase, 0) =
280 (**mIntfVoltage)(phase, 0) -
281 Math::complexFromVectorElement(leftVector, matrixNodeIndex(0, phase));
282 }
283}
