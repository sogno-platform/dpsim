33void DP::Ph3::TwoTerminalITypeSSNComp::mnaCompApplyRightSideVectorStamp(
34 Matrix &rightVector) {
35 MatrixComp iHist = -mW.inverse() * mYHist;
36
37 if (terminalNotGrounded(0)) {
38 for (Int phase = 0; phase < 3; ++phase)
39 Math::setVectorElement(rightVector, matrixNodeIndex(0, phase),
40 iHist(phase, 0));
41 }
42 if (terminalNotGrounded(1)) {
43 for (Int phase = 0; phase < 3; ++phase)
44 Math::setVectorElement(rightVector, matrixNodeIndex(1, phase),
45 -iHist(phase, 0));
46 }
47}

49void DP::Ph3::TwoTerminalITypeSSNComp::mnaCompUpdateCurrent(const Matrix &) {
50 // i = W^-1 (v - yHist)
51 **mIntfCurrent = mW.inverse() * (**mIntfVoltage - mYHist);
52}

54void DP::Ph3::TwoTerminalITypeSSNComp::mnaCompUpdateVoltage(
55 const Matrix &leftVector) {
56 **mIntfVoltage = MatrixComp::Zero(3, 1);
57
58 if (terminalNotGrounded(1)) {
59 for (Int phase = 0; phase < 3; ++phase)
60 (**mIntfVoltage)(phase, 0) =
61 Math::complexFromVectorElement(leftVector, matrixNodeIndex(1, phase));
62 }
63 if (terminalNotGrounded(0)) {
64 for (Int phase = 0; phase < 3; ++phase)
65 (**mIntfVoltage)(phase, 0) -=
66 Math::complexFromVectorElement(leftVector, matrixNodeIndex(0, phase));
67 }
