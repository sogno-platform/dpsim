28void DP::SSNComp::setParameters(const Matrix &A, const Matrix &B,
29 const Matrix &C, const Matrix &D,
30 const Matrix &D1) {
31 mParametersSet = false;
32
33 if (A.rows() != A.cols())
34 throw std::invalid_argument("A must be square.");
35
36 if (B.rows() != A.rows() || B.cols() != mInputSize)
37 throw std::invalid_argument("B has invalid dimensions.");
38
39 if (C.rows() != mOutputSize || C.cols() != A.rows())
40 throw std::invalid_argument("C has invalid dimensions.");
41
42 if (D.rows() != mOutputSize || D.cols() != mInputSize)
43 throw std::invalid_argument("D has invalid dimensions.");
44 if (D1.size() != 0 && (D1.rows() != mOutputSize || D1.cols() != mInputSize))
45 throw std::invalid_argument("D1 has invalid dimensions.");
46
47 mA = A;
48 mB = B;
49 mC = C;
50 mD = D;
51 mD1 = D1.size() == 0 ? Matrix::Zero(mOutputSize, mInputSize) : D1;
52
53 **mX = MatrixComp::Zero(mA.rows(), 1);
54
55 mDiscreteA = MatrixComp::Zero(mA.rows(), mA.cols());
56 mDiscreteB = MatrixComp::Zero(mB.rows(), mB.cols());
57
58 mW = MatrixComp::Zero(mOutputSize, mInputSize);
59 mYHist = MatrixComp::Zero(mOutputSize, 1);
60 mStateUpdateInput.resize(mInputSize, 1);
61 mStateUpdate.resize(mA.rows(), 1);
62
63 mParametersSet = true;
64}

66Matrix DP::SSNComp::buildAugmentedA(Real omega) const {
67 const Matrix::Index n = mA.rows();
68 const Matrix wI = omega * Matrix::Identity(n, n);
69
70 Matrix aAug = Matrix::Zero(2 * n, 2 * n);
71 aAug.topLeftCorner(n, n) = mA;
72 aAug.topRightCorner(n, n) = wI;
73 aAug.bottomLeftCorner(n, n) = -wI;
74 aAug.bottomRightCorner(n, n) = mA;
75 return aAug;
76}

78Matrix DP::SSNComp::buildAugmentedB() const {
79 const Matrix::Index n = mA.rows();
80
81 Matrix bAug = Matrix::Zero(2 * n, 2 * mInputSize);
82 bAug.topLeftCorner(n, mInputSize) = mB;
83 bAug.bottomRightCorner(n, mInputSize) = mB;
84 return bAug;
85}

87void DP::SSNComp::recomputeDiscreteModel(Real omega) {
88 const Matrix::Index n = mA.rows();
89
90 // Discretize the real-augmented model with the same helper as EMT-SSN.
91 const Matrix aAug = buildAugmentedA(omega);
92 const Matrix bAug = buildAugmentedB();
93 Matrix dAaug = Matrix::Zero(2 * n, 2 * n);
94 Matrix dBaug = Matrix::Zero(2 * n, 2 * mInputSize);
95 Math::calculateStateSpaceTrapezoidalMatrices(aAug, bAug, mTimeStep, dAaug,
96 dBaug);
97
98 // Fold the [[P, -Q], [Q, P]] rotation blocks back into complex P + jQ.
99 mDiscreteA = dAaug.topLeftCorner(n, n).cast<Complex>() +
100 Complex(0., 1.) * dAaug.bottomLeftCorner(n, n).cast<Complex>();
101 mDiscreteB =
102 dBaug.block(0, 0, n, mInputSize).cast<Complex>() +
103 Complex(0., 1.) * dBaug.block(n, 0, n, mInputSize).cast<Complex>();
104
105 mW = mC.cast<Complex>() * mDiscreteB + mD.cast<Complex>();
106 mW += Complex(2. / mTimeStep, omega) * mD1.cast<Complex>();
107}

109MatrixComp DP::SSNComp::calculateHistoryVector() const {

114void DP::SSNComp::updateHistoryVector(MatrixComp &history) {
115 mHistoryState.noalias() = mDiscreteA * (**mX);
116 mHistoryInput.noalias() = mDiscreteB * (**inputAttribute());
117 mHistoryState += mHistoryInput;
118 history.noalias() = mComplexC * mHistoryState;
119 history.noalias() -=
120 (2. / mTimeStep) * mD1.cast<Complex>() * (**inputAttribute());
121}

123MatrixComp DP::SSNComp::calculateSteadyStateStateFromInput(const MatrixComp &u,
124 Real omega) const {
125 MatrixComp h =
126 Complex(0., omega) * MatrixComp::Identity(mA.rows(), mA.cols()) -
127 mA.cast<Complex>();
128
129 return h.inverse() * mB.cast<Complex>() * u;

138void DP::SSNComp::updateState(const MatrixComp &uOld, const MatrixComp &uNew) {
139 mStateUpdateInput = uNew + uOld;
140 mStateUpdate.noalias() = mDiscreteA * (**mX);
141 mStateUpdate.noalias() += mDiscreteB * mStateUpdateInput;

149void DP::SSNComp::mnaCompInitialize(Real omega, Real timeStep,
150 Attribute<Matrix>::Ptr leftVector) {
151 if (!mParametersSet)
152 throw std::logic_error(
153 "setParameters() must be called before initialization.");
154
155 if (mNumFreqs != 1)
156 throw std::logic_error(
157 "DP SSN components currently support a single carrier frequency.");
158
159 mTimeStep = timeStep;
160 updateMatrixNodeIndices();
161
162 mRightVectorAffectedIndices.clear();
163 const UInt baseRowLimit = static_cast<UInt>(leftVector->get().rows() / 2);
164 for (UInt terminal = 0; terminal < terminalNumberConnected(); ++terminal) {
165 if (terminalNotGrounded(terminal)) {
166 for (UInt index : matrixNodeIndices(terminal))
167 if (index < baseRowLimit)
168 mRightVectorAffectedIndices.push_back(index);
169 }
170 }
171 for (UInt virtualNode = 0; virtualNode < virtualNodesNumber();
172 ++virtualNode) {
173 for (UInt index : virtualMatrixNodeIndices(virtualNode))
174 if (index < baseRowLimit)
175 mRightVectorAffectedIndices.push_back(index);
176 }
177 std::sort(mRightVectorAffectedIndices.begin(),
178 mRightVectorAffectedIndices.end());
179 mRightVectorAffectedIndices.erase(
180 std::unique(mRightVectorAffectedIndices.begin(),
181 mRightVectorAffectedIndices.end()),
182 mRightVectorAffectedIndices.end());
183
184 recomputeDiscreteModel(omega);
185 mComplexC = mC.cast<Complex>();
186 mHistoryState.resize(mA.rows(), 1);
187 mHistoryInput.resize(mA.rows(), 1);
188 mYHist.resize(mOutputSize, 1);
189 updateHistoryVector(mYHist);
190}
