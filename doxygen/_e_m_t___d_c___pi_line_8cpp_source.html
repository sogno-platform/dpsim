38void EMT::DC::PiLine::setParameters(Real seriesResistance,
39 Real seriesInductance,
40 Real parallelCapacitance,
41 Real parallelConductance,
42 Real initialCurrent) {
43 const Real epsilon = std::numeric_limits<Real>::epsilon();
44 if (!Math::isFinite(seriesResistance) || seriesResistance <= DOUBLE_EPSILON)
45 throw std::invalid_argument(
46 "DC pi-line series resistance must be finite and positive.");
47 if (!Math::isFinite(seriesInductance) || seriesInductance <= DOUBLE_EPSILON)
48 throw std::invalid_argument(
49 "DC pi-line series inductance must be finite and positive.");
50 if (!Math::isFinite(parallelCapacitance) || parallelCapacitance < 0.0 ||
51 (parallelCapacitance > 0.0 &&
52 parallelCapacitance / 2.0 <= DOUBLE_EPSILON))
53 throw std::invalid_argument(
54 "DC pi-line shunt capacitance must be zero or safely positive.");
55 if (!Math::isFinite(parallelConductance) || parallelConductance < 0.0 ||
56 (parallelConductance > 0.0 && parallelConductance <= DOUBLE_EPSILON))
57 throw std::invalid_argument(
58 "DC pi-line shunt conductance must be zero or safely positive.");
59 if (!Math::isFinite(initialCurrent))
60 throw std::invalid_argument("DC pi-line initial current must be finite.");
61
62 **mSeriesResistance = seriesResistance;
63 **mSeriesInductance = seriesInductance;
64 **mParallelCapacitance = parallelCapacitance;
65 **mParallelConductance = parallelConductance;
66 **mInitialCurrent = initialCurrent;
67 mParametersSet = true;
68}

70void EMT::DC::PiLine::validateDCTerminals() const {
71 for (UInt terminalIdx = 0; terminalIdx < 2; ++terminalIdx) {
72 const auto terminalNode = const_cast<PiLine *>(this)->node(terminalIdx);
73 if (!terminalNode->isGround() && terminalNode->phaseType() != PhaseType::DC)
74 throw std::invalid_argument(
75 "DC pi-line requires DC nodes or ground at both terminals.");
76 }

79void EMT::DC::PiLine::createSubComponents() {
80 if (mSubCompCreated)
81 return;
82 if (!mParametersSet)
83 throw std::logic_error(
84 "setParameters() must be called before pi-line construction.");
85 validateDCTerminals();
86 mSubCompCreated = true;
87
88 if (!mUseSubcomponents)
89 return;
90
91 mSeriesResistor = Resistor::make(**mName + "_series_R", mLogLevel);
92 mSeriesResistor->setParameters(**mSeriesResistance);
93 mSeriesResistor->connect({mTerminals[0]->node(), mVirtualNodes[0]});
94 addMNASubComponent(mSeriesResistor,
95 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
96 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
97
98 mSeriesInductor = Inductor::make(**mName + "_series_L", mLogLevel);
99 mSeriesInductor->setParameters(**mSeriesInductance, **mInitialCurrent);
100 mSeriesInductor->connect({mVirtualNodes[0], mTerminals[1]->node()});
101 addMNASubComponent(mSeriesInductor,
102 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
103 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
104
105 if (**mParallelConductance > 0.0) {
106 const Real halfShuntResistance = 2.0 / **mParallelConductance;
107 if (!Math::isFinite(halfShuntResistance))
108 throw std::invalid_argument("DC pi-line shunt resistance is non-finite.");
109
110 mShuntResistor0 = Resistor::make(**mName + "_shunt_R0", mLogLevel);
111 mShuntResistor0->setParameters(halfShuntResistance);
112 mShuntResistor0->connect({EMT::SimNode::GND, mTerminals[0]->node()});
113 addMNASubComponent(mShuntResistor0,
114 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
115 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
116
117 mShuntResistor1 = Resistor::make(**mName + "_shunt_R1", mLogLevel);
118 mShuntResistor1->setParameters(halfShuntResistance);
119 mShuntResistor1->connect({EMT::SimNode::GND, mTerminals[1]->node()});
120 addMNASubComponent(mShuntResistor1,
121 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
122 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
123 }
124
125 if (**mParallelCapacitance > 0.0) {
126 const Real halfCapacitance = **mParallelCapacitance / 2.0;
127 mShuntCapacitor0 = Capacitor::make(**mName + "_shunt_C0", mLogLevel);
128 mShuntCapacitor0->setParameters(halfCapacitance);
129 mShuntCapacitor0->connect({EMT::SimNode::GND, mTerminals[0]->node()});
130 addMNASubComponent(mShuntCapacitor0,
131 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
132 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
133
134 mShuntCapacitor1 = Capacitor::make(**mName + "_shunt_C1", mLogLevel);
135 mShuntCapacitor1->setParameters(halfCapacitance);
136 mShuntCapacitor1->connect({EMT::SimNode::GND, mTerminals[1]->node()});
137 addMNASubComponent(mShuntCapacitor1,
138 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
139 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
140 }

143void EMT::DC::PiLine::initializeParentFromNodesAndTerminals(Real) {
144 validateDCTerminals();
145 const Complex voltage0 = initialSingleVoltage(0);
146 const Complex voltage1 = initialSingleVoltage(1);
147 if (!Math::isFinite(voltage0) || !Math::isFinite(voltage1) ||
148 std::abs(voltage0.imag()) > std::numeric_limits<Real>::epsilon() ||
149 std::abs(voltage1.imag()) > std::numeric_limits<Real>::epsilon())
150 throw std::invalid_argument(
151 "DC pi-line initial node voltages must be finite real values.");
152
153 (**mIntfVoltage)(0, 0) = voltage1.real() - voltage0.real();
154 (**mIntfCurrent)(0, 0) = **mInitialCurrent;
155
156 if (!mUseSubcomponents) {
157 (**mDirectState).setZero();
158 (**mDirectState)(0, 0) = **mInitialCurrent;
159 (**mDirectState)(1, 0) = voltage1.real() - voltage0.real() -
160 **mSeriesResistance * **mInitialCurrent;
161 (**mDirectState)(2, 0) = voltage0.real();
162 (**mDirectState)(4, 0) = voltage1.real();
163 } else {
164 const Real virtualVoltage =
165 voltage0.real() + **mSeriesResistance * **mInitialCurrent;
166 mVirtualNodes[0]->setInitialVoltage(Complex(virtualVoltage, 0.0));
167 }
168}

170void EMT::DC::PiLine::mnaParentInitialize(Real, Real timeStep,
171 Attribute<Matrix>::Ptr) {
172 if (mUseSubcomponents)
173 return;
174 mDirectInductorAdmittance = timeStep / (2. * **mSeriesInductance);
175 mDirectSeriesAdmittance =
176 mDirectInductorAdmittance /
177 (1. + **mSeriesResistance * mDirectInductorAdmittance);
178 mDirectCapAdmittance = **mParallelCapacitance / timeStep;
179 const Real denominator = 1. + **mSeriesResistance * mDirectInductorAdmittance;
180 mDirectSeriesHistory = (mDirectInductorAdmittance * (**mDirectState)(1, 0) +
181 (**mDirectState)(0, 0)) /
182 denominator;
183}

185void EMT::DC::PiLine::mnaParentApplySystemMatrixStamp(
186 SparseMatrixRow &systemMatrix) {
187 if (mUseSubcomponents)
188 return;
189 updateMatrixNodeIndices();
190 mDirectTerminalRows = {matrixNodeIndex(0), matrixNodeIndex(1)};
191 mDirectTerminalNotGrounded = {terminalNotGrounded(0), terminalNotGrounded(1)};
192 MNAStampUtils::stampConductance(
193 mDirectSeriesAdmittance, systemMatrix, mDirectTerminalRows[0],
194 mDirectTerminalRows[1], mDirectTerminalNotGrounded[0],
195 mDirectTerminalNotGrounded[1], mSLog);
196 const Real shuntAdmittance =
197 mDirectCapAdmittance + **mParallelConductance / 2.;
198 for (UInt terminal = 0; terminal < 2; ++terminal)
199 MNAStampUtils::stampConductance(
200 shuntAdmittance, systemMatrix, matrixNodeIndex(terminal), 0,
201 mDirectTerminalNotGrounded[terminal], false, mSLog);
202}

204void EMT::DC::PiLine::mnaParentApplyRightSideVectorStamp(Matrix &rightVector) {
205 if (mUseSubcomponents)
206 return;
207 if (mDirectTerminalNotGrounded[0])
208 Math::setVectorElement(rightVector, mDirectTerminalRows[0],
209 mDirectSeriesHistory - mDirectCapHistory0);
210 if (mDirectTerminalNotGrounded[1])
211 Math::setVectorElement(rightVector, mDirectTerminalRows[1],
212 -mDirectSeriesHistory - mDirectCapHistory1);

221 if (mUseSubcomponents) {
222 CompositePowerComp<Real>::mnaAddCompactRightSideVectorStamp(rightVector);

225 for (UInt terminal = 0; terminal < 2; ++terminal)

234void EMT::DC::PiLine::mnaCompUpdateVoltage(const Matrix &leftVector) {
235 (**mIntfVoltage)(0, 0) = 0.0;
236 if (terminalNotGrounded(1))
237 (**mIntfVoltage)(0, 0) =
238 Math::realFromVectorElement(leftVector, matrixNodeIndex(1));
239 if (terminalNotGrounded(0))
240 (**mIntfVoltage)(0, 0) -=
241 Math::realFromVectorElement(leftVector, matrixNodeIndex(0));
242 if (!Math::isFinite((**mIntfVoltage)(0, 0)))
243 throw std::runtime_error(
244 "DC pi-line voltage update produced a non-finite value.");
245}

247void EMT::DC::PiLine::mnaCompUpdateCurrent(const Matrix &) {
248 if (!mUseSubcomponents) {
249 (**mIntfCurrent)(0, 0) =
250 mDirectSeriesAdmittance * (**mIntfVoltage)(0, 0) + mDirectSeriesHistory;
251 return;
252 }
253 **mIntfCurrent = mSeriesInductor->intfCurrent();
254 if (!(**mIntfCurrent).allFinite())
255 throw std::runtime_error(
256 "DC pi-line current update produced a non-finite value.");
257}

259void EMT::DC::PiLine::mnaParentPreStep(Real, Int) {
260 if (!mUseSubcomponents) {
261 const Real denominator =
262 1. + **mSeriesResistance * mDirectInductorAdmittance;
263 mDirectSeriesHistory = (mDirectInductorAdmittance * (**mDirectState)(1, 0) +
264 (**mDirectState)(0, 0)) /
265 denominator;
266 mDirectCapHistory0 = -((**mDirectState)(3, 0) +
267 mDirectCapAdmittance * (**mDirectState)(2, 0));
268 mDirectCapHistory1 = -((**mDirectState)(5, 0) +
269 mDirectCapAdmittance * (**mDirectState)(4, 0));
270 }
271 mnaCompApplyRightSideVectorStamp(**mRightVector);
272}

274void EMT::DC::PiLine::mnaParentPostStep(Real, Int,
275 Attribute<Matrix>::Ptr &leftVector) {
276 if (mUseSubcomponents) {
277 mnaCompUpdateVoltage(**leftVector);
278 mnaCompUpdateCurrent(**leftVector);
279 return;
280 }
281
282 const Real voltage0 =
283 mDirectTerminalNotGrounded[0]
284 ? Math::realFromVectorElement(**leftVector, mDirectTerminalRows[0])
285 : 0.;
286 const Real voltage1 =
287 mDirectTerminalNotGrounded[1]
288 ? Math::realFromVectorElement(**leftVector, mDirectTerminalRows[1])
289 : 0.;
290 (**mIntfVoltage)(0, 0) = voltage1;
291 (**mIntfVoltage)(0, 0) -= voltage0;
292 if (!Math::isFinite((**mIntfVoltage)(0, 0)))
293 throw std::runtime_error(
294 "DC pi-line voltage update produced a non-finite value.");
295 mnaCompUpdateCurrent(**leftVector);
296 (**mDirectState)(0, 0) = (**mIntfCurrent)(0, 0);
297 (**mDirectState)(1, 0) =
298 (**mIntfVoltage)(0, 0) - **mSeriesResistance * (**mIntfCurrent)(0, 0);
299 (**mDirectState)(2, 0) = voltage0;
300 (**mDirectState)(3, 0) = mDirectCapAdmittance * voltage0 + mDirectCapHistory0;
301 (**mDirectState)(4, 0) = voltage1;
302 (**mDirectState)(5, 0) = mDirectCapAdmittance * voltage1 + mDirectCapHistory1;
303}
