15void EMT::DC::TwoTerminalITypeSSNComp::validateDCTerminals() const {
16 for (UInt terminalIdx = 0; terminalIdx < 2; ++terminalIdx) {
17 const auto terminalNode =
18 const_cast<TwoTerminalITypeSSNComp *>(this)->node(terminalIdx);
19 if (!terminalNode->isGround() && terminalNode->phaseType() != PhaseType::DC)
20 throw std::invalid_argument(
21 "DC SSN components require DC nodes or ground at both terminals.");
22 }
23}

25Real EMT::DC::TwoTerminalITypeSSNComp::companionConductance() const {
26 if (mW.rows() != 1 || mW.cols() != 1 || !Math::isFinite(mW(0, 0)) ||
27 std::abs(mW(0, 0)) <= DOUBLE_EPSILON)
28 throw std::runtime_error(
29 "DC I-type SSN component has a singular companion impedance.");
30 const Real conductance = 1.0 / mW(0, 0);
31 if (!Math::isFinite(conductance))
32 throw std::runtime_error(
33 "DC I-type SSN companion conductance is non-finite.");
34 return conductance;
35}

60void EMT::DC::TwoTerminalITypeSSNComp::mnaCompUpdateCurrent(const Matrix &) {
61 (**mIntfCurrent)(0, 0) =

68void EMT::DC::TwoTerminalITypeSSNComp::mnaCompUpdateVoltage(
69 const Matrix &leftVector) {

77 if (!Math::isFinite((**mIntfVoltage)(0, 0)))
78 throw std::runtime_error(
79 "DC SSN voltage update produced a non-finite value.");
