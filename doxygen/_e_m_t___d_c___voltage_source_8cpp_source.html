33void EMT::DC::VoltageSource::validateDCTerminals() const {
34 for (UInt terminalIdx = 0; terminalIdx < 2; ++terminalIdx) {
35 const auto terminalNode =
36 const_cast<VoltageSource *>(this)->node(terminalIdx);
37 if (!terminalNode->isGround() && terminalNode->phaseType() != PhaseType::DC)
38 throw std::invalid_argument(
39 "DC voltage source requires DC nodes or ground at both terminals.");
40 }
41}

62void EMT::DC::VoltageSource::mnaCompApplySystemMatrixStamp(
63 SparseMatrixRow &systemMatrix) {
64 const UInt branchIdx = mVirtualNodes.at(0)->matrixNodeIndices().at(0);
65 if (terminalNotGrounded(0)) {
66 Math::addToMatrixElement(systemMatrix, matrixNodeIndex(0), branchIdx, -1.0);
67 Math::addToMatrixElement(systemMatrix, branchIdx, matrixNodeIndex(0), -1.0);
68 }
69 if (terminalNotGrounded(1)) {
70 Math::addToMatrixElement(systemMatrix, matrixNodeIndex(1), branchIdx, 1.0);
71 Math::addToMatrixElement(systemMatrix, branchIdx, matrixNodeIndex(1), 1.0);
72 }
73}

82void EMT::DC::VoltageSource::mnaCompUpdateCurrent(const Matrix &leftVector) {
83 (**mIntfCurrent)(0, 0) = Math::realFromVectorElement(
84 leftVector, mVirtualNodes.at(0)->matrixNodeIndices().at(0));
85 if (!Math::isFinite((**mIntfCurrent)(0, 0)))
86 throw std::runtime_error(
87 "DC voltage-source current contains a non-finite value.");
88}
