23void EMT::Ph1::SSN::Full_Serial_RLC::setParameters(Real resistance,
24 Real inductance,
25 Real capacitance) {
26 Base::Ph1::Resistor::setParameters(resistance);
27 Base::Ph1::Inductor::setParameters(inductance);
28 Base::Ph1::Capacitor::setParameters(capacitance);
29
30 Matrix a = Matrix::Zero(2, 2);
31 a(0, 1) = 1. / capacitance;
32 a(1, 0) = -1. / inductance;
33 a(1, 1) = -resistance / inductance;
34 Matrix b = Matrix::Zero(2, 1);
35 b(1, 0) = 1. / inductance;
36 Matrix c = Matrix::Zero(1, 2);
37 c(0, 1) = 1.;
38 SSNComp::setParameters(a, b, c, Matrix::Zero(1, 1));
39}
