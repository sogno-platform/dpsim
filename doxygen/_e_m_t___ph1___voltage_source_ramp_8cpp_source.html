53void EMT::Ph1::VoltageSourceRamp::initialize(Matrix frequencies) {
54 SimPowerComp<Real>::initialize(frequencies);
55
56 if (**mVoltageRef == Complex(0, 0))
57 **mVoltageRef =
58 RMS3PH_TO_PEAK1PH * (initialSingleVoltage(1) - initialSingleVoltage(0));
59
60 if (mUseSubcomponents) {
61 mSubVoltageSource = VoltageSource::make(**mName + "_src", mLogLevel);
62 mSubVoltageSource->setParameters(**mVoltageRef, 0);
63 mSubVoltageSource->connect({node(0), node(1)});
64 mSubVoltageSource->setVirtualNodeAt(mVirtualNodes[0], 0);
65 mSubVoltageSource->initialize(frequencies);
66 addMNASubComponent(mSubVoltageSource,
67 MNA_SUBCOMP_TASK_ORDER::TASK_AFTER_PARENT,
68 MNA_SUBCOMP_TASK_ORDER::NO_TASK, true);
69 }

72void EMT::Ph1::VoltageSourceRamp::updateState(Real time) {
73 Real voltageAbs = Math::abs(**mVoltageRef);
74 Real voltagePhase = Math::phase(**mVoltageRef);
75 (**mIntfVoltage)(0, 0) = voltageAbs * cos(voltagePhase + **mSrcFreq * time);
76
77 if (time >= mSwitchTime && time < mSwitchTime + mRampTime) {
78 voltageAbs = Math::abs(**mVoltageRef +
79 (time - mSwitchTime) / mRampTime * mAddVoltage);
80 voltagePhase = Math::phase(**mVoltageRef +
81 (time - mSwitchTime) / mRampTime * mAddVoltage);
82 Real fadeInOut =
83 0.5 + 0.5 * sin((time - mSwitchTime) / mRampTime * PI + -PI / 2);
84 (**mIntfVoltage)(0, 0) =
85 voltageAbs *
86 cos(voltagePhase + (**mSrcFreq + fadeInOut * mAddSrcFreq) * time);
87 } else if (time >= mSwitchTime + mRampTime) {
88 voltageAbs = Math::abs(**mVoltageRef + mAddVoltage);
89 voltagePhase = Math::phase(**mVoltageRef + mAddVoltage);
90 (**mIntfVoltage)(0, 0) =
91 voltageAbs * cos(voltagePhase + (**mSrcFreq + mAddSrcFreq) * time);
92 }
93}

105void EMT::Ph1::VoltageSourceRamp::mnaParentApplySystemMatrixStamp(
106 SparseMatrixRow &systemMatrix) {
107 if (mUseSubcomponents)
108 return;
109 if (terminalNotGrounded(0)) {
110 Math::addToMatrixElement(systemMatrix, matrixNodeIndex(0),
111 mVirtualNodes[0]->matrixNodeIndex(), -1.);
112 Math::addToMatrixElement(systemMatrix, mVirtualNodes[0]->matrixNodeIndex(),
113 matrixNodeIndex(0), -1.);
114 }
115 if (terminalNotGrounded(1)) {
116 Math::addToMatrixElement(systemMatrix, matrixNodeIndex(1),
117 mVirtualNodes[0]->matrixNodeIndex(), 1.);
118 Math::addToMatrixElement(systemMatrix, mVirtualNodes[0]->matrixNodeIndex(),
119 matrixNodeIndex(1), 1.);
120 }
121}
