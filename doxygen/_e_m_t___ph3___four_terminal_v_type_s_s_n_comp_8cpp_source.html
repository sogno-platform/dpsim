42void EMT::Ph3::FourTerminalVTypeSSNComp::mnaCompUpdateVoltage(
43 const Matrix &leftVector) {
44
45 Matrix vTerm = Matrix::Zero(12, 1);
46
47 for (UInt terminal = 0; terminal < 4; ++terminal) {
48 if (!terminalNotGrounded(terminal))
49 continue;
50
51 for (UInt phase = 0; phase < 3; ++phase) {
52 vTerm(3 * terminal + phase, 0) = Math::realFromVectorElement(
53 leftVector, matrixNodeIndex(terminal, phase));
54 }
55 }
56
57 **mIntfVoltage = mTerminalIncidence * vTerm;
58}

60MatrixComp
61EMT::Ph3::FourTerminalVTypeSSNComp::buildInitialInputFromNodes(Real) {
62 MatrixComp vInitABC = MatrixComp::Zero(6, 1);
63
64 // Interface voltage convention:
65 // u = [v_terminal2 - v_terminal0;
66 // v_terminal3 - v_terminal1]
67
68 // First three-phase interface voltage: v20 = v2 - v0
69 vInitABC(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(2) -
70 RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
71 vInitABC(1, 0) = vInitABC(0, 0) * SHIFT_TO_PHASE_B;
72 vInitABC(2, 0) = vInitABC(0, 0) * SHIFT_TO_PHASE_C;
73
74 // Second three-phase interface voltage: v31 = v3 - v1
75 vInitABC(3, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(3) -
76 RMS3PH_TO_PEAK1PH * initialSingleVoltage(1);
77 vInitABC(4, 0) = vInitABC(3, 0) * SHIFT_TO_PHASE_B;
78 vInitABC(5, 0) = vInitABC(3, 0) * SHIFT_TO_PHASE_C;
79
80 return vInitABC;
81}

83void EMT::Ph3::FourTerminalVTypeSSNComp::mnaCompApplySystemMatrixStamp(
84 SparseMatrixRow &systemMatrix) {
85
86 const Matrix yTerm = mTerminalIncidence.transpose() * mW * mTerminalIncidence;
87
88 for (UInt terminalRow = 0; terminalRow < 4; ++terminalRow) {
89 if (!terminalNotGrounded(terminalRow))
90 continue;
91
92 for (UInt phaseRow = 0; phaseRow < 3; ++phaseRow) {
93 const UInt row = 3 * terminalRow + phaseRow;
94
95 for (UInt terminalCol = 0; terminalCol < 4; ++terminalCol) {
96 if (!terminalNotGrounded(terminalCol))
97 continue;
98
99 for (UInt phaseCol = 0; phaseCol < 3; ++phaseCol) {
100 const UInt col = 3 * terminalCol + phaseCol;
101
102 Math::addToMatrixElement(
103 systemMatrix, matrixNodeIndex(terminalRow, phaseRow),
104 matrixNodeIndex(terminalCol, phaseCol), yTerm(row, col));
105 }
106 }
107 }
108 }
