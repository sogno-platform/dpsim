37void EMT::Ph3::HalfDecouplingLine::setParameters(Matrix resistance,
38 Matrix inductance,
39 Matrix capacitance) {
40
41 mResistance = resistance;
42 mInductance = inductance;
43 mCapacitance = capacitance;
44
45 mSurgeImpedance = (inductance * capacitance.inverse()).array().sqrt();
46 mDelay = (inductance.array() * capacitance.array()).sqrt().maxCoeff();
47
48 **mSrcRes = mSurgeImpedance + mResistance / 4;
49 mSrcResInverse = (**mSrcRes).inverse();
50 mSrcResSquaredInverse = ((**mSrcRes) * (**mSrcRes)).inverse();
51
52 SPDLOG_LOGGER_INFO(mSLog, "surge impedance: {}", mSurgeImpedance);
53 SPDLOG_LOGGER_INFO(mSLog, "delay: {}", mDelay);
54
55 mParametersSet = true;
56}

113void EMT::Ph3::HalfDecouplingLine::mnaParentInitialize(
114 Real omega, Real timeStep, Attribute<Matrix>::Ptr leftVector) {
115 if (mDelay < timeStep)
116 throw SystemError("Timestep too large for decoupling");
117
118 mBufSize = static_cast<UInt>(ceil(mDelay / timeStep));
119 mAlpha = 1 - (mBufSize - mDelay / timeStep);
120 SPDLOG_LOGGER_INFO(mSLog, "bufsize {} alpha {}", mBufSize, mAlpha);
121
122 MatrixComp voltNear = -initialVoltage(0);
123 MatrixComp voltFar = (**mReceivingVolt).cast<Complex>();
124
125 MatrixComp seriesAdmittance =
126 (mResistance + Complex(0, omega) * mInductance).inverse();
127 MatrixComp initAdmittance =
128 seriesAdmittance + Complex(0, omega) * mCapacitance / 2;
129 MatrixComp curNear = initAdmittance * voltNear - seriesAdmittance * voltFar;
130
131 SPDLOG_LOGGER_INFO(mSLog, "initial voltage: v_k {}", voltNear);
132 SPDLOG_LOGGER_INFO(mSLog, "initial current: i_k {}", curNear);
133
134 mVoltBuf = voltNear.real().transpose().replicate(mBufSize, 1);
135 mCurBuf = curNear.real().transpose().replicate(mBufSize, 1);
136
137 **mSendingVolt = voltNear.real();
138 **mSendingCur = curNear.real();
139}

141MatrixFixedSize<3, 1>
142EMT::Ph3::HalfDecouplingLine::interpolate(const Matrix &data) const {
143 const UInt nextIdx = (mBufIdx + 1 == mBufSize) ? 0 : mBufIdx + 1;
144 MatrixFixedSize<3, 1> result;
145 const Real nextWeight = 1 - mAlpha;
146 for (UInt phase = 0; phase < 3; ++phase)
147 result(phase) =
148 mAlpha * data(mBufIdx, phase) + nextWeight * data(nextIdx, phase);
149 return result;
150}

152void EMT::Ph3::HalfDecouplingLine::step(Real time, Int timeStepCount) {
153 const Matrix &voltNear = **mSendingVolt;
154 const Matrix &curNear = **mSendingCur;
155 const Matrix &voltFar = **mReceivingVolt;
156 const Matrix &curFar = **mReceivingCur;
157
158 if (timeStepCount == 0) {
159 **mSrcCtrledCurrent = curNear - mSrcResInverse * voltNear;
160 } else {
161 **mSrcCtrledCurrent =
162 -mSurgeImpedance * mSrcResSquaredInverse *
163 (voltFar + (mSurgeImpedance - mResistance / 4) * curFar) -
164 mResistance / 4 * mSrcResSquaredInverse *
165 (voltNear + (mSurgeImpedance - mResistance / 4) * curNear);
166 }
167
168 if (mUseSubcomponents)
169 mSubCtrledCurrentSource->mCurrentRef->set(**mSrcCtrledCurrent);
170}

172void EMT::Ph3::HalfDecouplingLine::postStep() {
173 if (mUseSubcomponents) {
174 mVoltBuf.row(mBufIdx) = -mSubRes->intfVoltage().transpose();
175 mCurBuf.row(mBufIdx) =
176 -mSubRes->intfCurrent().transpose() + (**mSrcCtrledCurrent).transpose();
177 } else {
178 mVoltBuf.row(mBufIdx) = -(**mIntfVoltage).transpose();
179 mCurBuf.row(mBufIdx) = (**mIntfCurrent).transpose();
180 }
181
182 mBufIdx++;
183 if (mBufIdx == mBufSize)
184 mBufIdx = 0;
185
186 **mSendingVolt = interpolate(mVoltBuf);
187 **mSendingCur = interpolate(mCurBuf);
188}

215void EMT::Ph3::HalfDecouplingLine::mnaCompUpdateCurrent(
216 const Matrix &leftVector) {
217 if (mUseSubcomponents) {
218 **mIntfCurrent = -mSubRes->intfCurrent() + **mSrcCtrledCurrent;
