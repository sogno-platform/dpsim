17EMT::Ph3::PiLine::PiLine(String uid, String name, Bool useSubcomponents,
18 Logger::Level logLevel)
19 : Base::Ph3::PiLine(mAttributes),
20 CompositePowerComp<Real>(uid, name, true, true, logLevel) {
21 mUseSubcomponents = useSubcomponents;
22 mDirectState = mAttributes->create<Matrix>("direct_state");
23 **mDirectState = Matrix::Zero(18, 1);
24 mPhaseType = PhaseType::ABC;
25 // The direct path still keeps the series midpoint as an MNA unknown. This
26 // preserves the exact conventional R-L companion equations while removing
27 // the separate child components and their scheduler tasks.
28 setVirtualNodeNumber(1);
29 setTerminalNumber(2);
30
31 SPDLOG_LOGGER_INFO(mSLog, "Create {} {}", this->type(), name);
32 **mIntfVoltage = Matrix::Zero(3, 1);
33 **mIntfCurrent = Matrix::Zero(3, 1);
34
35 mSLog->flush();
36}

46void EMT::Ph3::PiLine::createSubComponents() {
47 if (mSubCompCreated)
48 return;
49 mSubCompCreated = true;
50
51 if (!mUseSubcomponents) {
52 if ((**mParallelCond)(0, 0) <= 0.) {
53 **mParallelCond = Matrix::Zero(3, 3);
54 (**mParallelCond).diagonal().setConstant(1e-6);
55 }
56 return;
57 }
58
59 // By default there is always a small conductance to ground to
60 // avoid problems with floating nodes.
61 Matrix defaultParallelCond = Matrix::Zero(3, 3);
62 defaultParallelCond << 1e-6, 0, 0, 0, 1e-6, 0, 0, 0, 1e-6;
63 **mParallelCond =
64 ((**mParallelCond)(0, 0) > 0) ? **mParallelCond : defaultParallelCond;
65
66 // Create series sub components
67 mSubSeriesResistor =
68 std::make_shared<EMT::Ph3::Resistor>(**mName + "_res", mLogLevel);
69 mSubSeriesResistor->setParameters(**mSeriesRes);
70 mSubSeriesResistor->connect({mTerminals[0]->node(), mVirtualNodes[0]});
71 addMNASubComponent(mSubSeriesResistor,
72 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
73 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
74
75 mSubSeriesInductor =
76 std::make_shared<EMT::Ph3::Inductor>(**mName + "_ind", mLogLevel);
77 mSubSeriesInductor->setParameters(**mSeriesInd);
78 mSubSeriesInductor->connect({mVirtualNodes[0], mTerminals[1]->node()});
79 addMNASubComponent(mSubSeriesInductor,
80 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
81 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
82
83 // Create parallel sub components
84 mSubParallelResistor0 =
85 std::make_shared<EMT::Ph3::Resistor>(**mName + "_con0", mLogLevel);
86 mSubParallelResistor0->setParameters(2. * (**mParallelCond).inverse());
87 mSubParallelResistor0->connect(
88 SimNode::List{SimNode::GND, mTerminals[0]->node()});
89 addMNASubComponent(mSubParallelResistor0,
90 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
91 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
92
93 mSubParallelResistor1 =
94 std::make_shared<EMT::Ph3::Resistor>(**mName + "_con1", mLogLevel);
95 mSubParallelResistor1->setParameters(2. * (**mParallelCond).inverse());
96 mSubParallelResistor1->connect(
97 SimNode::List{SimNode::GND, mTerminals[1]->node()});
98 addMNASubComponent(mSubParallelResistor1,
99 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
100 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
101
102 if ((**mParallelCap)(0, 0) > 0) {
103 mSubParallelCapacitor0 =
104 std::make_shared<EMT::Ph3::Capacitor>(**mName + "_cap0", mLogLevel);
105 mSubParallelCapacitor0->setParameters(**mParallelCap / 2.);
106 mSubParallelCapacitor0->connect(
107 SimNode::List{SimNode::GND, mTerminals[0]->node()});
108 addMNASubComponent(mSubParallelCapacitor0,
109 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
110 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
111
112 mSubParallelCapacitor1 =
113 std::make_shared<EMT::Ph3::Capacitor>(**mName + "_cap1", mLogLevel);
114 mSubParallelCapacitor1->setParameters(**mParallelCap / 2.);
115 mSubParallelCapacitor1->connect(
116 SimNode::List{SimNode::GND, mTerminals[1]->node()});
117 addMNASubComponent(mSubParallelCapacitor1,
118 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
119 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
120 }

123void EMT::Ph3::PiLine::initializeParentFromNodesAndTerminals(Real frequency) {
124 if (!mUseSubcomponents) {
125 const Real omega = 2. * PI * frequency;
126 MatrixComp v0 = MatrixComp::Zero(3, 1), v1 = MatrixComp::Zero(3, 1);
127 v0(0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
128 v1(0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(1);
129 v0(1) = v0(0) * SHIFT_TO_PHASE_B;
130 v0(2) = v0(0) * SHIFT_TO_PHASE_C;
131 v1(1) = v1(0) * SHIFT_TO_PHASE_B;
132 v1(2) = v1(0) * SHIFT_TO_PHASE_C;
133 const MatrixComp voltage = v1 - v0;
134 const MatrixComp impedance =
135 (**mSeriesRes).cast<Complex>() +
136 Complex(0., omega) * (**mSeriesInd).cast<Complex>();
137 const MatrixComp current = impedance.inverse() * voltage;
138 const MatrixComp halfCap = ((**mParallelCap) / 2.).cast<Complex>();
139 **mIntfVoltage = voltage.real();
140 **mIntfCurrent = current.real();
141 mVirtualNodes[0]->setInitialVoltage(
142 PEAK1PH_TO_RMS3PH *
143 (v0 + (**mSeriesRes).cast<Complex>() * current).real());
144 (**mDirectState).setZero();
145 (**mDirectState).block<3, 1>(0, 0) = current.real();
146 (**mDirectState).block<3, 1>(3, 0) =
147 (voltage - (**mSeriesRes).cast<Complex>() * current).real();
148 (**mDirectState).block<3, 1>(6, 0) = v0.real();
149 (**mDirectState).block<3, 1>(9, 0) =
150 (Complex(0., omega) * halfCap * v0).real();
151 (**mDirectState).block<3, 1>(12, 0) = v1.real();
152 (**mDirectState).block<3, 1>(15, 0) =
153 (Complex(0., omega) * halfCap * v1).real();
154 return;
155 }
156 // Static calculation
157 Real omega = 2. * PI * frequency;
158 MatrixComp impedance = MatrixComp::Zero(3, 3);
159 impedance << Complex((**mSeriesRes)(0, 0), omega * (**mSeriesInd)(0, 0)),
160 Complex((**mSeriesRes)(0, 1), omega * (**mSeriesInd)(0, 1)),
161 Complex((**mSeriesRes)(0, 2), omega * (**mSeriesInd)(0, 2)),
162 Complex((**mSeriesRes)(1, 0), omega * (**mSeriesInd)(1, 0)),
163 Complex((**mSeriesRes)(1, 1), omega * (**mSeriesInd)(1, 1)),
164 Complex((**mSeriesRes)(1, 2), omega * (**mSeriesInd)(1, 2)),
165 Complex((**mSeriesRes)(2, 0), omega * (**mSeriesInd)(2, 0)),
166 Complex((**mSeriesRes)(2, 1), omega * (**mSeriesInd)(2, 1)),
167 Complex((**mSeriesRes)(2, 2), omega * (**mSeriesInd)(2, 2));
168
169 MatrixComp vInitABC = MatrixComp::Zero(3, 1);
170 vInitABC(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(1) -
171 RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
172 vInitABC(1, 0) = vInitABC(0, 0) * SHIFT_TO_PHASE_B;
173 vInitABC(2, 0) = vInitABC(0, 0) * SHIFT_TO_PHASE_C;
174 MatrixComp iInit = impedance.inverse() * vInitABC;
175 **mIntfCurrent = iInit.real();
176 **mIntfVoltage = vInitABC.real();
177
178 // Initialization of virtual node
179 // Initial voltage of phase B,C is set after A
180 MatrixComp vInitTerm0 = MatrixComp::Zero(3, 1);
181 vInitTerm0(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
182 vInitTerm0(1, 0) = vInitTerm0(0, 0) * SHIFT_TO_PHASE_B;
183 vInitTerm0(2, 0) = vInitTerm0(0, 0) * SHIFT_TO_PHASE_C;
184
185 mVirtualNodes[0]->setInitialVoltage(PEAK1PH_TO_RMS3PH *
186 (vInitTerm0 + **mSeriesRes * iInit));
187
188 SPDLOG_LOGGER_DEBUG(mSLog,
189 "\n--debug--"
190 "\n seriesRes: {:s}"
191 "\n seriesInd: {:s}"
192 "\n Impedance: {:s}"
193 "\n vInit: {:s}"
194 "\n iInit: {:s}",
195 Logger::matrixToString(**mSeriesRes),
196 Logger::matrixToString(**mSeriesInd),
197 Logger::matrixCompToString(impedance),
198 Logger::matrixCompToString(vInitABC),
199 Logger::matrixCompToString(iInit));
200
201 SPDLOG_LOGGER_INFO(
202 mSLog,
203 "\n--- Initialization from powerflow ---"
204 "\nVoltage across: {:s}"
205 "\nCurrent: {:s}"
206 "\nTerminal 0 voltage: {:s}"
207 "\nTerminal 1 voltage: {:s}"
208 "\nVirtual Node 1 voltage: {:s}"
209 "\n--- Initialization from powerflow finished ---",
210 Logger::matrixToString(**mIntfVoltage),
211 Logger::matrixToString(**mIntfCurrent),
212 Logger::phasorToString(RMS3PH_TO_PEAK1PH * initialSingleVoltage(0)),
213 Logger::phasorToString(RMS3PH_TO_PEAK1PH * initialSingleVoltage(1)),
214 Logger::phasorToString(mVirtualNodes[0]->initialSingleVoltage()));
215 mSLog->flush();
216}

218void EMT::Ph3::PiLine::mnaParentInitialize(Real, Real timeStep,
219 Attribute<Matrix>::Ptr) {
220 if (mUseSubcomponents)
221 return;
222 mDirectResistanceConductance = (**mSeriesRes).inverse();
223 mDirectInductorAdmittance = (timeStep / 2.) * (**mSeriesInd).inverse();
224 mDirectCapAdmittance = (**mParallelCap) / timeStep;
225}

227void EMT::Ph3::PiLine::mnaParentApplySystemMatrixStamp(
228 SparseMatrixRow &systemMatrix) {
229 if (mUseSubcomponents)
230 return;
231 updateMatrixNodeIndices();
232 const UInt midpoint = mVirtualNodes[0]->matrixNodeIndex(PhaseType::A);
233 MNAStampUtils::stampConductanceMatrix(
234 mDirectResistanceConductance, systemMatrix, matrixNodeIndex(0), midpoint,
235 terminalNotGrounded(0), true, mSLog);
236 MNAStampUtils::stampConductanceMatrix(mDirectInductorAdmittance, systemMatrix,
237 midpoint, matrixNodeIndex(1), true,
238 terminalNotGrounded(1), mSLog);
239 const Matrix shunt = mDirectCapAdmittance + (**mParallelCond) / 2.;
240 for (UInt terminal = 0; terminal < 2; ++terminal)
241 MNAStampUtils::stampConductanceMatrix(
242 shunt, systemMatrix, matrixNodeIndex(terminal), 0,
243 terminalNotGrounded(terminal), false, mSLog);
244}

246void EMT::Ph3::PiLine::mnaParentApplyRightSideVectorStamp(Matrix &rightVector) {
247 if (mUseSubcomponents)
248 return;
249 const UInt midpoint = mVirtualNodes[0]->matrixNodeIndex(PhaseType::A);
250 for (UInt phase = 0; phase < 3; ++phase)
251 Math::setVectorElement(rightVector, midpoint + phase,
252 mDirectSeriesHistory(phase));
253 if (terminalNotGrounded(1))
254 for (UInt phase = 0; phase < 3; ++phase)
255 Math::setVectorElement(rightVector, matrixNodeIndex(1, phase),
256 -mDirectSeriesHistory(phase) -
257 mDirectCapHistory1(phase));
258 if (terminalNotGrounded(0))
259 for (UInt phase = 0; phase < 3; ++phase)
260 Math::setVectorElement(rightVector, matrixNodeIndex(0, phase),
261 -mDirectCapHistory0(phase));
262}

269void EMT::Ph3::PiLine::mnaAddCompactRightSideVectorStamp(Matrix &rightVector) {
270 if (mUseSubcomponents) {
271 CompositePowerComp<Real>::mnaAddCompactRightSideVectorStamp(rightVector);
272 return;
273 }
274 const UInt midpoint = mVirtualNodes[0]->matrixNodeIndex(PhaseType::A);
275 for (UInt phase = 0; phase < 3; ++phase)
276 Math::addToVectorElement(
277 rightVector, midpoint + phase,
278 Math::realFromVectorElement(**mRightVector, midpoint + phase));
279 for (UInt terminal = 0; terminal < 2; ++terminal) {
280 if (!terminalNotGrounded(terminal))
281 continue;
282 for (UInt phase = 0; phase < 3; ++phase) {
283 const UInt idx = matrixNodeIndex(terminal, phase);
284 Math::addToVectorElement(
285 rightVector, idx, Math::realFromVectorElement(**mRightVector, idx));
286 }
287 }

303void EMT::Ph3::PiLine::mnaParentPreStep(Real time, Int timeStepCount) {
304 if (!mUseSubcomponents) {
305 mDirectInductorHistory.noalias() =
306 mDirectInductorAdmittance * (**mDirectState).block<3, 1>(3, 0);
307 mDirectInductorHistory += (**mDirectState).block<3, 1>(0, 0);
308 mDirectSeriesHistory = mDirectInductorHistory;
309 mDirectCapHistory0 =
310 -(**mDirectState).block<3, 1>(9, 0) -
311 mDirectCapAdmittance * (**mDirectState).block<3, 1>(6, 0);
312 mDirectCapHistory1 =
313 -(**mDirectState).block<3, 1>(15, 0) -
314 mDirectCapAdmittance * (**mDirectState).block<3, 1>(12, 0);
315 }
316 mnaCompApplyRightSideVectorStamp(**mRightVector);
317}

331void EMT::Ph3::PiLine::mnaParentPostStep(Real time, Int timeStepCount,
332 Attribute<Matrix>::Ptr &leftVector) {
333 if (!mUseSubcomponents) {
334 const auto voltage = [&](UInt terminal) {
335 Eigen::Matrix<Real, 3, 1> result = Eigen::Matrix<Real, 3, 1>::Zero();
336 if (terminalNotGrounded(terminal))
337 for (UInt phase = 0; phase < 3; ++phase)
338 result(phase) = Math::realFromVectorElement(
339 **leftVector, matrixNodeIndex(terminal, phase));
340 return result;
341 };
342 const auto v0 = voltage(0), v1 = voltage(1);
343 Eigen::Matrix<Real, 3, 1> vmid;
344 for (UInt phase = 0; phase < 3; ++phase)
345 vmid(phase) = Math::realFromVectorElement(
346 **leftVector,
347 mVirtualNodes[0]->matrixNodeIndex(static_cast<PhaseType>(phase)));
348 **mIntfVoltage = v1 - v0;
349 **mIntfCurrent =
350 mDirectInductorAdmittance * (v1 - vmid) + mDirectSeriesHistory;
351 (**mDirectState).block<3, 1>(0, 0) = **mIntfCurrent;
352 (**mDirectState).block<3, 1>(3, 0) = v1 - vmid;
353 (**mDirectState).block<3, 1>(6, 0) = v0;
354 (**mDirectState).block<3, 1>(9, 0) =
355 mDirectCapAdmittance * v0 + mDirectCapHistory0;
356 (**mDirectState).block<3, 1>(12, 0) = v1;
357 (**mDirectState).block<3, 1>(15, 0) =
358 mDirectCapAdmittance * v1 + mDirectCapHistory1;
359 return;
360 }
361 mnaCompUpdateVoltage(**leftVector);
362 mnaCompUpdateCurrent(**leftVector);
363}

365void EMT::Ph3::PiLine::mnaCompUpdateVoltage(const Matrix &leftVector) {
366 // v1 - v0
367 (**mIntfVoltage).setZero();
368 if (terminalNotGrounded(1)) {
369 (**mIntfVoltage)(0, 0) =
370 Math::realFromVectorElement(leftVector, matrixNodeIndex(1, 0));
371 (**mIntfVoltage)(1, 0) =
372 Math::realFromVectorElement(leftVector, matrixNodeIndex(1, 1));
373 (**mIntfVoltage)(2, 0) =
374 Math::realFromVectorElement(leftVector, matrixNodeIndex(1, 2));
375 }
376 if (terminalNotGrounded(0)) {
377 (**mIntfVoltage)(0, 0) =
378 (**mIntfVoltage)(0, 0) -
379 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 0));
380 (**mIntfVoltage)(1, 0) =
381 (**mIntfVoltage)(1, 0) -
382 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 1));
383 (**mIntfVoltage)(2, 0) =
384 (**mIntfVoltage)(2, 0) -
385 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 2));
386 }
387}

389void EMT::Ph3::PiLine::mnaCompUpdateCurrent(const Matrix &leftVector) {
390 if (!mUseSubcomponents) {
391 Eigen::Matrix<Real, 3, 1> midpointVoltage;
392 for (UInt phase = 0; phase < 3; ++phase)
393 midpointVoltage(phase) = Math::realFromVectorElement(
394 leftVector,
395 mVirtualNodes[0]->matrixNodeIndex(static_cast<PhaseType>(phase)));
396 Eigen::Matrix<Real, 3, 1> terminal1 = Eigen::Matrix<Real, 3, 1>::Zero();
397 if (terminalNotGrounded(1))
398 for (UInt phase = 0; phase < 3; ++phase)
399 terminal1(phase) =
400 Math::realFromVectorElement(leftVector, matrixNodeIndex(1, phase));
401 **mIntfCurrent = mDirectInductorAdmittance * (terminal1 - midpointVoltage) +
402 mDirectSeriesHistory;
403 return;
404 }
405 **mIntfCurrent = mSubSeriesInductor->intfCurrent();
406}
