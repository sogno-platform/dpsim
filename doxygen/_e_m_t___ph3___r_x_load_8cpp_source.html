67void EMT::Ph3::RXLoad::setParameters(Matrix activePower, Matrix reactivePower,
68 Real volt, bool reactanceInSeries) {
69 if (!mUseSubcomponents && reactanceInSeries)
70 throw std::invalid_argument(
71 "Direct RXLoad mode currently supports parallel R-X loads only.");
72 **mActivePower = activePower;
73 **mReactivePower = reactivePower;
74 mReactanceInSeries = reactanceInSeries;
75
76 if (mReactanceInSeries) {
77 setVirtualNodeNumber(1);
78 }
79 // complex power
80 mPower = MatrixComp::Zero(3, 3);
81 mPower(0, 0) = {(**mActivePower)(0, 0), (**mReactivePower)(0, 0)};
82 mPower(1, 1) = {(**mActivePower)(1, 1), (**mReactivePower)(1, 1)};
83 mPower(2, 2) = {(**mActivePower)(2, 2), (**mReactivePower)(2, 2)};
84
85 **mNomVoltage = volt;
86
87 SPDLOG_LOGGER_INFO(mSLog,
88 "\nActive Power [W]: {}"
89 "\nReactive Power [VAr]: {}",
90 Logger::matrixToString(**mActivePower),
91 Logger::matrixToString(**mReactivePower));
92 SPDLOG_LOGGER_INFO(mSLog, "Nominal Voltage={} [V]", **mNomVoltage);
93
94 initPowerFromTerminal = false;
95}

97void EMT::Ph3::RXLoad::createSubComponents() {
98 if (mSubCompCreated)
99 return;
100 mSubCompCreated = true;
101 if (!mUseSubcomponents)
102 return;
103
104 // Intentionally empty: which of R/L/C exist depends on the load power sign, known only in
105 // initializeParentFromNodesAndTerminals(), where the sub-components are created. Safe: any
106 // series-mode virtual node is already declared in setParameters().

109void EMT::Ph3::RXLoad::initializeParentFromNodesAndTerminals(Real frequency) {
110 Real omega = 2. * PI * frequency;
111
112 if (initPowerFromTerminal) {
113 **mActivePower = Matrix::Zero(3, 3);
114 (**mActivePower)(0, 0) = mTerminals[0]->singleActivePower() / 3.;
115 (**mActivePower)(1, 1) = mTerminals[0]->singleActivePower() / 3.;
116 (**mActivePower)(2, 2) = mTerminals[0]->singleActivePower() / 3.;
117
118 **mReactivePower = Matrix::Zero(3, 3);
119 (**mReactivePower)(0, 0) = mTerminals[0]->singleReactivePower() / 3.;
120 (**mReactivePower)(1, 1) = mTerminals[0]->singleReactivePower() / 3.;
121 (**mReactivePower)(2, 2) = mTerminals[0]->singleReactivePower() / 3.;
122
123 // complex power
124 mPower = MatrixComp::Zero(3, 3);
125 mPower(0, 0) = {(**mActivePower)(0, 0), (**mReactivePower)(0, 0)};
126 mPower(1, 1) = {(**mActivePower)(1, 1), (**mReactivePower)(1, 1)};
127 mPower(2, 2) = {(**mActivePower)(2, 2), (**mReactivePower)(2, 2)};
128
129 **mNomVoltage = std::abs(mTerminals[0]->initialSingleVoltage());
130
131 SPDLOG_LOGGER_INFO(mSLog,
132 "\nActive Power [W]: {}"
133 "\nReactive Power [VAr]: {}",
134 Logger::matrixToString(**mActivePower),
135 Logger::matrixToString(**mReactivePower));
136 SPDLOG_LOGGER_INFO(mSLog, "Nominal Voltage={} [V]", **mNomVoltage);
137 }
138
139 // Compute derived impedance values and create+parametrize sub-components
140 // now that power and voltage are guaranteed to be known.
141 if ((**mActivePower)(0, 0) != 0)
142 mResistance =
143 std::pow(**mNomVoltage / sqrt(3), 2) * (**mActivePower).inverse();
144 else
145 mResistance = Matrix::Zero(3, 3);
146
147 if ((**mReactivePower)(0, 0) != 0)
148 mReactance =
149 std::pow(**mNomVoltage / sqrt(3), 2) * (**mReactivePower).inverse();
150 else
151 mReactance = Matrix::Zero(3, 3);
152
153 if (!mUseSubcomponents) {
154 const Real omega = 2. * PI * frequency;
155 if (mReactance(0, 0) > 0.)
156 mInductance = mReactance / omega;
157 else if (mReactance(0, 0) < 0.)
158 mCapacitance = -mReactance.inverse() / omega;
159 mDirectTotalConductance =
160 mResistance.norm() > 0. ? mResistance.inverse()(0, 0) : 0.;
161 mDirectHasInductor = mReactance(0, 0) > 0.;
162 mDirectHasCapacitor = mReactance(0, 0) < 0.;
163 MatrixComp voltage = MatrixComp::Zero(3, 1);
164 voltage(0) = RMS3PH_TO_PEAK1PH * mTerminals[0]->initialSingleVoltage();
165 voltage(1) = voltage(0) * SHIFT_TO_PHASE_B;
166 voltage(2) = voltage(0) * SHIFT_TO_PHASE_C;
167 MatrixComp loadG = MatrixComp::Zero(3, 3);
168 if (mResistance.norm() > 0.)
169 loadG = mResistance.inverse().cast<Complex>();
170 **mIntfVoltage = voltage.real();
171 (**mDirectState).setZero();
172 if (mDirectHasInductor) {
173 const MatrixComp yL =
174 (Complex(0., omega) * mInductance.cast<Complex>()).inverse();
175 (**mDirectState).block<3, 1>(0, 0) = voltage.real();
176 (**mDirectState).block<3, 1>(3, 0) = (yL * voltage).real();
177 }
178 if (mDirectHasCapacitor) {
179 const MatrixComp yC = Complex(0., omega) * mCapacitance.cast<Complex>();
180 (**mDirectState).block<3, 1>(6, 0) = voltage.real();
181 (**mDirectState).block<3, 1>(9, 0) = (yC * voltage).real();
182 }
183 **mIntfCurrent = loadG.real() * voltage.real();
184 if (mDirectHasInductor)
185 **mIntfCurrent += (**mDirectState).block<3, 1>(3, 0);
186 if (mDirectHasCapacitor)
187 **mIntfCurrent += (**mDirectState).block<3, 1>(9, 0);
188 return;
189 }
190
191 if ((**mActivePower)(0, 0) != 0) {
192 mSubResistor =
193 std::make_shared<EMT::Ph3::Resistor>(**mName + "_res", mLogLevel);
194 mSubResistor->setParameters(mResistance);
195 if (mReactanceInSeries) {
196 mSubResistor->connect({mTerminals[0]->node(), mVirtualNodes[0]});
197 } else {
198 mSubResistor->connect({SimNode::GND, mTerminals[0]->node()});
199 }
200 addMNASubComponent(mSubResistor, MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
201 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
202 }
203
204 if (mReactance(0, 0) > 0) {
205 mInductance = mReactance / omega;
206
207 mSubInductor =
208 std::make_shared<EMT::Ph3::Inductor>(**mName + "_ind", mLogLevel);
209 mSubInductor->setParameters(mInductance);
210 if (mReactanceInSeries) {
211 mSubInductor->connect({SimNode::GND, mVirtualNodes[0]});
212 } else {
213 mSubInductor->connect({SimNode::GND, mTerminals[0]->node()});
214 }
215 addMNASubComponent(mSubInductor, MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
216 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
217 } else if (mReactance(0, 0) < 0) {
218 mCapacitance = -1. / omega * mReactance.inverse();
219
220 mSubCapacitor =
221 std::make_shared<EMT::Ph3::Capacitor>(**mName + "_cap", mLogLevel);
222 mSubCapacitor->setParameters(mCapacitance);
223 if (mReactanceInSeries) {
224 mSubCapacitor->connect({SimNode::GND, mVirtualNodes[0]});
225 } else {
226 mSubCapacitor->connect({SimNode::GND, mTerminals[0]->node()});
227 }
228 addMNASubComponent(mSubCapacitor,
229 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
230 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
231 }
232
233 MatrixComp vInitABC = MatrixComp::Zero(3, 1);
234 vInitABC(0, 0) = RMS3PH_TO_PEAK1PH * mTerminals[0]->initialSingleVoltage();
235 vInitABC(1, 0) = vInitABC(0, 0) * SHIFT_TO_PHASE_B;
236 vInitABC(2, 0) = vInitABC(0, 0) * SHIFT_TO_PHASE_C;
237 **mIntfVoltage = vInitABC.real();
238
239 if (mReactanceInSeries) {
240 MatrixComp impedance = MatrixComp::Zero(3, 3);
241 impedance << Complex(mResistance(0, 0), mReactance(0, 0)),
242 Complex(mResistance(0, 1), mReactance(0, 1)),
243 Complex(mResistance(0, 2), mReactance(0, 2)),
244 Complex(mResistance(1, 0), mReactance(1, 0)),
245 Complex(mResistance(1, 1), mReactance(1, 1)),
246 Complex(mResistance(1, 2), mReactance(1, 2)),
247 Complex(mResistance(2, 0), mReactance(2, 0)),
248 Complex(mResistance(2, 1), mReactance(2, 1)),
249 Complex(mResistance(2, 2), mReactance(2, 2));
250 MatrixComp iInit = impedance.inverse() * vInitABC;
251 **mIntfCurrent = iInit.real();
252
253 // Initialization of virtual node
254 // Initial voltage of phase B,C is set after A
255 MatrixComp vInitTerm0 = MatrixComp::Zero(3, 1);
256 vInitTerm0(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
257 vInitTerm0(1, 0) = vInitTerm0(0, 0) * SHIFT_TO_PHASE_B;
258 vInitTerm0(2, 0) = vInitTerm0(0, 0) * SHIFT_TO_PHASE_C;
259 mVirtualNodes[0]->setInitialVoltage(PEAK1PH_TO_RMS3PH *
260 (vInitTerm0 - mResistance * iInit));
261 }
262
263 if ((**mActivePower)(0, 0) != 0) {
264 mSubResistor->initializeFromNodesAndTerminals(frequency);
265 if (!mReactanceInSeries) {
266 **mIntfCurrent += mSubResistor->intfCurrent();
267 }
268 }
269
270 if (mReactance(0, 0) > 0) {
271 mSubInductor->initializeFromNodesAndTerminals(frequency);
272 if (!mReactanceInSeries) {
273 **mIntfCurrent += mSubInductor->intfCurrent();
274 }
275 } else if (mReactance(0, 0) < 0) {
276 mSubCapacitor->initializeFromNodesAndTerminals(frequency);
277 if (!mReactanceInSeries) {
278 **mIntfCurrent += mSubCapacitor->intfCurrent();
279 }
280 }
281
282 SPDLOG_LOGGER_INFO(
283 mSLog,
284 "\n--- Initialization from powerflow ---"
285 "\nVoltage across: {:s}"
286 "\nCurrent: {:s}"
287 "\nTerminal 0 voltage: {:s}"
288 "\nActive Power: {:s}"
289 "\nReactive Power: {:s}"
290 "\nResistance: {:s}"
291 "\nReactance: {:s}"
292 "\n--- Initialization from powerflow finished ---",
293 Logger::matrixToString(**mIntfVoltage),
294 Logger::matrixToString(**mIntfCurrent),
295 Logger::phasorToString(RMS3PH_TO_PEAK1PH * initialSingleVoltage(0)),
296 Logger::matrixToString(**mActivePower),
297 Logger::matrixToString(**mReactivePower),
298 Logger::matrixToString(mResistance), Logger::matrixToString(mReactance));
299 mSLog->flush();
300}

323void EMT::Ph3::RXLoad::mnaParentPreStep(Real time, Int timeStepCount) {
324 if (!mUseSubcomponents) {
325 const Matrix &state = **mDirectState;
326 for (Int phase = 0; phase < 3; ++phase) {
327 if (mDirectHasInductor)
328 mDirectHistoryLoad(phase, 0) =
329 mDirectInductorAdmittance * state(phase, 0) + state(phase + 3, 0);
330 if (mDirectHasCapacitor)
331 mDirectHistoryShunt(phase, 0) =
332 -state(phase + 9, 0) -
333 mDirectCapacitorAdmittance * state(phase + 6, 0);
334 }
335 }
336 mnaCompApplyRightSideVectorStamp(**mRightVector);
337}
338
339void EMT::Ph3::RXLoad::mnaParentPostStep(Real time, Int timeStepCount,
340 Attribute<Matrix>::Ptr &leftVector) {
341 if (!mUseSubcomponents) {
342 mnaCompUpdateVoltage(**leftVector);
343 const Matrix &voltage = **mIntfVoltage;
344 Matrix &state = **mDirectState;
345 Matrix &current = **mIntfCurrent;
346 for (Int phase = 0; phase < 3; ++phase) {
347 Real inductorCurrent = 0.;
348 Real capacitorCurrent = 0.;
349 if (mDirectHasInductor) {
350 inductorCurrent = mDirectInductorAdmittance * voltage(phase, 0) +
351 mDirectHistoryLoad(phase, 0);
352 state(phase, 0) = voltage(phase, 0);
353 state(phase + 3, 0) = inductorCurrent;
354 }
355 if (mDirectHasCapacitor) {
356 capacitorCurrent = mDirectCapacitorAdmittance * voltage(phase, 0) +
357 mDirectHistoryShunt(phase, 0);
358 state(phase + 6, 0) = voltage(phase, 0);
359 state(phase + 9, 0) = capacitorCurrent;
360 }
361 current(phase, 0) = mDirectTotalConductance * voltage(phase, 0) +
362 inductorCurrent + capacitorCurrent;
363 }
364 return;
365 }
366 mnaCompUpdateVoltage(**leftVector);
367 mnaCompUpdateCurrent(**leftVector);
368}

380void EMT::Ph3::RXLoad::mnaCompUpdateCurrent(const Matrix &leftVector) {
381 if (!mUseSubcomponents) {
382 const Matrix &voltage = **mIntfVoltage;
383 Matrix &current = **mIntfCurrent;
384 for (Int phase = 0; phase < 3; ++phase) {
385 Real inductorCurrent = 0.;
386 Real capacitorCurrent = 0.;
387 if (mDirectHasInductor)
388 inductorCurrent = mDirectInductorAdmittance * voltage(phase, 0) +
389 mDirectHistoryLoad(phase, 0);
390 if (mDirectHasCapacitor)
391 capacitorCurrent = mDirectCapacitorAdmittance * voltage(phase, 0) +
392 mDirectHistoryShunt(phase, 0);
393 current(phase, 0) = mDirectTotalConductance * voltage(phase, 0) +
394 inductorCurrent + capacitorCurrent;
395 }
396 return;
397 }
398 if (mReactanceInSeries) {
399 **mIntfCurrent = mSubInductor->intfCurrent();
400 } else {
401 (**mIntfCurrent).setZero();
402 for (auto &subc : mSubComponents) {
403 **mIntfCurrent += subc->intfCurrent();
404 }
405 }

408void EMT::Ph3::RXLoad::mnaParentInitialize(Real, Real timeStep,
409 Attribute<Matrix>::Ptr) {
410 if (mUseSubcomponents)
411 return;
412 mDirectInductorAdmittance =
413 mDirectHasInductor ? timeStep / (2. * mInductance(0, 0)) : 0.;
414 mDirectCapacitorAdmittance =
415 mDirectHasCapacitor ? 2. * mCapacitance(0, 0) / timeStep : 0.;
416 mDirectNodalAdmittance =
417 (mDirectTotalConductance + mDirectInductorAdmittance +
418 mDirectCapacitorAdmittance) *
419 Matrix::Identity(3, 3);
420}
