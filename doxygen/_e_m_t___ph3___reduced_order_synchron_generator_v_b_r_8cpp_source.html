34void EMT::Ph3::ReducedOrderSynchronGeneratorVBR::initializeResistanceMatrix() {
35 // dq0 resistance matrix
36 mResistanceMatrixDq0 = MatrixFixedSize<3, 3>::Zero(3, 3);
37 mResistanceMatrixDq0 << 0.0, mA, 0.0, mB, 0.0, 0.0, 0.0, 0.0, mL0;
38
39 // This impedance is constant in the rotor frame. Compute its inverse once;
40 // each step only rotates the conductance into the abc frame.
41 mConductanceMatrixDq0 = mResistanceMatrixDq0.inverse() / mBase_Z;
42 mConductanceMatrix = MatrixFixedSize<3, 3>::Zero(3, 3);
43}
44
45void EMT::Ph3::ReducedOrderSynchronGeneratorVBR::calculateResistanceMatrix() {
46 mConductanceMatrix.noalias() = mDq0ToAbc * mConductanceMatrixDq0 * mAbcToDq0;
47}

187void EMT::Ph3::ReducedOrderSynchronGeneratorVBR::mnaCompApplySystemMatrixStamp(
188 SparseMatrixRow &systemMatrix) {
189
190 if (mDeviceMatrixManaged)
191 return;
192
193 if (mModelAsNortonSource) {
194 MNAStampUtils::stamp3x3ConductanceMatrixNodeToGround(
195 mConductanceMatrix, systemMatrix, matrixNodeIndex(0), mSLog);
196 } else {
197 // Stamp voltage source
198 Math::addToMatrixElement(
199 systemMatrix, mVirtualNodes[0]->matrixNodeIndex(PhaseType::A),
200 mVirtualNodes[1]->matrixNodeIndex(PhaseType::A), -1);
201 Math::addToMatrixElement(
202 systemMatrix, mVirtualNodes[1]->matrixNodeIndex(PhaseType::A),
203 mVirtualNodes[0]->matrixNodeIndex(PhaseType::A), 1);
204 Math::addToMatrixElement(
205 systemMatrix, mVirtualNodes[0]->matrixNodeIndex(PhaseType::B),
206 mVirtualNodes[1]->matrixNodeIndex(PhaseType::B), -1);
207 Math::addToMatrixElement(
208 systemMatrix, mVirtualNodes[1]->matrixNodeIndex(PhaseType::B),
209 mVirtualNodes[0]->matrixNodeIndex(PhaseType::B), 1);
210 Math::addToMatrixElement(
211 systemMatrix, mVirtualNodes[0]->matrixNodeIndex(PhaseType::C),
212 mVirtualNodes[1]->matrixNodeIndex(PhaseType::C), -1);
213 Math::addToMatrixElement(
214 systemMatrix, mVirtualNodes[1]->matrixNodeIndex(PhaseType::C),
215 mVirtualNodes[0]->matrixNodeIndex(PhaseType::C), 1);
216
217 // Stamp conductance matrix
218 MNAStampUtils::stamp3x3ConductanceMatrixBetween2Nodes(
219 mConductanceMatrix, systemMatrix,
220 mVirtualNodes[0]->matrixNodeIndex(PhaseType::A), matrixNodeIndex(0),
221 mSLog);
222 }
223}

225void EMT::Ph3::ReducedOrderSynchronGeneratorVBR::
226 mnaCompApplyRightSideVectorStamp(Matrix &rightVector) {
227 if (mModelAsNortonSource) {
228 // compute equivalent northon circuit in abc reference frame
229 mIvbr = mConductanceMatrix * mEvbr;
230
231 Math::setVectorElement(rightVector, matrixNodeIndex(0, 0), mIvbr(0, 0));
232 Math::setVectorElement(rightVector, matrixNodeIndex(0, 1), mIvbr(1, 0));
233 Math::setVectorElement(rightVector, matrixNodeIndex(0, 2), mIvbr(2, 0));
234 } else {
235 Math::setVectorElement(rightVector,
236 mVirtualNodes[1]->matrixNodeIndex(PhaseType::A),
237 mEvbr(0, 0));
238 Math::setVectorElement(rightVector,
239 mVirtualNodes[1]->matrixNodeIndex(PhaseType::B),
240 mEvbr(1, 0));
241 Math::setVectorElement(rightVector,
242 mVirtualNodes[1]->matrixNodeIndex(PhaseType::C),
243 mEvbr(2, 0));
244 }
245}

247void EMT::Ph3::ReducedOrderSynchronGeneratorVBR::
248 mnaAddCompactRightSideVectorStamp(Matrix &rightVector) {
249 if (mDeviceRhsManaged)
250 return;
251 const Matrix &stamp = **mRightVector;
252 const PhaseType phases[] = {PhaseType::A, PhaseType::B, PhaseType::C};
253 for (UInt phase = 0; phase < 3; ++phase) {
254 const UInt index = mModelAsNortonSource
255 ? matrixNodeIndex(0, phase)
256 : mVirtualNodes[1]->matrixNodeIndex(phases[phase]);
257 rightVector(index, 0) += stamp(index, 0);
258 }

261void EMT::Ph3::ReducedOrderSynchronGeneratorVBR::mnaCompPostStep(
262 const Matrix &leftVector) {
263 // update armature voltage
264 (**mIntfVoltage)(0, 0) =
265 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 0));
266 (**mIntfVoltage)(1, 0) =
267 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 1));
268 (**mIntfVoltage)(2, 0) =
269 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 2));
270
271 // convert terminal voltage into dq reference frame
272 **mVdq0 = mAbcToDq0 * **mIntfVoltage / mBase_V;
273
274 // update armature current
275 if (mModelAsNortonSource) {
276 **mIntfCurrent = mIvbr - mConductanceMatrix * **mIntfVoltage;
277 } else {
278 (**mIntfCurrent)(0, 0) = Math::realFromVectorElement(
279 leftVector, mVirtualNodes[1]->matrixNodeIndex(PhaseType::A));
280 (**mIntfCurrent)(1, 0) = Math::realFromVectorElement(
281 leftVector, mVirtualNodes[1]->matrixNodeIndex(PhaseType::B));
282 (**mIntfCurrent)(2, 0) = Math::realFromVectorElement(
283 leftVector, mVirtualNodes[1]->matrixNodeIndex(PhaseType::C));
284 }
285
286 // convert armature current into dq reference frame
287 **mIdq0 = mAbcToDq0 * **mIntfCurrent / mBase_I;
288}

290MatrixFixedSize<3, 3>
291EMT::Ph3::ReducedOrderSynchronGeneratorVBR::get_parkTransformMatrix() const {
292 MatrixFixedSize<3, 3> abcToDq0;
293
294 abcToDq0 << 2. / 3. * cos(**mThetaMech),
295 2. / 3. * cos(**mThetaMech - 2. * PI / 3.),
296 2. / 3. * cos(**mThetaMech + 2. * PI / 3.), -2. / 3. * sin(**mThetaMech),
297 -2. / 3. * sin(**mThetaMech - 2. * PI / 3.),
298 -2. / 3. * sin(**mThetaMech + 2. * PI / 3.), 1. / 3., 1. / 3., 1. / 3.;
299
300 return abcToDq0;
301}

303MatrixFixedSize<3, 3>
304EMT::Ph3::ReducedOrderSynchronGeneratorVBR::get_inverseParkTransformMatrix()
305 const {
306 MatrixFixedSize<3, 3> dq0ToAbc;
307
308 dq0ToAbc << cos(**mThetaMech), -sin(**mThetaMech), 1.,
309 cos(**mThetaMech - 2. * PI / 3.), -sin(**mThetaMech - 2. * PI / 3.), 1.,
310 cos(**mThetaMech + 2. * PI / 3.), -sin(**mThetaMech + 2. * PI / 3.), 1.;
311
312 return dq0ToAbc;
313}

315void EMT::Ph3::ReducedOrderSynchronGeneratorVBR::updateParkTransformMatrices() {
316 const Real theta = **mThetaMech;
317 const Real thetaB = theta - 2. * PI / 3.;
318 const Real thetaC = theta + 2. * PI / 3.;
319 const Real cosA = std::cos(theta);
320 const Real sinA = std::sin(theta);
321 const Real cosB = std::cos(thetaB);
322 const Real sinB = std::sin(thetaB);
323 const Real cosC = std::cos(thetaC);
324 const Real sinC = std::sin(thetaC);
325
326 mAbcToDq0 << 2. / 3. * cosA, 2. / 3. * cosB, 2. / 3. * cosC, -2. / 3. * sinA,
327 -2. / 3. * sinB, -2. / 3. * sinC, 1. / 3., 1. / 3., 1. / 3.;
328 mDq0ToAbc << cosA, -sinA, 1., cosB, -sinB, 1., cosC, -sinC, 1.;
329}
