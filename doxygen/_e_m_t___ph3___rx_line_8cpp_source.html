45void EMT::Ph3::RxLine::createSubComponents() {
46 if (mSubCompCreated)
47 return;
48 mSubCompCreated = true;
49
50 if (!mUseSubcomponents)
51 return;
52
53 // Default model with virtual node in between
54 mSubResistor =
55 std::make_shared<EMT::Ph3::Resistor>(**mName + "_res", mLogLevel);
56 mSubResistor->setParameters(**mSeriesRes);
57 mSubResistor->connect({mTerminals[0]->node(), mVirtualNodes[0]});
58 addMNASubComponent(mSubResistor, MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
59 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
60
61 mSubInductor =
62 std::make_shared<EMT::Ph3::Inductor>(**mName + "_ind", mLogLevel);
63 mSubInductor->setParameters(**mSeriesInd);
64 mSubInductor->connect({mVirtualNodes[0], mTerminals[1]->node()});
65 addMNASubComponent(mSubInductor, MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
66 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, true);
67
68 mInitialResistor =
69 std::make_shared<EMT::Ph3::Resistor>(**mName + "_snubber_res", mLogLevel);
70 Matrix defaultSnubRes = Matrix::Zero(3, 3);
71 defaultSnubRes << 1e6, 0, 0, 0, 1e6, 0, 0, 0, 1e6;
72 mInitialResistor->setParameters(defaultSnubRes);
73 mInitialResistor->connect({SimNode::GND, mTerminals[1]->node()});
74 addMNASubComponent(mInitialResistor,
75 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT,
76 MNA_SUBCOMP_TASK_ORDER::TASK_BEFORE_PARENT, false);
77}

79void EMT::Ph3::RxLine::initializeParentFromNodesAndTerminals(Real frequency) {
80 // Match PiLine's RMS line-to-neutral power-flow phasors to the peak
81 // phase-to-ground values used by the EMT subcomponents. The virtual node is
82 // not initialized yet here, so deriving the branch voltage from it would
83 // start the inductor with an arbitrary (usually zero) voltage.
84 Real omega = 2. * PI * frequency;
85 MatrixComp impedance = MatrixComp::Zero(3, 3);
86 impedance << Complex((**mSeriesRes)(0, 0), omega * (**mSeriesInd)(0, 0)),
87 Complex((**mSeriesRes)(0, 1), omega * (**mSeriesInd)(0, 1)),
88 Complex((**mSeriesRes)(0, 2), omega * (**mSeriesInd)(0, 2)),
89 Complex((**mSeriesRes)(1, 0), omega * (**mSeriesInd)(1, 0)),
90 Complex((**mSeriesRes)(1, 1), omega * (**mSeriesInd)(1, 1)),
91 Complex((**mSeriesRes)(1, 2), omega * (**mSeriesInd)(1, 2)),
92 Complex((**mSeriesRes)(2, 0), omega * (**mSeriesInd)(2, 0)),
93 Complex((**mSeriesRes)(2, 1), omega * (**mSeriesInd)(2, 1)),
94 Complex((**mSeriesRes)(2, 2), omega * (**mSeriesInd)(2, 2));
95
96 MatrixComp vInitABC = MatrixComp::Zero(3, 1);
97 vInitABC(0, 0) =
98 RMS3PH_TO_PEAK1PH * (initialSingleVoltage(1) - initialSingleVoltage(0));
99 vInitABC(1, 0) = vInitABC(0, 0) * SHIFT_TO_PHASE_B;
100 vInitABC(2, 0) = vInitABC(0, 0) * SHIFT_TO_PHASE_C;
101
102 MatrixComp iInit = impedance.inverse() * vInitABC;
103 **mIntfCurrent = iInit.real();
104 **mIntfVoltage = vInitABC.real();
105 // Initialization of virtual node
106 // Initial voltage of phase B,C is set after A
107 MatrixComp vInitTerm0 = MatrixComp::Zero(3, 1);
108 vInitTerm0(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
109 vInitTerm0(1, 0) = vInitTerm0(0, 0) * SHIFT_TO_PHASE_B;
110 vInitTerm0(2, 0) = vInitTerm0(0, 0) * SHIFT_TO_PHASE_C;
111
112 mVirtualNodes[0]->setInitialVoltage(PEAK1PH_TO_RMS3PH *
113 (vInitTerm0 + **mSeriesRes * iInit));
114 if (!mUseSubcomponents) {
115 (**mDirectState).setZero();
116 (**mDirectState).block<3, 1>(0, 0) = iInit.real();
117 (**mDirectState).block<3, 1>(3, 0) =
118 (vInitABC - (**mSeriesRes).cast<Complex>() * iInit).real();
119 }
120
121 SPDLOG_LOGGER_INFO(mSLog,
122 "\n--- Initialization from powerflow ---"
123 "\nVoltage across: {:s}"
124 "\nCurrent: {:s}"
125 "\nTerminal 0 voltage: {:s}"
126 "\nTerminal 1 voltage: {:s}"
127 "\n--- Initialization from powerflow finished ---",
128 Logger::matrixToString(**mIntfVoltage),
129 Logger::matrixToString(**mIntfCurrent),
130 Logger::phasorToString(initialSingleVoltage(0)),
131 Logger::phasorToString(initialSingleVoltage(1)));
132}

140void EMT::Ph3::RxLine::mnaParentApplySystemMatrixStamp(
141 SparseMatrixRow &systemMatrix) {
142 if (mUseSubcomponents)
143 return;
144 updateMatrixNodeIndices();
145 cacheDirectMatrixIndices();
146 const Matrix snubberConductance = Matrix::Identity(3, 3) * 1e-6;
147 MNAStampUtils::stampConductanceMatrix(
148 (**mSeriesRes).inverse(), systemMatrix, mDirectTerminalIndices[0][0],
149 mDirectMidpointIndices[0], mDirectTerminalConnected[0], true, mSLog);
150 MNAStampUtils::stampConductanceMatrix(
151 mDirectInductorAdmittance, systemMatrix, mDirectMidpointIndices[0],
152 mDirectTerminalIndices[1][0], true, mDirectTerminalConnected[1], mSLog);
153 MNAStampUtils::stampConductanceMatrix(
154 snubberConductance, systemMatrix, mDirectTerminalIndices[1][0], 0,
155 mDirectTerminalConnected[1], false, mSLog);
156}

158void EMT::Ph3::RxLine::cacheDirectMatrixIndices() {
159 for (UInt terminal = 0; terminal < 2; ++terminal) {
160 mDirectTerminalConnected[terminal] = terminalNotGrounded(terminal);
161 for (UInt phase = 0; phase < 3; ++phase)
162 mDirectTerminalIndices[terminal][phase] =
163 matrixNodeIndex(terminal, phase);
164 }
165 for (UInt phase = 0; phase < 3; ++phase)
166 mDirectMidpointIndices[phase] =
167 mVirtualNodes[0]->matrixNodeIndex(static_cast<PhaseType>(phase));
168}

170void EMT::Ph3::RxLine::mnaParentApplyRightSideVectorStamp(Matrix &rightVector) {
171 if (mUseSubcomponents)
172 return;
173 for (UInt phase = 0; phase < 3; ++phase) {
174 Math::setVectorElement(rightVector, mDirectMidpointIndices[phase],
175 mDirectHistory(phase));
176 if (mDirectTerminalConnected[1])
177 Math::setVectorElement(rightVector, mDirectTerminalIndices[1][phase],
178 -mDirectHistory(phase));
179 }
180}

187void EMT::Ph3::RxLine::mnaAddCompactRightSideVectorStamp(Matrix &rightVector) {
188 if (mUseSubcomponents) {
189 CompositePowerComp<Real>::mnaAddCompactRightSideVectorStamp(rightVector);
190 return;
191 }
192 for (UInt phase = 0; phase < 3; ++phase) {
193 Math::addToVectorElement(
194 rightVector, mDirectMidpointIndices[phase],
195 Math::realFromVectorElement(**mRightVector,
196 mDirectMidpointIndices[phase]));
197 if (mDirectTerminalConnected[1]) {
198 const UInt index = mDirectTerminalIndices[1][phase];
199 Math::addToVectorElement(
200 rightVector, index,
201 Math::realFromVectorElement(**mRightVector, index));
202 }
203 }

236void EMT::Ph3::RxLine::mnaParentPostStep(Real time, Int timeStepCount,
237 Attribute<Matrix>::Ptr &leftVector) {
238 if (!mUseSubcomponents) {
239 Eigen::Matrix<Real, 3, 1> v0 = Eigen::Matrix<Real, 3, 1>::Zero();
240 Eigen::Matrix<Real, 3, 1> v1 = Eigen::Matrix<Real, 3, 1>::Zero();
241 Eigen::Matrix<Real, 3, 1> vmid;
242 if (mDirectTerminalConnected[0])
243 for (UInt phase = 0; phase < 3; ++phase)
244 v0(phase) = Math::realFromVectorElement(
245 **leftVector, mDirectTerminalIndices[0][phase]);
246 if (mDirectTerminalConnected[1])
247 for (UInt phase = 0; phase < 3; ++phase)
248 v1(phase) = Math::realFromVectorElement(
249 **leftVector, mDirectTerminalIndices[1][phase]);
250 for (UInt phase = 0; phase < 3; ++phase)
251 vmid(phase) = Math::realFromVectorElement(**leftVector,
252 mDirectMidpointIndices[phase]);
253 **mIntfVoltage = v1 - v0;
254 **mIntfCurrent = mDirectInductorAdmittance * (v1 - vmid) + mDirectHistory;
255 (**mDirectState).block<3, 1>(0, 0) = **mIntfCurrent;
256 (**mDirectState).block<3, 1>(3, 0) = v1 - vmid;
257 return;
258 }
259 mnaCompUpdateVoltage(**leftVector);
260 mnaCompUpdateCurrent(**leftVector);
261}

263void EMT::Ph3::RxLine::mnaCompUpdateVoltage(const Matrix &leftVector) {
264 // v1 - v0
265 (**mIntfVoltage).setZero();
266 if (terminalNotGrounded(1)) {
267 (**mIntfVoltage)(0, 0) =
268 Math::realFromVectorElement(leftVector, matrixNodeIndex(1, 0));
269 (**mIntfVoltage)(1, 0) =
270 Math::realFromVectorElement(leftVector, matrixNodeIndex(1, 1));
271 (**mIntfVoltage)(2, 0) =
272 Math::realFromVectorElement(leftVector, matrixNodeIndex(1, 2));
273 }
274 if (terminalNotGrounded(0)) {
275 (**mIntfVoltage)(0, 0) =
276 (**mIntfVoltage)(0, 0) -
277 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 0));
278 (**mIntfVoltage)(1, 0) =
279 (**mIntfVoltage)(1, 0) -
280 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 1));
281 (**mIntfVoltage)(2, 0) =
282 (**mIntfVoltage)(2, 0) -
283 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 2));
284 }
285}

287void EMT::Ph3::RxLine::mnaCompUpdateCurrent(const Matrix &leftVector) {
288 if (!mUseSubcomponents) {
289 Eigen::Matrix<Real, 3, 1> v1 = Eigen::Matrix<Real, 3, 1>::Zero();
290 Eigen::Matrix<Real, 3, 1> vmid;
291 if (mDirectTerminalConnected[1])
292 for (UInt phase = 0; phase < 3; ++phase)
293 v1(phase) = Math::realFromVectorElement(
294 leftVector, mDirectTerminalIndices[1][phase]);
295 for (UInt phase = 0; phase < 3; ++phase)
296 vmid(phase) = Math::realFromVectorElement(leftVector,
297 mDirectMidpointIndices[phase]);
298 **mIntfCurrent = mDirectInductorAdmittance * (v1 - vmid) + mDirectHistory;
299 return;
300 }
301 **mIntfCurrent = mSubInductor->intfCurrent();
302}
