87void EMT::Ph3::SSN_GFL_Split::setParameters(Real lf, Real cf, Real rf, Real rc,
88 Real omegaN, Real kpPLL, Real kiPLL,
89 Real omegaCutoff, Real pRef,
90 Real qRef, Real kpPowerCtrl,
91 Real kiPowerCtrl, Real kpCurrCtrl,
92 Real kiCurrCtrl) {
93
94 if (!Math::isFinite(lf) || !Math::isFinite(cf) || !Math::isFinite(rf) ||
95 !Math::isFinite(rc) || !Math::isFinite(omegaN) ||
96 !Math::isFinite(kpPLL) || !Math::isFinite(kiPLL) ||
97 !Math::isFinite(omegaCutoff) || !Math::isFinite(pRef) ||
98 !Math::isFinite(qRef) || !Math::isFinite(kpPowerCtrl) ||
99 !Math::isFinite(kiPowerCtrl) || !Math::isFinite(kpCurrCtrl) ||
100 !Math::isFinite(kiCurrCtrl))
101 throw std::invalid_argument("SSN_GFL_Split parameters must be finite.");
102
103 if (lf <= 0.0)
104 throw std::invalid_argument("Filter inductance lf must be positive.");
105
106 if (cf <= 0.0)
107 throw std::invalid_argument("Filter capacitance cf must be positive.");
108
109 if (rf < 0.0)
110 throw std::invalid_argument("Filter resistance rf must be non-negative.");
111
112 if (rc <= 0.0)
113 throw std::invalid_argument("Coupling resistance rc must be positive.");
114
115 if (omegaN <= 0.0)
116 throw std::invalid_argument(
117 "Nominal angular frequency omegaN must be positive.");
118
119 if (omegaCutoff < 0.0)
120 throw std::invalid_argument(
121 "Power-filter cutoff frequency omegaCutoff must be non-negative.");
122
123 if (kiPLL == 0.0)
124 throw std::invalid_argument("PLL integral gain kiPLL must be non-zero.");
125
126 if (kiPowerCtrl == 0.0)
127 throw std::invalid_argument(
128 "Power-control integral gain kiPowerCtrl must be non-zero.");
129
130 if (kiCurrCtrl == 0.0)
131 throw std::invalid_argument(
132 "Current-control integral gain kiCurrCtrl must be non-zero.");
133
134 mLf = lf;
135 mCf = cf;
136 mRf = rf;
137 mRc = rc;
138
139 mOmegaN = omegaN;
140 mKpPLL = kpPLL;
141 mKiPLL = kiPLL;
142
143 mOmegaCutoff = omegaCutoff;
144 mPRef = pRef;
145 mQRef = qRef;
146 mKpPowerCtrl = kpPowerCtrl;
147 mKiPowerCtrl = kiPowerCtrl;
148 mKpCurrCtrl = kpCurrCtrl;
149 mKiCurrCtrl = kiCurrCtrl;
150
151 const Matrix identity3 = Matrix::Identity(3, 3);
152
153 // Constant electrical plant.
154 Matrix aNetwork = Matrix::Zero(mNetworkStateSize, mNetworkStateSize);
155 Matrix bTerminal = Matrix::Zero(mNetworkStateSize, mTerminalInputSize);
156 Matrix cNetwork = Matrix::Zero(mOutputSize, mNetworkStateSize);
157 Matrix dTerminal = Matrix::Zero(mOutputSize, mTerminalInputSize);
158
159 aNetwork.block(VcA, VcA, 3, 3) = -1.0 / (mCf * mRc) * identity3;
160 aNetwork.block(VcA, IfA, 3, 3) = 1.0 / mCf * identity3;
161
162 aNetwork.block(IfA, VcA, 3, 3) = -1.0 / mLf * identity3;
163 aNetwork.block(IfA, IfA, 3, 3) = -mRf / mLf * identity3;
164
165 bTerminal.block(VcA, 0, 3, 3) = 1.0 / (mCf * mRc) * identity3;
166
167 Matrix bControllerOutput = Matrix::Zero(mNetworkStateSize, 3);
168 bControllerOutput.block(IfA, 0, 3, 3) = 1.0 / mLf * identity3;
169
170 cNetwork.block(0, VcA, 3, 3) = -1.0 / mRc * identity3;
171 dTerminal = 1.0 / mRc * identity3;
172
173 Matrix measurementState =
174 Matrix::Zero(mControllerInputSize, mNetworkStateSize);
175 measurementState.block(0, VcA, 3, 3) = identity3;
176 measurementState.block(3, VcA, 3, 3) = (1.0 / mRc) * identity3;
177
178 Matrix measurementTerminal =
179 Matrix::Zero(mControllerInputSize, mTerminalInputSize);
180 measurementTerminal.block(3, 0, 3, 3) = (-1.0 / mRc) * identity3;
181
182 // The fixed network stamp is not changed by controller re-linearization.
183 setSplitParameters(aNetwork, bTerminal, cNetwork, dTerminal,
184 bControllerOutput, measurementState, measurementTerminal);
185}

187Matrix EMT::Ph3::SSN_GFL_Split::getParkTransformMatrix(Real theta) const {
188 Matrix transform(2, 3);
189 const Real scale = std::sqrt(2.0 / 3.0);
190
191 transform.row(0) << scale * std::cos(theta),
192 scale * std::cos(theta - 2.0 * PI / 3.0),
193 scale * std::cos(theta + 2.0 * PI / 3.0);
194
195 transform.row(1) << -scale * std::sin(theta),
196 -scale * std::sin(theta - 2.0 * PI / 3.0),
197 -scale * std::sin(theta + 2.0 * PI / 3.0);
198
199 return transform;
200}
201
202Matrix
203EMT::Ph3::SSN_GFL_Split::getInverseParkTransformMatrix(Real theta) const {
204 Matrix transform(3, 2);
205 const Real scale = std::sqrt(2.0 / 3.0);
206
207 transform << scale * std::cos(theta), -scale * std::sin(theta),
208 scale * std::cos(theta - 2.0 * PI / 3.0),
209 -scale * std::sin(theta - 2.0 * PI / 3.0),
210 scale * std::cos(theta + 2.0 * PI / 3.0),
211 -scale * std::sin(theta + 2.0 * PI / 3.0);
212
213 return transform;
214}

216void EMT::Ph3::SSN_GFL_Split::evaluateControllerStateDerivative(
217 const Matrix &xController, const Matrix &measurement,
218 Matrix &stateDerivative) const {
219
220 if (xController.rows() != mControllerStateSize || xController.cols() != 1)
221 throw std::invalid_argument(
222 "SSN_GFL_Split controller state has an invalid dimension.");
223
224 if (measurement.rows() != mControllerInputSize || measurement.cols() != 1)
225 throw std::invalid_argument(
226 "SSN_GFL_Split controller measurement has an invalid dimension.");
227
228 stateDerivative.setZero(mControllerStateSize, 1);
229
230 const Real thetaPLL = xController(ThetaPLL, 0);
231 const Real phiPLL = xController(PhiPLL, 0);
232 const Real pFiltered = xController(PFiltered, 0);
233 const Real qFiltered = xController(QFiltered, 0);
234 const Real phiD = xController(PhiD, 0);
235 const Real phiQ = xController(PhiQ, 0);
236
237 const Matrix vcAbc = measurement.block(0, 0, 3, 1);
238 const Matrix iGridAbc = measurement.block(3, 0, 3, 1);
239
240 const Matrix parkTransform = getParkTransformMatrix(thetaPLL);
241
242 const Matrix vcDq = parkTransform * vcAbc;
243 const Matrix iGridDq = parkTransform * iGridAbc;
244
245 const Real vcD = vcDq(0, 0);
246 const Real vcQ = vcDq(1, 0);
247 const Real iGridD = iGridDq(0, 0);
248 const Real iGridQ = iGridDq(1, 0);
249
250 const Real pInstantaneous = vcD * iGridD + vcQ * iGridQ;
251 const Real qInstantaneous = -vcD * iGridQ + vcQ * iGridD;
252
253 // 1. PLL
254 stateDerivative(ThetaPLL, 0) = mOmegaN + mKpPLL * vcQ + mKiPLL * phiPLL;
255 stateDerivative(PhiPLL, 0) = vcQ;
256
257 // 2. Active-/reactive-power filters
258 stateDerivative(PFiltered, 0) = mOmegaCutoff * (pInstantaneous - pFiltered);
259 stateDerivative(QFiltered, 0) = mOmegaCutoff * (qInstantaneous - qFiltered);
260
261 // 3. Outer power loop
262 stateDerivative(PhiD, 0) = mPRef - pFiltered;
263 stateDerivative(PhiQ, 0) = qFiltered - mQRef;
264
265 const Real currentReferenceD =
266 mKpPowerCtrl * (mPRef - pFiltered) + mKiPowerCtrl * phiD;
267 const Real currentReferenceQ =
268 mKpPowerCtrl * (qFiltered - mQRef) + mKiPowerCtrl * phiQ;
269
270 // 4. Inner current-loop integrators
271 stateDerivative(GammaD, 0) = currentReferenceD - iGridD;
272 stateDerivative(GammaQ, 0) = currentReferenceQ - iGridQ;
273}

275void EMT::Ph3::SSN_GFL_Split::evaluateControllerOutput(
276 const Matrix &xController, const Matrix &measurement,
277 Matrix &output) const {
278
279 if (xController.rows() != mControllerStateSize || xController.cols() != 1)
280 throw std::invalid_argument(
281 "SSN_GFL_Split controller state has an invalid dimension.");
282
283 if (measurement.rows() != mControllerInputSize || measurement.cols() != 1)
284 throw std::invalid_argument(
285 "SSN_GFL_Split controller measurement has an invalid dimension.");
286
287 const Real thetaPLL = xController(ThetaPLL, 0);
288 const Real pFiltered = xController(PFiltered, 0);
289 const Real qFiltered = xController(QFiltered, 0);
290 const Real phiD = xController(PhiD, 0);
291 const Real phiQ = xController(PhiQ, 0);
292 const Real gammaD = xController(GammaD, 0);
293 const Real gammaQ = xController(GammaQ, 0);
294
295 const Matrix iGridAbc = measurement.block(3, 0, 3, 1);
296
297 const Matrix parkTransform = getParkTransformMatrix(thetaPLL);
298 const Matrix inverseParkTransform = getInverseParkTransformMatrix(thetaPLL);
299
300 const Matrix iGridDq = parkTransform * iGridAbc;
301 const Real iGridD = iGridDq(0, 0);
302 const Real iGridQ = iGridDq(1, 0);
303
304 const Real currentReferenceD =
305 mKpPowerCtrl * (mPRef - pFiltered) + mKiPowerCtrl * phiD;
306 const Real currentReferenceQ =
307 mKpPowerCtrl * (qFiltered - mQRef) + mKiPowerCtrl * phiQ;
308
309 const Real currentErrorD = currentReferenceD - iGridD;
310 const Real currentErrorQ = currentReferenceQ - iGridQ;
311
312 const Real converterVoltageReferenceD =
313 mKpCurrCtrl * currentErrorD + mKiCurrCtrl * gammaD;
314 const Real converterVoltageReferenceQ =
315 mKpCurrCtrl * currentErrorQ + mKiCurrCtrl * gammaQ;
316
317 Matrix converterVoltageReferenceDq(2, 1);
318 converterVoltageReferenceDq << converterVoltageReferenceD,
319 converterVoltageReferenceQ;
320
321 output = inverseParkTransform * converterVoltageReferenceDq;
322}

324void EMT::Ph3::SSN_GFL_Split::calculateControllerAnalyticalJacobians(
325 const Matrix &xController, const Matrix &measurement, Matrix &A, Matrix &B,
326 Matrix &C, Matrix &D) const {
327
328 if (xController.rows() != mControllerStateSize || xController.cols() != 1)
329 throw std::invalid_argument(
330 "SSN_GFL_Split controller state has an invalid dimension.");
331
332 if (measurement.rows() != mControllerInputSize || measurement.cols() != 1)
333 throw std::invalid_argument(
334 "SSN_GFL_Split controller measurement has an invalid dimension.");
335
336 const Real thetaPLL = xController(ThetaPLL, 0);
337 const Real pFiltered = xController(PFiltered, 0);
338 const Real qFiltered = xController(QFiltered, 0);
339 const Real phiD = xController(PhiD, 0);
340 const Real phiQ = xController(PhiQ, 0);
341 const Real gammaD = xController(GammaD, 0);
342 const Real gammaQ = xController(GammaQ, 0);
343
344 const Matrix vcAbc = measurement.block(0, 0, 3, 1);
345 const Matrix iGridAbc = measurement.block(3, 0, 3, 1);
346
347 const Matrix parkTransform = getParkTransformMatrix(thetaPLL);
348 const Matrix tD = parkTransform.row(0);
349 const Matrix tQ = parkTransform.row(1);
350
351 const Matrix inverseParkTransform = getInverseParkTransformMatrix(thetaPLL);
352 const Matrix sD = inverseParkTransform.col(0);
353 const Matrix sQ = inverseParkTransform.col(1);
354
355 // Exact derivatives of the orthonormal Park transformation.
356 const Matrix dTdTheta = tQ;
357 const Matrix dTqTheta = -tD;
358 const Matrix dSdTheta = sQ;
359 const Matrix dSqTheta = -sD;
360
361 const Real vcD = (tD * vcAbc)(0, 0);
362 const Real vcQ = (tQ * vcAbc)(0, 0);
363 const Real iGridD = (tD * iGridAbc)(0, 0);
364 const Real iGridQ = (tQ * iGridAbc)(0, 0);
365
366 const Real dVcDByTheta = (dTdTheta * vcAbc)(0, 0);
367 const Real dVcQByTheta = (dTqTheta * vcAbc)(0, 0);
368 const Real dIGridDByTheta = (dTdTheta * iGridAbc)(0, 0);
369 const Real dIGridQByTheta = (dTqTheta * iGridAbc)(0, 0);
370
371 const Matrix dPByVc = iGridD * tD + iGridQ * tQ;
372 const Matrix dPByI = vcD * tD + vcQ * tQ;
373
374 const Real dPByTheta = iGridD * dVcDByTheta + vcD * dIGridDByTheta +
375 iGridQ * dVcQByTheta + vcQ * dIGridQByTheta;
376
377 const Matrix dQByVc = -iGridQ * tD + iGridD * tQ;
378 const Matrix dQByI = -vcD * tQ + vcQ * tD;
379
380 const Real dQByTheta = -iGridQ * dVcDByTheta - vcD * dIGridQByTheta +
381 iGridD * dVcQByTheta + vcQ * dIGridDByTheta;
382
383 const Real currentReferenceD =
384 mKpPowerCtrl * (mPRef - pFiltered) + mKiPowerCtrl * phiD;
385 const Real currentReferenceQ =
386 mKpPowerCtrl * (qFiltered - mQRef) + mKiPowerCtrl * phiQ;
387
388 const Real currentErrorD = currentReferenceD - iGridD;
389 const Real currentErrorQ = currentReferenceQ - iGridQ;
390
391 const Real converterVoltageReferenceD =
392 mKpCurrCtrl * currentErrorD + mKiCurrCtrl * gammaD;
393 const Real converterVoltageReferenceQ =
394 mKpCurrCtrl * currentErrorQ + mKiCurrCtrl * gammaQ;
395
396 A.setZero(mControllerStateSize, mControllerStateSize);
397 B.setZero(mControllerStateSize, mControllerInputSize);
398 C.setZero(mControllerOutputSize, mControllerStateSize);
399 D.setZero(mControllerOutputSize, mControllerInputSize);
400
401 // -----------------------------------------------------------------------
402 // Controller state Jacobian A_c and measurement Jacobian B_c
403 // -----------------------------------------------------------------------
404
405 // PLL
406 A(ThetaPLL, ThetaPLL) = mKpPLL * dVcQByTheta;
407 A(ThetaPLL, PhiPLL) = mKiPLL;
408 B.block(ThetaPLL, 0, 1, 3) = mKpPLL * tQ;
409
410 A(PhiPLL, ThetaPLL) = dVcQByTheta;
411 B.block(PhiPLL, 0, 1, 3) = tQ;
412
413 // Power filters
414 A(PFiltered, ThetaPLL) = mOmegaCutoff * dPByTheta;
415 A(PFiltered, PFiltered) = -mOmegaCutoff;
416 B.block(PFiltered, 0, 1, 3) = mOmegaCutoff * dPByVc;
417 B.block(PFiltered, 3, 1, 3) = mOmegaCutoff * dPByI;
418
419 A(QFiltered, ThetaPLL) = mOmegaCutoff * dQByTheta;
420 A(QFiltered, QFiltered) = -mOmegaCutoff;
421 B.block(QFiltered, 0, 1, 3) = mOmegaCutoff * dQByVc;
422 B.block(QFiltered, 3, 1, 3) = mOmegaCutoff * dQByI;
423
424 // Outer-loop integrators
425 A(PhiD, PFiltered) = -1.0;
426 A(PhiQ, QFiltered) = 1.0;
427
428 // Inner-loop integrators
429 A(GammaD, ThetaPLL) = -dIGridDByTheta;
430 A(GammaD, PFiltered) = -mKpPowerCtrl;
431 A(GammaD, PhiD) = mKiPowerCtrl;
432 B.block(GammaD, 3, 1, 3) = -tD;
433
434 A(GammaQ, ThetaPLL) = -dIGridQByTheta;
435 A(GammaQ, QFiltered) = mKpPowerCtrl;
436 A(GammaQ, PhiQ) = mKiPowerCtrl;
437 B.block(GammaQ, 3, 1, 3) = -tQ;
438
439 // -----------------------------------------------------------------------
440 // Controller output Jacobian C_c and D_c
441 //
442 // v_inv,abc = S(theta) [v_ref,d; v_ref,q]
443 // -----------------------------------------------------------------------
444 C.col(ThetaPLL) = dSdTheta * converterVoltageReferenceD +
445 dSqTheta * converterVoltageReferenceQ +
446 sD * (-mKpCurrCtrl * dIGridDByTheta) +
447 sQ * (-mKpCurrCtrl * dIGridQByTheta);
448
449 C.col(PFiltered) += sD * (-mKpCurrCtrl * mKpPowerCtrl);
450 C.col(PhiD) += sD * (mKpCurrCtrl * mKiPowerCtrl);
451 C.col(GammaD) += sD * mKiCurrCtrl;
452
453 C.col(QFiltered) += sQ * (mKpCurrCtrl * mKpPowerCtrl);
454 C.col(PhiQ) += sQ * (mKpCurrCtrl * mKiPowerCtrl);
455 C.col(GammaQ) += sQ * mKiCurrCtrl;
456
457 D.block(0, 3, 3, 3) = sD * (-mKpCurrCtrl * tD) + sQ * (-mKpCurrCtrl * tQ);
458}

460void EMT::Ph3::SSN_GFL_Split::buildControllerStateSpaceModel(
461 const Matrix &xController, const Matrix &measurement, Matrix &A, Matrix &B,
462 Matrix &C, Matrix &D, Matrix &E, Matrix &F) const {
463
464 calculateControllerAnalyticalJacobians(xController, measurement, A, B, C, D);
465
466 Matrix stateDerivative = Matrix::Zero(mControllerStateSize, 1);
467 Matrix output = Matrix::Zero(mControllerOutputSize, 1);
468
469 evaluateControllerStateDerivative(xController, measurement, stateDerivative);
470 evaluateControllerOutput(xController, measurement, output);
471
472 E = stateDerivative - A * xController - B * measurement;
473 F = output - C * xController - D * measurement;
474}

476void EMT::Ph3::SSN_GFL_Split::updateLogAttributes(const Matrix &u) const {
477 const Matrix measurement = buildControllerMeasurement(**mX, u);
478
479 const Real thetaPLL = mControllerState(ThetaPLL, 0);
480 const Matrix parkTransform = getParkTransformMatrix(thetaPLL);
481
482 const Matrix vcAbc = measurement.block(0, 0, 3, 1);
483 const Matrix iGridAbc = measurement.block(3, 0, 3, 1);
484
485 **mVcD = (parkTransform.row(0) * vcAbc)(0, 0);
486 **mVcQ = (parkTransform.row(1) * vcAbc)(0, 0);
487 **mIrcD = (parkTransform.row(0) * iGridAbc)(0, 0);
488 **mIrcQ = (parkTransform.row(1) * iGridAbc)(0, 0);
489
490 **mPInst = **mVcD * **mIrcD + **mVcQ * **mIrcQ;
491 **mQInst = -**mVcD * **mIrcQ + **mVcQ * **mIrcD;
492
493 **mOmegaPLL =
494 mOmegaN + mKpPLL * **mVcQ + mKiPLL * mControllerState(PhiPLL, 0);
495
496 **mVInvRefA = mControllerOutput(0, 0);
497 **mVInvRefB = mControllerOutput(1, 0);
498 **mVInvRefC = mControllerOutput(2, 0);
499}

501void EMT::Ph3::SSN_GFL_Split::initializeFromNodesAndTerminals(Real frequency) {
502
503 if (!mParametersSet)
504 throw std::logic_error("setParameters() must be called before "
505 "initializeFromNodesAndTerminals().");
506
507 // Initialize electrical and controller states at the same operating point.
508 const Real omega = 2.0 * PI * frequency;
509 const Complex imaginaryUnit(0.0, 1.0);
510 const Complex powerReference(mPRef, mQRef);
511
512 const MatrixComp uPhasor = buildInitialInputFromNodes(frequency);
513
514 MatrixComp vcPhasor = uPhasor;
515 MatrixComp injectionCurrentPhasor = MatrixComp::Zero(3, 1);
516
517 constexpr Int initializationMaxIterations = 10;
518 constexpr Real initializationTolerance = 1e-9;
519
520 for (Int iteration = 0; iteration < initializationMaxIterations;
521 ++iteration) {
522 const Complex vcA = vcPhasor(0, 0);
523
524 if (std::abs(vcA) < initializationTolerance) {
525 injectionCurrentPhasor.setZero();
526 break;
527 }
528
529 // Peak-valued phase phasors use S = 1.5*V_peak*conj(I_peak).
530 const Complex currentA = std::conj(powerReference / (1.5 * vcA));
531
532 MatrixComp nextInjectionCurrent(3, 1);
533 nextInjectionCurrent << currentA, currentA * SHIFT_TO_PHASE_B,
534 currentA * SHIFT_TO_PHASE_C;
535
536 const MatrixComp nextVcPhasor = uPhasor + mRc * nextInjectionCurrent;
537
538 injectionCurrentPhasor = nextInjectionCurrent;
539
540 if ((nextVcPhasor - vcPhasor).norm() < initializationTolerance) {
541 vcPhasor = nextVcPhasor;
542 break;
543 }
544
545 vcPhasor = nextVcPhasor;
546 }
547
548 const MatrixComp ifPhasor =
549 imaginaryUnit * omega * mCf * vcPhasor + injectionCurrentPhasor;
550
551 const MatrixComp converterVoltageReferencePhasor =
552 vcPhasor + (mRf + imaginaryUnit * omega * mLf) * ifPhasor;
553
554 const Matrix vcAbc0 = vcPhasor.real();
555 const Matrix ifAbc0 = ifPhasor.real();
556 const Matrix injectionCurrentAbc0 = injectionCurrentPhasor.real();
557 const Matrix converterVoltageReferenceAbc0 =
558 converterVoltageReferencePhasor.real();
559
560 const Real theta0 = std::arg(vcPhasor(0, 0));
561 const Matrix parkTransform = getParkTransformMatrix(theta0);
562
563 const Matrix vcDq0 = parkTransform * vcAbc0;
564 const Matrix injectionCurrentDq0 = parkTransform * injectionCurrentAbc0;
565 const Matrix converterVoltageReferenceDq0 =
566 parkTransform * converterVoltageReferenceAbc0;
567
568 const Real vcD0 = vcDq0(0, 0);
569 const Real vcQ0 = vcDq0(1, 0);
570 const Real iGridD0 = injectionCurrentDq0(0, 0);
571 const Real iGridQ0 = injectionCurrentDq0(1, 0);
572
573 const Real pInitial = vcD0 * iGridD0 + vcQ0 * iGridQ0;
574 const Real qInitial = -vcD0 * iGridQ0 + vcQ0 * iGridD0;
575
576 // -----------------------------------------------------------------------
577 // Network state
578 // -----------------------------------------------------------------------
579 Matrix xNetwork0 = Matrix::Zero(mNetworkStateSize, 1);
580 xNetwork0.block(VcA, 0, 3, 1) = vcAbc0;
581 xNetwork0.block(IfA, 0, 3, 1) = ifAbc0;
582
583 **mX = xNetwork0;
584
585 // -----------------------------------------------------------------------
586 // Controller state
587 // -----------------------------------------------------------------------
588 mControllerState.setZero();
589
590 mControllerState(ThetaPLL, 0) = theta0;
591 mControllerState(PhiPLL, 0) = (omega - mOmegaN) / mKiPLL;
592 mControllerState(PFiltered, 0) = pInitial;
593 mControllerState(QFiltered, 0) = qInitial;
594
595 mControllerState(PhiD, 0) =
596 (iGridD0 + mKpPowerCtrl * (pInitial - mPRef)) / mKiPowerCtrl;
597
598 mControllerState(PhiQ, 0) =
599 (iGridQ0 - mKpPowerCtrl * (qInitial - mQRef)) / mKiPowerCtrl;
600
601 const Real currentReferenceD = -mKpPowerCtrl * pInitial +
602 mKiPowerCtrl * mControllerState(PhiD, 0) +
603 mKpPowerCtrl * mPRef;
604
605 const Real currentReferenceQ = mKpPowerCtrl * qInitial +
606 mKiPowerCtrl * mControllerState(PhiQ, 0) -
607 mKpPowerCtrl * mQRef;
608
609 mControllerState(GammaD, 0) = (converterVoltageReferenceDq0(0, 0) +
610 mKpCurrCtrl * (iGridD0 - currentReferenceD)) /
611 mKiCurrCtrl;
612
613 mControllerState(GammaQ, 0) = (converterVoltageReferenceDq0(1, 0) +
614 mKpCurrCtrl * (iGridQ0 - currentReferenceQ)) /
615 mKiCurrCtrl;
616
617 // -----------------------------------------------------------------------
618 // Interface quantities and controller measurement
619 // -----------------------------------------------------------------------
620 **mIntfVoltage = uPhasor.real();
621 **mIntfCurrent = ((uPhasor - vcPhasor) / mRc).real();
622
623 mControllerMeasurementOld = buildControllerMeasurement(**mX, **mIntfVoltage);
624
625 // Avoid an artificial one-step zero command at t = 0.
626 mControllerOutput = converterVoltageReferenceAbc0;
627 mDelayedControllerOutput = converterVoltageReferenceAbc0;
628
629 // Build the controller model at the initialized operating point.
630 buildControllerStateSpaceModel(mControllerState, mControllerMeasurementOld,
631 mControllerA, mControllerB, mControllerC,
632 mControllerD, mControllerE, mControllerF);
633
634 // Usually mTimeStep is set later by mnaCompInitialize(), which will call
635 // recomputeDiscreteModel(). This also makes re-initialization robust if a
636 // timestep is already known.
637 if (mTimeStep > 0.0)
638 recomputeDiscreteModel();
639
640 updateLogAttributes(**mIntfVoltage);
641
642 SPDLOG_LOGGER_INFO(
643 mSLog,
644 "\n--- SSN GFL split initialization ---"
645 "\nInput u: {:s}"
646 "\nOutput y: {:s}"
647 "\nNetwork state x_n: {:s}"
648 "\nController state x_c: {:s}"
649 "\nP/Q init: [{:.6e}, {:.6e}]"
650 "\nVc dq: [{:.6e}, {:.6e}]"
651 "\nIgrid dq: [{:.6e}, {:.6e}]"
652 "\nInitial delayed v_inv: {:s}"
653 "\n--- SSN GFL split initialization finished ---",
654 Logger::matrixToString(**mIntfVoltage),
655 Logger::matrixToString(**mIntfCurrent), Logger::matrixToString(**mX),
656 Logger::matrixToString(mControllerState), pInitial, qInitial, vcD0, vcQ0,
657 iGridD0, iGridQ0, Logger::matrixToString(mDelayedControllerOutput));
658}
