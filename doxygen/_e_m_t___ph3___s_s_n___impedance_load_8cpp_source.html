39void EMT::Ph3::SSN::ImpedanceLoad::setParameters(Real activePower,
40 Real reactivePower,
41 Real nominalVoltage,
42 Real shuntConductance,
43 Real shuntSusceptance) {
44 if (!std::isfinite(activePower) || !std::isfinite(reactivePower) ||
45 !std::isfinite(nominalVoltage) || !std::isfinite(shuntConductance) ||
46 !std::isfinite(shuntSusceptance) ||
47 ((activePower != 0. || reactivePower != 0.) && nominalVoltage <= 0.))
48 throw std::invalid_argument("Invalid impedance-load parameters.");
49 mActivePower = activePower;
50 mReactivePower = reactivePower;
51 mNominalVoltage = nominalVoltage;
52 mShuntConductance = shuntConductance;
53 mShuntSusceptance = shuntSusceptance;
54 mLoadConductance = nominalVoltage > 0.
55 ? activePower / (nominalVoltage * nominalVoltage)
56 : 0.;
57 mTotalConductance = mLoadConductance + shuntConductance;
58 EMT::SSNComp::setParameters(Matrix::Zero(12, 12), Matrix::Zero(12, 3),
59 Matrix::Zero(3, 12), Matrix::Zero(3, 3));
60}

62void EMT::Ph3::SSN::ImpedanceLoad::initializeFromNodesAndTerminals(
63 Real frequency) {
64 const Real omega = 2. * PI * frequency;
65 const Real loadSusceptance =
66 mNominalVoltage > 0.
67 ? mReactivePower / (mNominalVoltage * mNominalVoltage)
68 : 0.;
69 const Real inductiveSusceptance =
70 std::max(loadSusceptance, 0.) + std::max(-mShuntSusceptance, 0.);
71 const Real capacitiveSusceptance =
72 std::max(-loadSusceptance, 0.) + std::max(mShuntSusceptance, 0.);
73 mLoadInductance =
74 inductiveSusceptance > 0. ? 1. / (omega * inductiveSusceptance) : 0.;
75 mLoadCapacitance = capacitiveSusceptance / omega;
76
77 MatrixComp voltagePhasor = MatrixComp::Zero(3, 1);
78 voltagePhasor(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
79 voltagePhasor(1, 0) = voltagePhasor(0, 0) * SHIFT_TO_PHASE_B;
80 voltagePhasor(2, 0) = voltagePhasor(0, 0) * SHIFT_TO_PHASE_C;
81 MatrixComp inductorCurrentPhasor = MatrixComp::Zero(3, 1);
82 MatrixComp capacitorCurrentPhasor = MatrixComp::Zero(3, 1);
83 if (mLoadInductance > 0.)
84 inductorCurrentPhasor =
85 Complex(0., -1. / (omega * mLoadInductance)) * voltagePhasor;
86 if (mLoadCapacitance > 0.)
87 capacitorCurrentPhasor =
88 Complex(0., omega * mLoadCapacitance) * voltagePhasor;
89
90 const Matrix voltage = voltagePhasor.real();
91 const Matrix inductorCurrent = inductorCurrentPhasor.real();
92 const Matrix capacitorCurrent = capacitorCurrentPhasor.real();
93 *mIntfVoltageValue = voltage;
94 mStateValue->setZero();
95 if (mLoadInductance > 0.) {
96 mStateValue->block<3, 1>(kLoadInductorVoltage, 0) = voltage;
97 mStateValue->block<3, 1>(kLoadInductorCurrent, 0) = inductorCurrent;
98 }
99 if (mLoadCapacitance > 0.) {
100 mStateValue->block<3, 1>(kLoadCapacitorVoltage, 0) = voltage;
101 mStateValue->block<3, 1>(kLoadCapacitorCurrent, 0) = capacitorCurrent;
102 }
103 *mIntfCurrentValue =
104 mTotalConductance * voltage + inductorCurrent + capacitorCurrent;
105}

107MatrixComp EMT::Ph3::SSN::ImpedanceLoad::buildInitialInputFromNodes(Real) {
108 MatrixComp voltage = MatrixComp::Zero(3, 1);
109 const Complex phaseA = RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
110 voltage(0, 0) = phaseA;
111 voltage(1, 0) = phaseA * SHIFT_TO_PHASE_B;
112 voltage(2, 0) = phaseA * SHIFT_TO_PHASE_C;
113 return voltage;

116void EMT::Ph3::SSN::ImpedanceLoad::recomputeDiscreteModel() {
117 mNodalAdmittance = mTotalConductance * Matrix::Identity(3, 3);
118 if (mLoadInductance > 0.) {
119 mInductorAdmittanceScalar = mTimeStep / (2. * mLoadInductance);
120 mInductorAdmittance = mInductorAdmittanceScalar * Matrix::Identity(3, 3);
121 mNodalAdmittance += mInductorAdmittance;
122 } else {
123 mInductorAdmittanceScalar = 0.;
124 mInductorAdmittance.setZero();
125 }
126 if (mLoadCapacitance > 0.) {
127 mCapacitorAdmittanceScalar = 2. * mLoadCapacitance / mTimeStep;
128 mCapacitorAdmittance = mCapacitorAdmittanceScalar * Matrix::Identity(3, 3);
129 mNodalAdmittance += mCapacitorAdmittance;
130 } else {
131 mCapacitorAdmittanceScalar = 0.;
132 mCapacitorAdmittance.setZero();
133 }
134 mW = mNodalAdmittance;
135 mdA.setZero(12, 12);
136 mdB.setZero(12, 3);
137}

163void EMT::Ph3::SSN::ImpedanceLoad::calculateHistoryVector(
164 Matrix &history) const {
165 mHistoryLoad.setZero();
166 mHistoryShunt.setZero();
167 if (mLoadInductance > 0.)
168 for (UInt phase = 0; phase < 3; ++phase)
169 mHistoryLoad(phase, 0) =
170 mInductorAdmittanceScalar * (*mStateValue)(phase, 0) +
171 (*mStateValue)(3 + phase, 0);
172 if (mLoadCapacitance > 0.)
173 for (UInt phase = 0; phase < 3; ++phase)
174 mHistoryShunt(phase, 0) =
175 -(*mStateValue)(9 + phase, 0) -
176 mCapacitorAdmittanceScalar * (*mStateValue)(6 + phase, 0);
177 history = mHistoryLoad + mHistoryShunt;
178}

180void EMT::Ph3::SSN::ImpedanceLoad::updateState(const Matrix &,
181 const Matrix &voltage) {
182 if (mLoadInductance > 0.) {
183 const PhaseVector current =
184 mInductorAdmittanceScalar * voltage + mHistoryLoad;
185 mStateValue->block<3, 1>(0, 0) = voltage;
186 mStateValue->block<3, 1>(3, 0) = current;
187 }
188 if (mLoadCapacitance > 0.) {
189 const PhaseVector current =
190 mCapacitorAdmittanceScalar * voltage + mHistoryShunt;
191 mStateValue->block<3, 1>(6, 0) = voltage;
192 mStateValue->block<3, 1>(9, 0) = current;
193 }
