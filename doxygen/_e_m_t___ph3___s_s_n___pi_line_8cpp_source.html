51void EMT::Ph3::SSN::PiLine::initializeFromNodesAndTerminals(Real frequency) {
52 if ((**mParallelCond)(0, 0) <= 0.) {
53 **mParallelCond = Matrix::Zero(3, 3);
54 (**mParallelCond).diagonal().setConstant(1e-6);
55 }
56
57 const Real omega = 2. * PI * frequency;
58 MatrixComp v0Phasor = MatrixComp::Zero(3, 1);
59 MatrixComp v1Phasor = MatrixComp::Zero(3, 1);
60 v0Phasor(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
61 v1Phasor(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(1);
62 v0Phasor(1, 0) = v0Phasor(0, 0) * SHIFT_TO_PHASE_B;
63 v0Phasor(2, 0) = v0Phasor(0, 0) * SHIFT_TO_PHASE_C;
64 v1Phasor(1, 0) = v1Phasor(0, 0) * SHIFT_TO_PHASE_B;
65 v1Phasor(2, 0) = v1Phasor(0, 0) * SHIFT_TO_PHASE_C;
66
67 const MatrixComp voltagePhasor = v1Phasor - v0Phasor;
68 const MatrixComp impedance =
69 (**mSeriesRes).cast<Complex>() +
70 Complex(0., omega) * (**mSeriesInd).cast<Complex>();
71 const MatrixComp currentPhasor = impedance.inverse() * voltagePhasor;
72 const MatrixComp inductorVoltagePhasor =
73 voltagePhasor - (**mSeriesRes).cast<Complex>() * currentPhasor;
74 const MatrixComp capHalf = ((**mParallelCap) / 2.).cast<Complex>();
75 const MatrixComp capCurrent0 = Complex(0., omega) * capHalf * v0Phasor;
76 const MatrixComp capCurrent1 = Complex(0., omega) * capHalf * v1Phasor;
77
78 *mIntfVoltageValue = Matrix::Zero(6, 1);
79 mIntfVoltageValue->block(0, 0, 3, 1) = v0Phasor.real();
80 mIntfVoltageValue->block(3, 0, 3, 1) = v1Phasor.real();
81 *mIntfCurrentValue = Matrix::Zero(6, 1);
82 mIntfCurrentValue->block(0, 0, 3, 1) =
83 -currentPhasor.real() + capCurrent0.real();
84 mIntfCurrentValue->block(3, 0, 3, 1) =
85 currentPhasor.real() + capCurrent1.real();
86 mStateValue->setZero();
87 mStateValue->block<3, 1>(kSeriesCurrent, 0) = currentPhasor.real();
88 mStateValue->block<3, 1>(kSeriesInductorVoltage, 0) =
89 inductorVoltagePhasor.real();
90 mStateValue->block<3, 1>(kCapVoltage0, 0) = v0Phasor.real();
91 mStateValue->block<3, 1>(kCapCurrent0, 0) = capCurrent0.real();
92 mStateValue->block<3, 1>(kCapVoltage1, 0) = v1Phasor.real();
93 mStateValue->block<3, 1>(kCapCurrent1, 0) = capCurrent1.real();
94}

96void EMT::Ph3::SSN::PiLine::recomputeDiscreteModel() {
97 const Real timeStep = mTimeStep;
98 mInductorAdmittance = (timeStep / 2.) * (**mSeriesInd).inverse();
99 mSeriesSystemInverse =
100 (Matrix::Identity(3, 3) + mInductorAdmittance * **mSeriesRes).inverse();
101 mSeriesAdmittance = mSeriesSystemInverse * mInductorAdmittance;
102 mCapAdmittance = (**mParallelCap) / timeStep;
103 const Matrix shunt = mCapAdmittance + (**mParallelCond) / 2.;
104 mW = Matrix::Zero(6, 6);
105 mW.block<3, 3>(0, 0) = mSeriesAdmittance + shunt;
106 mW.block<3, 3>(0, 3) = -mSeriesAdmittance;
107 mW.block<3, 3>(3, 0) = -mSeriesAdmittance;
108 mW.block<3, 3>(3, 3) = mSeriesAdmittance + shunt;
109 mdA.setZero(18, 18);
110 mdB.setZero(18, 6);
111}

113void EMT::Ph3::SSN::PiLine::calculateHistoryVector(Matrix &history) const {
114 mInductorHistory.noalias() =
115 mInductorAdmittance * mStateValue->block<3, 1>(kSeriesInductorVoltage, 0);
116 mInductorHistory += mStateValue->block<3, 1>(kSeriesCurrent, 0);
117 mSeriesHistory.noalias() = mSeriesSystemInverse * mInductorHistory;
118 mCapHistory0 = -mStateValue->block<3, 1>(kCapCurrent0, 0) -
119 mCapAdmittance * mStateValue->block<3, 1>(kCapVoltage0, 0);
120 mCapHistory1 = -mStateValue->block<3, 1>(kCapCurrent1, 0) -
121 mCapAdmittance * mStateValue->block<3, 1>(kCapVoltage1, 0);
122 history.resize(6, 1);
123 history.block(0, 0, 3, 1) = -mSeriesHistory + mCapHistory0;
124 history.block(3, 0, 3, 1) = mSeriesHistory + mCapHistory1;
125}

127void EMT::Ph3::SSN::PiLine::updateState(const Matrix &, const Matrix &input) {
128 const TerminalVoltage v0 = input.block(0, 0, 3, 1);
129 const TerminalVoltage v1 = input.block(3, 0, 3, 1);
130 const TerminalVoltage interfaceVoltage = v1 - v0;
131 const TerminalVoltage seriesCurrent =
132 mSeriesAdmittance * interfaceVoltage + mSeriesHistory;
133 const TerminalVoltage inductorVoltage =
134 interfaceVoltage - **mSeriesRes * seriesCurrent;
135 const TerminalVoltage capCurrent0 = mCapAdmittance * v0 + mCapHistory0;
136 const TerminalVoltage capCurrent1 = mCapAdmittance * v1 + mCapHistory1;
137 mStateValue->block<3, 1>(kSeriesCurrent, 0) = seriesCurrent;
138 mStateValue->block<3, 1>(kSeriesInductorVoltage, 0) = inductorVoltage;
139 mStateValue->block<3, 1>(kCapVoltage0, 0) = v0;
140 mStateValue->block<3, 1>(kCapCurrent0, 0) = capCurrent0;
141 mStateValue->block<3, 1>(kCapVoltage1, 0) = v1;
142 mStateValue->block<3, 1>(kCapCurrent1, 0) = capCurrent1;
143}
