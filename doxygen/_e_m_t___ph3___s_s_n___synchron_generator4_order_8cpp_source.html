23void EMT::Ph3::SSN::SynchronGenerator4Order::setParametersOperationalPerUnit(
24 Real nominalPower, Real nominalVoltage, Real nominalFrequency, Real inertia,
25 Real resistance, Real xdTransient, Real xqTransient, Real xd, Real xq,
26 Real td0Transient, Real tq0Transient, Real mechanicalDamping) {

60void EMT::Ph3::SSN::SynchronGenerator4Order::evaluate(
61 const Matrix &x, const Matrix &u, MatrixFixedSize<4, 1> &f,
62 MatrixFixedSize<3, 1> &y) const {
63 const Real theta = x(kTheta, 0);
64 const Real phaseStep = 2. * PI / 3.;
65 Real vd = 0., vq = 0.;
66 Real cosPhase[3], sinPhase[3];
67 for (Int p = 0; p < 3; ++p) {
68 const Real angle = theta - phaseStep * p;
69 cosPhase[p] = std::cos(angle);
70 sinPhase[p] = std::sin(angle);
71 vd += (2. / 3.) * cosPhase[p] * u(p, 0) / mBaseV;
72 vq -= (2. / 3.) * sinPhase[p] * u(p, 0) / mBaseV;
73 }
74 const Real dd = vd - x(kEd, 0), dq = vq - x(kEq, 0);
75 const Real det = mRa * mRa + mXdp * mXqp;
76 const Real id = (mRa * dd + mXqp * dq) / det;
77 const Real iq = (-mXdp * dd + mRa * dq) / det;
78 for (Int p = 0; p < 3; ++p)
79 y(p, 0) = mBaseI * (cosPhase[p] * id - sinPhase[p] * iq);
80
81 f.setZero(4, 1);
82 f(kEd, 0) = (-(x(kEd, 0) - mEd0) - (mXq - mXqp) * (iq - mIq0)) / mTq0p;
83 f(kEq, 0) = (-(x(kEq, 0) - mEq0) + (mXd - mXdp) * (id - mId0)) / mTd0p;
84 const Real pe = vd * id + vq * iq;
85 f(kOmega, 0) = (mPm + pe - mDamping * (x(kOmega, 0) - 1.)) / (2. * mH);
86 f(kTheta, 0) = mOmega * x(kOmega, 0);
87}

89void EMT::Ph3::SSN::SynchronGenerator4Order::linearize(const Matrix &x,
90 const Matrix &u,
91 Matrix &A, Matrix &B,
92 Matrix &C,
93 Matrix &D) const {
94 // This model used to be linearized with 14 finite-difference evaluations of
95 // evaluate() per timestep and generator. Besides the repeated arithmetic,
96 // that path rebuilt the Park transforms and several temporary Eigen
97 // matrices for every perturbation. The equations below are the exact
98 // analytical Jacobian of evaluate(), avoiding that hot-path overhead.
99 A.setZero(4, 4);
100 B.setZero(4, 3);
101 C.setZero(3, 4);
102 D.setZero(3, 3);
103 const Real theta = x(kTheta, 0);
104 const Real thetaStep = 2. * PI / 3.;
105 Real vd = 0., vq = 0.;
106 Real dVdDu[3], dVqDu[3];
107 Real cosPhase[3], sinPhase[3];
108 for (Int p = 0; p < 3; ++p) {
109 const Real angle = theta - thetaStep * p;
110 cosPhase[p] = std::cos(angle);
111 sinPhase[p] = std::sin(angle);
112 vd += (2. / 3.) * cosPhase[p] * u(p, 0) / mBaseV;
113 vq -= (2. / 3.) * sinPhase[p] * u(p, 0) / mBaseV;
114 dVdDu[p] = (2. / 3.) * cosPhase[p] / mBaseV;
115 dVqDu[p] = -(2. / 3.) * sinPhase[p] / mBaseV;
116 }
117
118 const Real det = mRa * mRa + mXdp * mXqp;
119 const Real id = (mRa * (vd - x(kEd, 0)) + mXqp * (vq - x(kEq, 0))) / det;
120 const Real iq = (-mXdp * (vd - x(kEd, 0)) + mRa * (vq - x(kEq, 0))) / det;
121 const Real idVd = mRa / det;
122 const Real idVq = mXqp / det;
123 const Real iqVd = -mXdp / det;
124 const Real iqVq = mRa / det;
125 const Real idEd = -idVd, idEq = -idVq;
126 const Real iqEd = -iqVd, iqEq = -iqVq;
127 const Real vdTheta = vq;
128 const Real vqTheta = -vd;
129 const Real idTheta = idVd * vdTheta + idVq * vqTheta;
130 const Real iqTheta = iqVd * vdTheta + iqVq * vqTheta;
131 const Real edScale = -(mXq - mXqp) / mTq0p;
132 const Real eqScale = (mXd - mXdp) / mTd0p;
133 const Real powerScale = 1. / (2. * mH);
134
135 A(0, 0) = -1. / mTq0p + edScale * iqEd;
136 A(0, 1) = edScale * iqEq;
137 A(0, 3) = edScale * iqTheta;
138 A(1, 0) = eqScale * idEd;
139 A(1, 1) = -1. / mTd0p + eqScale * idEq;
140 A(1, 3) = eqScale * idTheta;
141 A(2, 0) = powerScale * (vd * idEd + vq * iqEd);
142 A(2, 1) = powerScale * (vd * idEq + vq * iqEq);
143 A(2, 2) = -mDamping * powerScale;
144 A(2, 3) =
145 powerScale * (vdTheta * id + vd * idTheta + vqTheta * iq + vq * iqTheta);
146 A(3, 2) = mOmega;
147
148 for (Int p = 0; p < 3; ++p) {
149 B(0, p) = edScale * (iqVd * dVdDu[p] + iqVq * dVqDu[p]);
150 B(1, p) = eqScale * (idVd * dVdDu[p] + idVq * dVqDu[p]);
151 B(2, p) = powerScale * (vd * (idVd * dVdDu[p] + idVq * dVqDu[p]) +
152 vq * (iqVd * dVdDu[p] + iqVq * dVqDu[p]));
153 }
154
155 for (Int p = 0; p < 3; ++p) {
156 const Real rD = cosPhase[p];
157 const Real rQ = -sinPhase[p];
158 const Real drD = -sinPhase[p];
159 const Real drQ = -cosPhase[p];
160 C(p, 0) = mBaseI * (rD * idEd + rQ * iqEd);
161 C(p, 1) = mBaseI * (rD * idEq + rQ * iqEq);
162 C(p, 3) = mBaseI * (drD * id + drQ * iq + rD * idTheta + rQ * iqTheta);
163 for (Int j = 0; j < 3; ++j) {
164 const Real idU = idVd * dVdDu[j] + idVq * dVqDu[j];
165 const Real iqU = iqVd * dVdDu[j] + iqVq * dVqDu[j];
166 D(p, j) = mBaseI * (rD * idU + rQ * iqU);
167 }
168 }
169}

171Bool EMT::Ph3::SSN::SynchronGenerator4Order::updateComponentParameters() {
172 linearize(**mX, **mIntfVoltage, mA, mB, mC, mD);
173 evaluate(**mX, **mIntfVoltage, mEvaluationF, mEvaluationY);
174 mStateOffset = mEvaluationF;
175 mStateOffset.noalias() -= mA * (**mX);
176 mStateOffset.noalias() -= mB * (**mIntfVoltage);
177 mOutputOffset = mEvaluationY;
178 mOutputOffset.noalias() -= mC * (**mX);
179 mOutputOffset.noalias() -= mD * (**mIntfVoltage);
180 setStateOffset(mStateOffset);
181 setOutputOffset(mOutputOffset);
182 return true;
183}

185void EMT::Ph3::SSN::SynchronGenerator4Order::initializeState(
186 const Matrix &phaseVoltage) {
187 if (!mConfigured || !mInitialValuesConfigured)
188 throw std::logic_error(
189 "Configure SSN generator parameters and PF values first.");
190 const Complex v = mVpu * std::polar(1., mVangle);
191 const Complex s(mPm, mQm);
192 // Positive-sequence current is positive into the generator component.
193 const Complex i = -std::conj(s / v);
194 const Complex e = v - Complex(mRa, 0.5 * (mXdp + mXqp)) * i;
195 const Real theta = std::arg(e) - PI / 2.;
196 const Complex rot = std::polar(1., -theta);
197 const Complex vdq = v * rot, idq = i * rot;
198 (**mX)(kTheta, 0) = theta;
199 (**mX)(kOmega, 0) = 1.;
200 (**mX)(kEd, 0) = vdq.real() - mRa * idq.real() + mXqp * idq.imag();
201 (**mX)(kEq, 0) = vdq.imag() - mXdp * idq.real() - mRa * idq.imag();
202 mEd0 = (**mX)(kEd, 0);
203 mEq0 = (**mX)(kEq, 0);
204 mId0 = idq.real();
205 mIq0 = idq.imag();
206 (void)phaseVoltage;
207}

209void EMT::Ph3::SSN::SynchronGenerator4Order::initializeFromNodesAndTerminals(
210 Real) {
211 if (!mConfigured)
212 throw std::logic_error("SSN generator parameters are unset.");
213 MatrixComp ph = MatrixComp::Zero(3, 1);
214 ph(0, 0) = kSqrtTwoThirds * initialSingleVoltage(1);
215 ph(1, 0) = ph(0, 0) * SHIFT_TO_PHASE_B;
216 ph(2, 0) = ph(0, 0) * SHIFT_TO_PHASE_C;
217 const Complex solvedPower = -terminal(0)->singlePower();
218 if (std::isfinite(solvedPower.real()) && std::isfinite(solvedPower.imag()) &&
219 std::abs(solvedPower) > 1.) {
220 mPm = solvedPower.real() / mNominalPower;
221 mQm = solvedPower.imag() / mNominalPower;
222 }
223 mVpu = std::abs(ph(0, 0)) / mBaseV;
224 mVangle = std::arg(ph(0, 0));
225 **mIntfVoltage = ph.real();
226 initializeState(**mIntfVoltage);
227 updateComponentParameters();
228 evaluate(**mX, **mIntfVoltage, mEvaluationF, mEvaluationY);
229 **mIntfCurrent = mEvaluationY;
230}
