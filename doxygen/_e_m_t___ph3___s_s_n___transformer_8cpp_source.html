64void EMT::Ph3::SSN::Transformer::resolveWindingRoles() {
65 switch (mReferenceWinding) {
66 case WindingReference::Primary:
67 mReferenceTerminal = 0;
68 break;
69 case WindingReference::Secondary:
70 mReferenceTerminal = 1;
71 break;
72 case WindingReference::Tertiary:
73 throw std::invalid_argument("Three-winding transformers are unsupported.");
74 case WindingReference::Auto:
75 mReferenceTerminal =
76 (mNominalVoltagePrimary >= mNominalVoltageSecondary) ? 0 : 1;
77 break;
78 }
79 mRatioFromReference =
80 (mReferenceTerminal == 0) ? effectiveRatio() : 1. / effectiveRatio();
81 if (std::abs(std::arg(mRatioFromReference)) > 1e-9)
82 throw std::invalid_argument(
83 "EMT SSN transformers support in-phase turns ratios only.");
84}

86void EMT::Ph3::SSN::Transformer::initializeFromNodesAndTerminals(
87 Real frequency) {
88 if (frequency <= 0.)
89 throw std::invalid_argument("Frequency must be positive.");
90 resolveWindingRoles();
91 const Real omega = 2. * PI * frequency;
92 const Real magnetizingSusceptance =
93 std::sqrt(mNoLoadCurrent * mNoLoadCurrent - mNoLoadLoss * mNoLoadLoss) *
94 mRatedPower / std::pow(nominalVoltageAt(mReferenceTerminal), 2);
95 const Real magnetizingInductance = 1. / (omega * magnetizingSusceptance);
96 const Real magnetizingResistance =
97 std::pow(nominalVoltageAt(mReferenceTerminal), 2) /
98 (mNoLoadLoss * mRatedPower);
99 mMagnetizingInductance =
100 Math::singlePhaseParameterToThreePhase(magnetizingInductance);
101 mMagnetizingResistance =
102 Math::singlePhaseParameterToThreePhase(magnetizingResistance);
103
104 mSeriesInductanceHalf = mInductance / 2.;
105 mSeriesResistanceHalf.setZero();
106 if (mWithResistiveLosses)
107 mSeriesResistanceHalf = mResistance / 2.;
108
109 // State x = [i_ref; i_other], with the magnetizing current eliminated by
110 // midpoint KCL: i_mag = i_ref + i_other.
111 Matrix mass = Matrix::Zero(6, 6);
112 Matrix resistance = Matrix::Zero(6, 6);
113 const Matrix magnetizingI = mMagnetizingInductance;
114 const Matrix magnetizingR = mMagnetizingResistance;
115 mass.block<3, 3>(0, 0) = mSeriesInductanceHalf + magnetizingI;
116 mass.block<3, 3>(0, 3) = magnetizingI;
117 mass.block<3, 3>(3, 0) = magnetizingI;
118 mass.block<3, 3>(3, 3) = mSeriesInductanceHalf + magnetizingI;
119 resistance.block<3, 3>(0, 0) = mSeriesResistanceHalf + magnetizingR;
120 resistance.block<3, 3>(0, 3) = magnetizingR;
121 resistance.block<3, 3>(3, 0) = magnetizingR;
122 resistance.block<3, 3>(3, 3) = mSeriesResistanceHalf + magnetizingR;
123
124 const Matrix inverseMass = mass.inverse();
125 Matrix inputMap = Matrix::Zero(6, 6);
126 Matrix outputMap = Matrix::Zero(6, 6);
127 const Matrix identity = Matrix::Identity(3, 3);
128 const Real ratio = mRatioFromReference.real();
129 if (mReferenceTerminal == 0) {
130 inputMap.block<3, 3>(0, 0) = identity;
131 inputMap.block<3, 3>(3, 3) = ratio * identity;
132 outputMap.block<3, 3>(0, 0) = identity;
133 outputMap.block<3, 3>(3, 3) = ratio * identity;
134 } else {
135 inputMap.block<3, 3>(0, 3) = identity;
136 inputMap.block<3, 3>(3, 0) = ratio * identity;
137 outputMap.block<3, 3>(0, 3) = ratio * identity;
138 outputMap.block<3, 3>(3, 0) = identity;
139 }
140 SSNComp::setParameters(-inverseMass * resistance, inverseMass * inputMap,
141 outputMap, Matrix::Zero(6, 6));
142 MultiPortVTypeSSNComp::initializeFromNodesAndTerminals(frequency);
143}
