100void EMT::Ph3::SSN::Transformer::resolveWindingRoles() {
101 switch (mReferenceWinding) {
102 case WindingReference::Primary:
103 mReferenceTerminal = 0;
104 break;
105 case WindingReference::Secondary:
106 mReferenceTerminal = 1;
107 break;
108 case WindingReference::Tertiary:
109 throw std::invalid_argument("Three-winding transformers are unsupported.");
110 case WindingReference::Auto:
111 mReferenceTerminal =
112 (mNominalVoltagePrimary >= mNominalVoltageSecondary) ? 0 : 1;
113 break;
114 }
115 mOrientationSign = (mReferenceTerminal == 0) ? 1. : -1.;
116 mRatioFromReference =
117 (mReferenceTerminal == 0) ? effectiveRatio() : 1. / effectiveRatio();
118 if (std::abs(std::arg(mRatioFromReference)) > 1e-9)
119 throw std::invalid_argument(
120 "EMT SSN transformers support in-phase turns ratios only.");
121}

123void EMT::Ph3::SSN::Transformer::initializeFromNodesAndTerminals(
124 Real frequency) {
125 resolveWindingRoles();
126 const Real omega = 2. * PI * frequency;
127 const Real magnetizingSusceptance =
128 std::sqrt(mNoLoadCurrent * mNoLoadCurrent - mNoLoadLoss * mNoLoadLoss) *
129 mRatedPower / std::pow(nominalVoltageAt(mReferenceTerminal), 2);
130 const Real magnetizingInductance = 1. / (omega * magnetizingSusceptance);
131 const Real magnetizingResistance =
132 std::pow(nominalVoltageAt(mReferenceTerminal), 2) /
133 (mNoLoadLoss * mRatedPower);
134 mMagnetizingInductance =
135 Math::singlePhaseParameterToThreePhase(magnetizingInductance);
136 mMagnetizingResistance =
137 Math::singlePhaseParameterToThreePhase(magnetizingResistance);
138 mMagnetizingConductance = mMagnetizingResistance.inverse();
139
140 mSeriesInductanceHalf = mInductance / 2.;
141 mSeriesResistanceHalf.setZero();
142 if (mWithResistiveLosses)
143 mSeriesResistanceHalf = mResistance / 2.;
144
145 const PhaseVectorComp voltage0 =
146 phaseVoltage(RMS3PH_TO_PEAK1PH * initialSingleVoltage(0));
147 const PhaseVectorComp voltage1 =
148 phaseVoltage(RMS3PH_TO_PEAK1PH * initialSingleVoltage(1));
149 const PhaseVectorComp referenceVoltage =
150 (mReferenceTerminal == 0) ? voltage0 : voltage1;
151 const PhaseVectorComp otherVoltage =
152 (mReferenceTerminal == 0) ? voltage1 : voltage0;
153 const PhaseVectorComp referredOtherVoltage =
154 mRatioFromReference.real() * otherVoltage;
155
156 const MatrixComp halfImpedance =
157 mSeriesResistanceHalf.cast<Complex>() +
158 Complex(0., omega) * mSeriesInductanceHalf.cast<Complex>();
159 const MatrixComp seriesAdmittance = halfImpedance.inverse();
160 const MatrixComp magnetizingAdmittance =
161 mMagnetizingConductance.cast<Complex>() +
162 Complex(0., -1. / omega) *
163 mMagnetizingInductance.inverse().cast<Complex>();
164 const MatrixComp midpointInverse =
165 (2. * seriesAdmittance + magnetizingAdmittance).inverse();
166 const PhaseVectorComp midpointVoltage =
167 midpointInverse * seriesAdmittance *
168 (referenceVoltage + referredOtherVoltage);
169 const PhaseVectorComp current0 =
170 seriesAdmittance * (midpointVoltage - referenceVoltage);
171 const PhaseVectorComp current1 =
172 seriesAdmittance * (referredOtherVoltage - midpointVoltage);
173 const PhaseVectorComp currentMagnetizing =
174 (magnetizingAdmittance - mMagnetizingConductance.cast<Complex>()) *
175 (-midpointVoltage);
176
177 const PhaseVectorComp voltageL0 =
178 Complex(0., omega) * mSeriesInductanceHalf.cast<Complex>() * current0;
179 const PhaseVectorComp voltageL1 =
180 Complex(0., omega) * mSeriesInductanceHalf.cast<Complex>() * current1;
181 const PhaseVectorComp voltageLMag = Complex(0., omega) *
182 mMagnetizingInductance.cast<Complex>() *
183 currentMagnetizing;
184 mStateValue->block<3, 1>(kVoltage0, 0) = voltageL0.real();
185 mStateValue->block<3, 1>(kCurrent0, 0) = current0.real();
186 mStateValue->block<3, 1>(kVoltage1, 0) = voltageL1.real();
187 mStateValue->block<3, 1>(kCurrent1, 0) = current1.real();
188 mStateValue->block<3, 1>(kVoltageMag, 0) = voltageLMag.real();
189 mStateValue->block<3, 1>(kCurrentMag, 0) = currentMagnetizing.real();
190 *mIntfVoltageValue = Matrix::Zero(6, 1);
191 mIntfVoltageValue->block(0, 0, 3, 1) = voltage0.real();
192 mIntfVoltageValue->block(3, 0, 3, 1) = voltage1.real();
193 *mIntfCurrentValue = Matrix::Zero(6, 1);
194 if (mReferenceTerminal == 0) {
195 mIntfCurrentValue->block(0, 0, 3, 1) = -current0.real();
196 mIntfCurrentValue->block(3, 0, 3, 1) = -current1.real();
197 } else {
198 mIntfCurrentValue->block(3, 0, 3, 1) = -current0.real();
199 mIntfCurrentValue->block(0, 0, 3, 1) = -current1.real();
200 }
201}

203void EMT::Ph3::SSN::Transformer::recomputeDiscreteModel() {
204 const Real timeStep = mTimeStep;
205 mInductorAdmittance = (timeStep / 2.) * mSeriesInductanceHalf.inverse();
206 mSeriesSystemInverse =
207 (Matrix::Identity(3, 3) + mInductorAdmittance * mSeriesResistanceHalf)
208 .inverse();
209 mSeriesAdmittance = mSeriesSystemInverse * mInductorAdmittance;
210 mMagnetizingInductorAdmittance =
211 (timeStep / 2.) * mMagnetizingInductance.inverse();
212 mMidpointInverse = (2. * mSeriesAdmittance + mMagnetizingInductorAdmittance +
213 mMagnetizingConductance)
214 .inverse();
215 const Matrix diagonal =
216 mSeriesAdmittance * mMidpointInverse * mSeriesAdmittance;
217 mPortAdmittance00 = mSeriesAdmittance - diagonal;
218 mPortAdmittance01 = -diagonal * mRatioFromReference.real();
219 mPortAdmittance10 = mPortAdmittance01;
220 mPortAdmittance11 = std::norm(mRatioFromReference) * mPortAdmittance00;
221 mW = Matrix::Zero(6, 6);
222 if (mReferenceTerminal == 0) {
223 mW.block<3, 3>(0, 0) = mPortAdmittance00;
224 mW.block<3, 3>(0, 3) = mPortAdmittance01;
225 mW.block<3, 3>(3, 0) = mPortAdmittance10;
226 mW.block<3, 3>(3, 3) = mPortAdmittance11;
227 } else {
228 mW.block<3, 3>(0, 0) = mPortAdmittance11;
229 mW.block<3, 3>(0, 3) = mPortAdmittance10;
230 mW.block<3, 3>(3, 0) = mPortAdmittance01;
231 mW.block<3, 3>(3, 3) = mPortAdmittance00;
232 }
233 mdA.setZero(18, 18);
234 mdB.setZero(18, 6);
235}

237void EMT::Ph3::SSN::Transformer::calculateHistoryVector(Matrix &history) const {
238 const auto &state = *mStateValue;
239 mHistory0 = mSeriesSystemInverse *
240 (mInductorAdmittance * state.block<3, 1>(kVoltage0, 0) +
241 state.block<3, 1>(kCurrent0, 0));
242 mHistory1 = mSeriesSystemInverse *
243 (mInductorAdmittance * state.block<3, 1>(kVoltage1, 0) +
244 state.block<3, 1>(kCurrent1, 0));
245 mHistoryMagnetizing =
246 mMagnetizingInductorAdmittance * state.block<3, 1>(kVoltageMag, 0) +
247 state.block<3, 1>(kCurrentMag, 0);
248 mMidpointHistory = -mHistory0 + mHistory1 + mHistoryMagnetizing;
249 mMidpointSource = mMidpointInverse * mMidpointHistory;
250 mSource0 = mHistory0 + mSeriesAdmittance * mMidpointSource;
251 mSource1 = mRatioFromReference.real() *
252 (mSeriesAdmittance * mMidpointSource - mHistory1);
253 history = Matrix::Zero(6, 1);
254 const UInt ref = 3 * mReferenceTerminal;
255 const UInt other = 3 * (1 - mReferenceTerminal);
256 history.block<3, 1>(ref, 0) = -mSource0;
257 history.block<3, 1>(other, 0) = -mSource1;
258}

260void EMT::Ph3::SSN::Transformer::updateState(const Matrix &,
261 const Matrix &input) {
262 const PhaseVector voltage0 = input.block<3, 1>(0, 0);
263 const PhaseVector voltage1 = input.block<3, 1>(3, 0);
264 const PhaseVector referenceVoltage =
265 (mReferenceTerminal == 0) ? voltage0 : voltage1;
266 const PhaseVector otherVoltage =
267 (mReferenceTerminal == 0) ? voltage1 : voltage0;
268 const PhaseVector referredOtherVoltage =
269 mRatioFromReference.real() * otherVoltage;
270 const PhaseVector midpointVoltage =
271 mMidpointInverse *
272 (mSeriesAdmittance * referenceVoltage +
273 mSeriesAdmittance * referredOtherVoltage + mMidpointHistory);
274 const PhaseVector totalVoltage0 = midpointVoltage - referenceVoltage;
275 const PhaseVector current0 = mSeriesAdmittance * totalVoltage0 + mHistory0;
276 const PhaseVector voltageL0 =
277 totalVoltage0 - mSeriesResistanceHalf * current0;
278 const PhaseVector totalVoltage1 = referredOtherVoltage - midpointVoltage;
279 const PhaseVector current1 = mSeriesAdmittance * totalVoltage1 + mHistory1;
280 const PhaseVector voltageL1 =
281 totalVoltage1 - mSeriesResistanceHalf * current1;
282 const PhaseVector voltageLMag = -midpointVoltage;
283 const PhaseVector currentMagnetizing =
284 mMagnetizingInductorAdmittance * voltageLMag + mHistoryMagnetizing;
285 mStateValue->block<3, 1>(kVoltage0, 0) = voltageL0;
286 mStateValue->block<3, 1>(kCurrent0, 0) = current0;
287 mStateValue->block<3, 1>(kVoltage1, 0) = voltageL1;
288 mStateValue->block<3, 1>(kCurrent1, 0) = current1;
289 mStateValue->block<3, 1>(kVoltageMag, 0) = voltageLMag;
290 mStateValue->block<3, 1>(kCurrentMag, 0) = currentMagnetizing;
291}
