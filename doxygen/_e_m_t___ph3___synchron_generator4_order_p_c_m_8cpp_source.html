14MatrixFixedSize<3, 1> parkTransformFixed(Real theta, const Matrix &abcVector) {
15 MatrixFixedSize<3, 3> abcToDq0;
16 MatrixFixedSize<3, 1> abc;
17 abc << abcVector(0, 0), abcVector(1, 0), abcVector(2, 0);
18
19 const Real cosTheta = std::cos(theta);
20 const Real sinTheta = std::sin(theta);
21 constexpr Real sqrtThreeOverTwo = 0.86602540378443864676;
22 const Real cosMinus120 = -0.5 * cosTheta + sqrtThreeOverTwo * sinTheta;
23 const Real sinMinus120 = -0.5 * sinTheta - sqrtThreeOverTwo * cosTheta;
24 const Real cosPlus120 = -0.5 * cosTheta - sqrtThreeOverTwo * sinTheta;
25 const Real sinPlus120 = -0.5 * sinTheta + sqrtThreeOverTwo * cosTheta;
26
27 abcToDq0 << 2. / 3. * cosTheta, 2. / 3. * cosMinus120, 2. / 3. * cosPlus120,
28 -2. / 3. * sinTheta, -2. / 3. * sinMinus120, -2. / 3. * sinPlus120,
29 1. / 3., 1. / 3., 1. / 3.;
30
31 return abcToDq0 * abc;
32}

34MatrixFixedSize<3, 1> inverseParkTransformFixed(Real theta,
35 const Matrix &dq0Vector) {
36 MatrixFixedSize<3, 3> dq0ToAbc;
37 MatrixFixedSize<3, 1> dq0;
38 dq0 << dq0Vector(0, 0), dq0Vector(1, 0), dq0Vector(2, 0);
39
40 const Real cosTheta = std::cos(theta);
41 const Real sinTheta = std::sin(theta);
42 constexpr Real sqrtThreeOverTwo = 0.86602540378443864676;
43 const Real cosMinus120 = -0.5 * cosTheta + sqrtThreeOverTwo * sinTheta;
44 const Real sinMinus120 = -0.5 * sinTheta - sqrtThreeOverTwo * cosTheta;
45 const Real cosPlus120 = -0.5 * cosTheta - sqrtThreeOverTwo * sinTheta;
46 const Real sinPlus120 = -0.5 * sinTheta + sqrtThreeOverTwo * cosTheta;
47
48 dq0ToAbc << cosTheta, -sinTheta, 1., cosMinus120, -sinMinus120, 1.,
49 cosPlus120, -sinPlus120, 1.;
50
51 return dq0ToAbc * dq0;
52}

98void EMT::Ph3::SynchronGenerator4OrderPCM::calculateStateSpaceMatrices() {
99 // Initialize matrices of state representation
100 mAStateSpace << -mLq / mTq0_t / mLq_t, 0.0, 0.0, 0, -mLd / mTd0_t / mLd_t,
101 0.0, 0, 0.0, 0.0;
102 mBStateSpace << (mLq - mLq_t) / mTq0_t / mLq_t, 0.0, 0.0, 0.0,
103 (mLd - mLd_t) / mTd0_t / mLd_t, 0.0, 0.0, 0.0, 0.0;
104 mCStateSpace << 0.0, 1. / mTd0_t, 0.0;
105 // Precalculate trapezoidal based matrices (avoids redundant matrix inversions in correction steps)
106 Math::calculateStateSpaceTrapezoidalMatrices(
107 mAStateSpace, mBStateSpace, mCStateSpace, mTimeStep, mAdTrapezoidal,
108 mBdTrapezoidal, mCdTrapezoidal);
109}

111void EMT::Ph3::SynchronGenerator4OrderPCM::stepInPerUnit() {
112 // set number of iterations equal to zero
113 **mNumIter = 0;
114
115 // Predictor step (euler)
116
117 // store values currently at t=k for later use
118 mEdq0tPrevStep = **mEdq0_t;
119 mIdq0PrevStep = **mIdq0;
120 mVdq0PrevStep = **mVdq0;
121
122 // predict emf at t=k+1 (euler) using
123 (**mEdq0_t) = Math::StateSpaceEuler(**mEdq0_t, mAStateSpace, mBStateSpace,
124 mCStateSpace * **mEf, mTimeStep, **mVdq0);
125
126 // predict stator currents at t=k+1 (assuming Vdq0(k+1)=Vdq0(k))
127 (**mIdq0)(0, 0) = ((**mEdq0_t)(1, 0) - (**mVdq0)(1, 0)) / mLd_t;
128 (**mIdq0)(1, 0) = ((**mVdq0)(0, 0) - (**mEdq0_t)(0, 0)) / mLq_t;
129 (**mIdq0)(2, 0) = 0.0;
130
131 // convert currents into the abc domain
132 **mIntfCurrent = inverseParkTransformFixed(**mThetaMech, **mIdq0);
133 **mIntfCurrent = **mIntfCurrent * mBase_I;
134
135 // The current equations have voltage Jacobian
136 // d(Id, Iq) / d(Vd, Vq) = [[0, -1/Ld_t], [1/Lq_t, 0]]. Stamp its
137 // negative in MNA so that the source-side compensation cancels this
138 // voltage dependence in the network iteration. The transform depends on
139 // rotor position and therefore the matrix is updated every step.
140 Matrix dq0ToAbc(3, 3);
141 Matrix abcToDq0(3, 3);
142 const Real theta = **mThetaMech;
143 dq0ToAbc << cos(theta), -sin(theta), 1., cos(theta - 2. * PI / 3.),
144 -sin(theta - 2. * PI / 3.), 1., cos(theta + 2. * PI / 3.),
145 -sin(theta + 2. * PI / 3.), 1.;
146 abcToDq0 << 2. / 3. * cos(theta), 2. / 3. * cos(theta - 2. * PI / 3.),
147 2. / 3. * cos(theta + 2. * PI / 3.), -2. / 3. * sin(theta),
148 -2. / 3. * sin(theta - 2. * PI / 3.),
149 -2. / 3. * sin(theta + 2. * PI / 3.), 1. / 3., 1. / 3., 1. / 3.;
150 Matrix dq0Conductance = Matrix::Zero(3, 3);
151 dq0Conductance(0, 1) = 1.0 / mLd_t;
152 dq0Conductance(1, 0) = -1.0 / mLq_t;
153 mNortonConductance = dq0ToAbc * dq0Conductance * abcToDq0 / mBase_Z;
154}

156void EMT::Ph3::SynchronGenerator4OrderPCM::mnaCompApplyRightSideVectorStamp(
157 Matrix &rightVector) {
158 const Matrix compensationCurrent = mNortonConductance * **mIntfVoltage;
159 Math::setVectorElement(rightVector, matrixNodeIndex(0, 0),
160 (**mIntfCurrent)(0, 0) + compensationCurrent(0, 0));
161 Math::setVectorElement(rightVector, matrixNodeIndex(0, 1),
162 (**mIntfCurrent)(1, 0) + compensationCurrent(1, 0));
163 Math::setVectorElement(rightVector, matrixNodeIndex(0, 2),
164 (**mIntfCurrent)(2, 0) + compensationCurrent(2, 0));
165}

191void EMT::Ph3::SynchronGenerator4OrderPCM::correctorStep() {
192 // increase number of iterations
193 **mNumIter = **mNumIter + 1;
194
195 // correct electrical vars
196 // calculate emf at j and k+1 (trapezoidal rule)
197 (**mEdq0_t) = Math::applyStateSpaceTrapezoidalMatrices(
198 mAdTrapezoidal, mBdTrapezoidal, mCdTrapezoidal * **mEf, mEdq0tPrevStep,
199 **mVdq0, mVdq0PrevStep);
200
201 // calculate corrected stator currents at t=k+1 (assuming Vdq(k+1)=VdqPrevIter(k+1))
202 (**mIdq0)(0, 0) = ((**mEdq0_t)(1, 0) - (**mVdq0)(1, 0)) / mLd_t;
203 (**mIdq0)(1, 0) = ((**mVdq0)(0, 0) - (**mEdq0_t)(0, 0)) / mLq_t;
204
205 // convert currents into the abc domain
206 **mIntfCurrent = inverseParkTransformFixed(**mThetaMech, **mIdq0);
207 **mIntfCurrent = **mIntfCurrent * mBase_I;
208
209 // stamp currents
210 mnaCompApplyRightSideVectorStamp(**mRightVector);
211}

213void EMT::Ph3::SynchronGenerator4OrderPCM::updateVoltage(
214 const Matrix &leftVector) {
215 // store voltage value currently at j-1 for later use
216 mVdq0PrevIter = **mVdq0;
217
218 (**mIntfVoltage)(0, 0) =
219 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 0));
220 (**mIntfVoltage)(1, 0) =
221 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 1));
222 (**mIntfVoltage)(2, 0) =
223 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 2));
224
225 // convert Vdq into abc domain
226 **mVdq0 = parkTransformFixed(**mThetaMech, **mIntfVoltage);
227 **mVdq0 = **mVdq0 / mBase_V;
228}

230bool EMT::Ph3::SynchronGenerator4OrderPCM::requiresIteration() {
231 if (**mNumIter >= mMaxIter) {
232 // maximum number of iterations reached
233 return false;
234 } else if (**mNumIter == 0) {
235 // no corrector step has been performed yet,
236 // convergence cannot be confirmed
237 return true;
238 } else {
239 // check voltage convergence according to tolerance
240 if (Math::abs((**mVdq0)(0, 0) - mVdq0PrevIter(0, 0)) > mTolerance ||
241 Math::abs((**mVdq0)(1, 0) - mVdq0PrevIter(1, 0)) > mTolerance)
242 return true;
243 else
244 return false;
245 }
