33EMT::Ph3::SynchronGenerator4OrderVBR::metalDeviceData() {
34 if (!mModelAsNortonSource || mHasExciter || mHasPSS || mHasTurbineGovernor ||
35 mHasGovernorAndTurbine)
36 throw std::logic_error(
37 "Metal SG4 VBR export supports only controller-free Norton models.");
38
39 MetalDeviceData data{};
40 for (UInt i = 0; i < 3; ++i) {
41 data.voltageDq0[i] = (**mVdq0)(i, 0);
42 data.currentDq0[i] = (**mIdq0)(i, 0);
43 data.transientVoltageDq0[i] = (**mEdq0_t)(i, 0);
44 data.matrixNodeIndices[i] = matrixNodeIndex(0, i);
45 }
46 for (UInt row = 0; row < 3; ++row)
47 for (UInt col = 0; col < 3; ++col) {
48 data.conductanceDq0[row * 3 + col] = mConductanceMatrixDq0(row, col);
49 }
50 // During DPsim initialize() the first component pre-step has not run yet,
51 // therefore mConductanceMatrix can still be zero. Export the exact stamp
52 // that calculateResistanceMatrix() will create on the first SG4 pre-step.
53 const MatrixFixedSize<3, 3> abcToDq = get_parkTransformMatrix();
54 const MatrixFixedSize<3, 3> dqToAbc = get_inverseParkTransformMatrix();
55 const MatrixFixedSize<3, 3> conductanceAbc =
56 dqToAbc * mConductanceMatrixDq0 * abcToDq;
57 for (UInt row = 0; row < 3; ++row)
58 for (UInt col = 0; col < 3; ++col)
59 data.conductanceAbc[row * 3 + col] = conductanceAbc(row, col);
60
61 data.timeStep = mTimeStep;
62 data.baseVoltage = mBase_V;
63 data.baseCurrent = mBase_I;
64 data.baseImpedance = mBase_Z;
65 data.baseOmegaElec = mBase_OmElec;
66 data.baseOmegaMech = mBase_OmMech;
67 data.inertia = mH;
68 data.damping = mDamping;
69 data.ldTransient = mLd_t;
70 data.lqTransient = mLq_t;
71 data.adTransient = mAd_t;
72 data.bdTransient = mBd_t;
73 data.aqTransient = mAq_t;
74 data.bqTransient = mBq_t;
75 data.dqTransient = mDq_t;
76 data.fieldVoltage = **mEf;
77 data.previousFieldVoltage = mEf_prev;
78 data.mechanicalTorque = **mMechTorque;
79 data.previousMechanicalTorque = mMechTorque_prev;
80 data.electricalTorque = **mElecTorque;
81 data.rotorSpeed = **mOmMech;
82 data.rotorAngle = **mThetaMech;
83 data.loadAngle = **mDelta;
84 return data;
85}

87void EMT::Ph3::SynchronGenerator4OrderVBR::specificInitialization() {
88 // initial voltage behind the transient reactance in the dq0 reference frame
89 (**mEdq0_t)(0, 0) = (**mVdq0)(0, 0) - (**mIdq0)(1, 0) * mLq_t;
90 (**mEdq0_t)(1, 0) = (**mVdq0)(1, 0) + (**mIdq0)(0, 0) * mLd_t;
91
92 SPDLOG_LOGGER_INFO(mSLog,
93 "\n--- Model specific initialization ---"
94 "\nSG model: 4th order"
95 "\nInitial Ed_t (per unit): {:f}"
96 "\nInitial Eq_t (per unit): {:f}"
97 "\n--- Model specific initialization finished ---",
98
99 (**mEdq0_t)(0, 0), (**mEdq0_t)(1, 0));
100 mSLog->flush();
101}

103void EMT::Ph3::SynchronGenerator4OrderVBR::stepInPerUnit() {
104
105 const Matrix &idq0 = *mIdq0Value;
106 const Matrix &vdq0 = *mVdq0Value;
107
108 if (mSimTime > 0.0) {
109 // calculate Edq_t at t=k
110 (**mEdq0_t)(0, 0) = -idq0(1, 0) * mLq_t + vdq0(0, 0);
111 (**mEdq0_t)(1, 0) = idq0(0, 0) * mLd_t + vdq0(1, 0);
112 (**mEdq0_t)(2, 0) = 0.0;
113 }
114
115 updateParkTransformMatrices();
116
117 // calculate resistance matrix at t=k+1
118 calculateResistanceMatrix();
119
120 // VBR history voltage
121 mEhs_vbr(0, 0) = mAd_t * idq0(1, 0) + mBd_t * (**mEdq0_t)(0, 0);
122 mEhs_vbr(1, 0) = mAq_t * idq0(0, 0) + mBq_t * (**mEdq0_t)(1, 0) +
123 mDq_t * mEf_prev + mDq_t * *mEfValue;
124 mEhs_vbr(2, 0) = 0.0;
125
126 // convert Edq_t into the abc reference frame
127 mEvbr = mDq0ToAbc * mEhs_vbr * mBase_V;
128}

136EMT::Ph3::HipSG4State EMT::Ph3::SynchronGenerator4OrderVBR::exportHipState() {
137 if (!supportsHipBatch())
138 throw std::invalid_argument(
139 "HIP SG4 batch supports only ungrounded Norton generators without "
140 "controllers");
141 HipSG4State state;
142 state.theta = **mThetaMech;
143 state.delta = **mDelta;
144 state.omega = **mOmMech;
145 state.electricalTorque = **mElecTorque;
146 state.mechanicalTorque = **mMechTorque;
147 state.previousMechanicalTorque = mMechTorque_prev;
148 state.fieldVoltage = **mEf;
149 state.previousFieldVoltage = mEf_prev;
150 state.timestep = mTimeStep;
151 state.inertia = mH;
152 state.damping = mDamping;
153 state.baseOmegaMech = mBase_OmMech;
154 state.baseVoltage = mBase_V;
155 state.baseCurrent = mBase_I;
156 state.lqTransient = mLq_t;
157 state.ldTransient = mLd_t;
158 state.adTransient = mAd_t;
159 state.bdTransient = mBd_t;
160 state.aqTransient = mAq_t;
161 state.bqTransient = mBq_t;
162 state.dqTransient = mDq_t;
163 for (int index = 0; index < 3; ++index) {
164 state.idq0[index] = (**mIdq0)(index, 0);
165 state.vdq0[index] = (**mVdq0)(index, 0);
166 state.edq0Transient[index] = (**mEdq0_t)(index, 0);
167 state.busRows[index] = static_cast<int>(matrixNodeIndex(0, index));
168 }
169 for (int col = 0; col < 3; ++col)
170 for (int row = 0; row < 3; ++row)
171 state.conductanceDq0[col * 3 + row] = mConductanceMatrixDq0(row, col);
172 state.grounded = 0;
173 return state;
174}

176void EMT::Ph3::SynchronGenerator4OrderVBR::applyHipPreState(
177 const HipSG4State &state, Real time) {
178 if (mEvbr.size() != 3)
179 mEvbr.resize(3, 1);
180 if (mIvbr.size() != 3)
181 mIvbr.resize(3, 1);
182 mSimTime = time;
183 **mThetaMech = state.theta;
184 **mDelta = state.delta;
185 **mOmMech = state.omega;
186 **mElecTorque = state.electricalTorque;
187 **mMechTorque = state.mechanicalTorque;
188 mMechTorque_prev = state.previousMechanicalTorque;
189 **mEf = state.fieldVoltage;
190 mEf_prev = state.previousFieldVoltage;
191 for (int index = 0; index < 3; ++index) {
192 (**mEdq0_t)(index, 0) = state.edq0Transient[index];
193 mEhs_vbr(index, 0) = state.historyDq0[index];
194 mEvbr(index, 0) = state.emfAbc[index];
195 mIvbr(index, 0) = state.currentSourceAbc[index];
196 }
197 for (int col = 0; col < 3; ++col)
198 for (int row = 0; row < 3; ++row)
199 mConductanceMatrix(row, col) = state.conductanceAbc[col * 3 + row];
200 (**mRightVector).setZero();
201 if (!state.grounded)
202 for (int phase = 0; phase < 3; ++phase)
203 (**mRightVector)(state.busRows[phase], 0) = state.currentSourceAbc[phase];
204}
