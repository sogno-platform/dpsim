53void EMT::Ph3::SynchronGeneratorDQTrapez::stepInPerUnit(Real time) {
54 // Receives the voltage of time step k and has to
55 // calculate the current output for time step k+1
56
57 // Calculate per unit values and
58 // transform per unit voltages from abc to dq0
59 mVdq0 = abcToDq0Transform(mThetaMech, **mIntfVoltage) / mBase_V;
60 mVsr(0, 0) = mVdq0(0, 0);
61 mVsr(3, 0) = mVdq0(1, 0);
62 mVsr(6, 0) = mVdq0(2, 0);
63 // Current is already available in pu dq0 from previous step
64
65 // Update of fd winding voltage from exciter
66 if (mHasExciter) {
67 Real Vpss = mHasPSS ? mPSS->step(**mOmMech, **mElecTorque, mVsr(0, 0),
68 mVsr(3, 0), mTimeStep)
69 : 0.0;
70 // Note: scaled by Rfd/Lmd to transform from exciter pu system
71 // to the synchronous generator pu system
72 mVsr(1, 0) =
73 (mRfd / mLmd) * mExciter->step(mVsr(0, 0), mVsr(3, 0), mTimeStep, Vpss);
74 }
75
76 // Update of mechanical torque from turbine governor
77 if (mHasGovernorAndTurbine) {
78 Real Pgv = mGovernor->step(**mOmMech, mTimeStep);
79 **mMechTorque = mTurbine->step(Pgv, mTimeStep);
80 } else if (mHasTurbineGovernorType1)
81 **mMechTorque = mTurbineGovernorType1->step(**mOmMech, mTimeStep);
82 else if (mHasTurbineGovernor)
83 **mMechTorque = mTurbineGovernor->step(
84 **mOmMech, 1, mInitElecPower.real() / mNomPower, mTimeStep);
85
86 // Calculation of electrical torque
87 **mElecTorque = (mPsisr(3, 0) * mIsr(0, 0) - mPsisr(0, 0) * mIsr(3, 0));
88
89 // Update mechanical rotor angle with respect to electrical angle
90 // using Euler and previous states
91 mThetaMech = mThetaMech + mTimeStep * (**mOmMech * mBase_OmMech);
92
93 // Update of omega using Euler
94 **mOmMech = **mOmMech + mTimeStep * (1. / (2. * **mInertia) *
95 (**mMechTorque - **mElecTorque));
96
97 // Update using euler
98 **mDelta = **mDelta + mTimeStep * (**mOmMech - 1);
99
100 // Integrate the machine states as a linear function of the newly solved
101 // terminal voltage. The Norton matrix below carries that voltage
102 // dependence into the network solve, avoiding an unstable explicit
103 // current-source/line-capacitance feedback loop.
104 mPreviousFluxes = mPsisr;
105 const Matrix stateMatrix =
106 mBase_OmElec * (mFluxStateSpaceMat + mOmegaFluxMat * **mOmMech);
107 const Matrix identity = Matrix::Identity(mPsisr.rows(), mPsisr.rows());
108 const Matrix lhs = identity - (mTimeStep / 2.0) * stateMatrix;
109 const Matrix lhsInverse = lhs.inverse();
110 mFluxF = lhsInverse * (identity + (mTimeStep / 2.0) * stateMatrix);
111
112 Matrix inputSelector = Matrix::Zero(mPsisr.rows(), mPsisr.rows());
113 inputSelector(0, 0) = 1.0;
114 inputSelector(3, 3) = 1.0;
115 inputSelector(6, 6) = 1.0;
116 mFluxK = lhsInverse * (mTimeStep * mBase_OmElec) * inputSelector;
117 mPsisr = mFluxF * mPreviousFluxes + mFluxK * mVsr;
118 mIsr = mFluxToCurrentMat * mPsisr;
119
120 Matrix currentSelector = Matrix::Zero(3, mIsr.rows());
121 Matrix voltageSelector = Matrix::Zero(mVsr.rows(), 3);
122 currentSelector(0, 0) = 1.0;
123 currentSelector(1, 3) = 1.0;
124 currentSelector(2, 6) = 1.0;
125 voltageSelector(0, 0) = 1.0;
126 voltageSelector(3, 1) = 1.0;
127 voltageSelector(6, 2) = 1.0;
128 const Matrix dqJacobian =
129 currentSelector * mFluxToCurrentMat * mFluxK * voltageSelector;
130 const Real theta = mThetaMech;
131 Matrix dq0ToAbc(3, 3);
132 Matrix abcToDq0(3, 3);
133 dq0ToAbc << cos(theta), -sin(theta), 1., cos(theta - 2. * PI / 3.),
134 -sin(theta - 2. * PI / 3.), 1., cos(theta + 2. * PI / 3.),
135 -sin(theta + 2. * PI / 3.), 1.;
136 abcToDq0 << 2. / 3. * cos(theta), 2. / 3. * cos(theta - 2. * PI / 3.),
137 2. / 3. * cos(theta + 2. * PI / 3.), -2. / 3. * sin(theta),
138 -2. / 3. * sin(theta - 2. * PI / 3.),
139 -2. / 3. * sin(theta + 2. * PI / 3.), 1. / 3., 1. / 3., 1. / 3.;
140 mNortonConductance = (mBase_I / mBase_V) * dq0ToAbc * dqJacobian * abcToDq0;
141 mIdq0(0, 0) = mIsr(0, 0);
142 mIdq0(1, 0) = mIsr(3, 0);
143 mIdq0(2, 0) = mIsr(6, 0);
144 **mIntfCurrent = mBase_I * dq0ToAbcTransform(mThetaMech, mIdq0);
145}

147void EMT::Ph3::SynchronGeneratorDQTrapez::mnaCompApplySystemMatrixStamp(
148 SparseMatrixRow &systemMatrix) {
149 if (!terminalNotGrounded(0))
150 return;
151 for (UInt row = 0; row < 3; ++row)
152 for (UInt col = 0; col < 3; ++col)
153 Math::addToMatrixElement(systemMatrix, matrixNodeIndex(0, row),
154 matrixNodeIndex(0, col),
155 mNortonConductance(row, col));
156}

158void EMT::Ph3::SynchronGeneratorDQTrapez::mnaCompApplyRightSideVectorStamp(
159 Matrix &rightVector) {
160 const Matrix compensationCurrent = mNortonConductance * **mIntfVoltage;
161 for (UInt phase = 0; phase < 3; ++phase)
162 Math::setVectorElement(rightVector, matrixNodeIndex(0, phase),
163 -(**mIntfCurrent)(phase, 0) +
164 compensationCurrent(phase, 0));
165}

167void EMT::Ph3::SynchronGeneratorDQTrapez::mnaCompPostStep(
168 Real time, Int timeStepCount, Attribute<Matrix>::Ptr &leftVector) {
169 mnaCompUpdateVoltage(**leftVector);
170 mVdq0 = abcToDq0Transform(mThetaMech, **mIntfVoltage) / mBase_V;
171 mVsr(0, 0) = mVdq0(0, 0);
172 mVsr(3, 0) = mVdq0(1, 0);
173 mVsr(6, 0) = mVdq0(2, 0);
174
175 mPsisr = mFluxF * mPreviousFluxes + mFluxK * mVsr;
176 mIsr = mFluxToCurrentMat * mPsisr;
177 mIdq0(0, 0) = mIsr(0, 0);
178 mIdq0(1, 0) = mIsr(3, 0);
179 mIdq0(2, 0) = mIsr(6, 0);
180 **mIntfCurrent = mBase_I * dq0ToAbcTransform(mThetaMech, mIdq0);
181}
