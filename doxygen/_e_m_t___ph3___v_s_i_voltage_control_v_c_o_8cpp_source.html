12using ParkTransform = MatrixFixedSize<2, 3>;
13using InverseParkTransform = MatrixFixedSize<3, 2>;
14using PhaseVector = MatrixFixedSize<3, 1>;
15using DqVector = MatrixFixedSize<2, 1>;
16
17ParkTransform makePowerInvariantParkTransform(Real theta) {
18 const Real k = std::sqrt(2.0 / 3.0);
19 ParkTransform transform;
20 transform << k * std::cos(theta), k * std::cos(theta - 2.0 * M_PI / 3.0),
21 k * std::cos(theta + 2.0 * M_PI / 3.0), -k * std::sin(theta),
22 -k * std::sin(theta - 2.0 * M_PI / 3.0),
23 -k * std::sin(theta + 2.0 * M_PI / 3.0);
24 return transform;
25}
26
27InverseParkTransform makePowerInvariantInverseParkTransform(Real theta) {
28 const Real k = std::sqrt(2.0 / 3.0);
29 InverseParkTransform transform;
30 transform << k * std::cos(theta), -k * std::sin(theta),
31 k * std::cos(theta - 2.0 * M_PI / 3.0),
32 -k * std::sin(theta - 2.0 * M_PI / 3.0),
33 k * std::cos(theta + 2.0 * M_PI / 3.0),
34 -k * std::sin(theta + 2.0 * M_PI / 3.0);
35 return transform;
36}

252void EMT::Ph3::VSIVoltageControlVCO::initializeParentFromNodesAndTerminals(
253 Real frequency) {
254
255 // use complex interface quantities for initialization calculations
256 MatrixComp intfVoltageComplex = Matrix::Zero(3, 1);
257 MatrixComp intfCurrentComplex = Matrix::Zero(3, 1);
258 // terminal powers in consumer system -> convert to generator system
259 Real activePower = terminal(0)->singlePower().real();
260 Real reactivePower = terminal(0)->singlePower().imag();
261
262 // derive complex threephase initialization from single phase initial values (only valid for balanced systems)
263 intfVoltageComplex(0, 0) = RMS3PH_TO_PEAK1PH * initialSingleVoltage(0);
264 intfVoltageComplex(1, 0) = intfVoltageComplex(0, 0) * SHIFT_TO_PHASE_B;
265 intfVoltageComplex(2, 0) = intfVoltageComplex(0, 0) * SHIFT_TO_PHASE_C;
266 intfCurrentComplex(0, 0) = -std::conj(
267 2. / 3. * Complex(activePower, reactivePower) / intfVoltageComplex(0, 0));
268 intfCurrentComplex(1, 0) = intfCurrentComplex(0, 0) * SHIFT_TO_PHASE_B;
269 intfCurrentComplex(2, 0) = intfCurrentComplex(0, 0) * SHIFT_TO_PHASE_C;
270
271 MatrixComp filterInterfaceInitialVoltage = MatrixComp::Zero(3, 1);
272 MatrixComp filterInterfaceInitialCurrent = MatrixComp::Zero(3, 1);
273 if (mWithConnectionTransformer) {
274 // calculate quantities of low voltage side of transformer (being the interface quantities of the filter, calculations only valid for symmetrical systems)
275 filterInterfaceInitialVoltage =
276 (intfVoltageComplex -
277 Complex(mTransformerResistance, mTransformerInductance * **mOmegaN) *
278 intfCurrentComplex) /
279 Complex(mTransformerRatioAbs, mTransformerRatioPhase);
280 filterInterfaceInitialCurrent =
281 intfCurrentComplex *
282 Complex(mTransformerRatioAbs, mTransformerRatioPhase);
283
284 // connect and init transformer
285 mVirtualNodes[3]->setInitialVoltage(PEAK1PH_TO_RMS3PH *
286 filterInterfaceInitialVoltage);
287 mConnectionTransformer->connect({mTerminals[0]->node(), mVirtualNodes[3]});
288 } else {
289 // if no transformer used, filter interface equal to inverter interface
290 filterInterfaceInitialVoltage = intfVoltageComplex;
291 filterInterfaceInitialCurrent = intfCurrentComplex;
292 }
293
294 // derive initialization quantities of filter (calculations only valid for symmetrical systems)
295 MatrixComp vcInit = filterInterfaceInitialVoltage -
296 filterInterfaceInitialCurrent * Complex(mRc, 0);
297 MatrixComp icfInit = vcInit * Complex(0., 2. * PI * frequency * mCf);
298 MatrixComp vfInit = vcInit - (filterInterfaceInitialCurrent - icfInit) *
299 Complex(0., 2. * PI * frequency * mLf);
300 MatrixComp vsInit =
301 vfInit - (filterInterfaceInitialCurrent - icfInit) * Complex(mRf, 0);
302 mVirtualNodes[0]->setInitialVoltage(PEAK1PH_TO_RMS3PH * vsInit);
303 mVirtualNodes[1]->setInitialVoltage(PEAK1PH_TO_RMS3PH * vfInit);
304 mVirtualNodes[2]->setInitialVoltage(PEAK1PH_TO_RMS3PH * vcInit);
305
306 // save real interface quantities calculated from complex ones
307 **mIntfVoltage = intfVoltageComplex.real();
308 **mIntfCurrent = intfCurrentComplex.real();
309 **mElecActivePower = (3. / 2. * intfVoltageComplex(0, 0) *
310 std::conj(-intfCurrentComplex(0, 0)))
311 .real();
312 **mElecReactivePower = (3. / 2. * intfVoltageComplex(0, 0) *
313 std::conj(-intfCurrentComplex(0, 0)))
314 .imag();
315
316 if (!mUseSubcomponents) {
317 mDirectVlf = (vcInit - vfInit).real();
318 mDirectVcf = vcInit.real();
319 mDirectIf = (filterInterfaceInitialCurrent - icfInit).real();
320 mDirectIcf = -icfInit.real();
321 }
322
323 // Initialize controlled source
324 if (mUseSubcomponents) {
325 mSubCtrledVoltageSource->setParameters(mVirtualNodes[0]->initialVoltage(),
326 0.0);
327 mSubCtrledVoltageSource->connect({SimNode::GND, mVirtualNodes[0]});
328 mSubResistorF->connect({mVirtualNodes[0], mVirtualNodes[1]});
329 mSubInductorF->connect({mVirtualNodes[1], mVirtualNodes[2]});
330 mSubCapacitorF->connect({mVirtualNodes[2], SimNode::GND});
331 if (mWithConnectionTransformer)
332 mSubResistorC->connect({mVirtualNodes[2], mVirtualNodes[3]});
333 else
334 mSubResistorC->connect({mVirtualNodes[2], mTerminals[0]->node()});
335
336 for (auto subcomp : mSubComponents) {
337 subcomp->initialize(mFrequencies);
338 subcomp->initializeFromNodesAndTerminals(frequency);
339 }
340 } else {
341 mDirectSourceVoltage = vsInit.real();
342 mVs->set(mDirectSourceVoltage);
343 }
344
345 if (mWithConnectionTransformer) {
346 // Initialize control subcomponents
347 // current and voltage inputs to VCO and voltage controller
348 Matrix vcdq, ircdq;
349 Real theta = std::arg(mVirtualNodes[3]->initialSingleVoltage());
350 vcdq = Math::parkTransformPowerInvariant(
351 theta, filterInterfaceInitialVoltage.real());
352 ircdq = Math::parkTransformPowerInvariant(
353 theta, -1 * (filterInterfaceInitialCurrent - icfInit).real());
354
355 **mVcd = vcdq(0, 0);
356 **mVcq = vcdq(1, 0);
357 **mIrcd = ircdq(0, 0);
358 **mIrcq = ircdq(1, 0);
359
360 // VCO start settings
361 mVCO->setInitialValues(**mOmegaN, theta, theta);
362 } else {
363 // Initialize control subcomponents
364 // current and voltage inputs to VCO and voltage controller
365 Matrix vcdq, ircdq;
366 Real theta = std::arg(mVirtualNodes[2]->initialSingleVoltage());
367 vcdq = Math::parkTransformPowerInvariant(
368 theta, filterInterfaceInitialVoltage.real());
369 ircdq = Math::parkTransformPowerInvariant(
370 theta, -1 * (filterInterfaceInitialCurrent - icfInit).real());
371
372 **mVcd = vcdq(0, 0);
373 **mVcq = vcdq(1, 0);
374 **mIrcd = ircdq(0, 0);
375 **mIrcq = ircdq(1, 0);
376
377 // VCO start settings
378 mVCO->setInitialValues(**mOmegaN, theta, theta);
379 }
380 if (mWithControl && !mManualInitialStatesSet) {
381 const Real theta0 = std::arg(
382 mWithConnectionTransformer ? mVirtualNodes[3]->initialSingleVoltage()
383 : mVirtualNodes[2]->initialSingleVoltage());
384 const Matrix sourceDq =
385 Math::parkTransformPowerInvariant(theta0, vsInit.real());
386 const Real phiDInit =
387 (**mIrcd + mKpVoltageCtrl * (**mVcd - **mVdRef)) / mKiVoltageCtrl;
388 const Real phiQInit =
389 (**mIrcq + mKpVoltageCtrl * (**mVcq - **mVqRef)) / mKiVoltageCtrl;
390 const Real iRefD =
391 mKpVoltageCtrl * (**mVdRef - **mVcd) + mKiVoltageCtrl * phiDInit;
392 const Real iRefQ =
393 mKpVoltageCtrl * (**mVqRef - **mVcq) + mKiVoltageCtrl * phiQInit;
394 const Real gammaDInit =
395 (sourceDq(0, 0) - **mVcd + mKpCurrCtrl * (**mIrcd - iRefD)) /
396 mKiCurrCtrl;
397 const Real gammaQInit =
398 (sourceDq(1, 0) - **mVcq + mKpCurrCtrl * (**mIrcq - iRefQ)) /
399 mKiCurrCtrl;
400 mVoltageControllerVSI->setInitialStateValues(phiDInit, phiQInit, gammaDInit,
401 gammaQInit);
402 }
403
404 SPDLOG_LOGGER_INFO(
405 mSLog,
406 "\n--- Initialization from powerflow ---"
407 "\nInterface voltage across: {:s}"
408 "\nInterface current: {:s}"
409 "\nTerminal 0 initial voltage: {:s}"
410 "\nTerminal 0 connected to {:s} = sim node {:d}"
411 "\nVirtual node 0 initial voltage: {:s}"
412 "\nVirtual node 1 initial voltage: {:s}"
413 "\nVirtual node 2 initial voltage: {:s}",
414 Logger::phasorToString(intfVoltageComplex(0, 0)),
415 Logger::phasorToString(intfCurrentComplex(0, 0)),
416 Logger::phasorToString(initialSingleVoltage(0)),
417 mTerminals[0]->node()->name(), mTerminals[0]->node()->matrixNodeIndex(),
418 Logger::phasorToString(mVirtualNodes[0]->initialSingleVoltage()),
419 Logger::phasorToString(mVirtualNodes[1]->initialSingleVoltage()),
420 Logger::phasorToString(mVirtualNodes[2]->initialSingleVoltage()));
421 if (mWithConnectionTransformer)
422 SPDLOG_LOGGER_INFO(
423 mSLog, "\nVirtual node 3 initial voltage: {:s}",
424 Logger::phasorToString(mVirtualNodes[3]->initialSingleVoltage()));
425 SPDLOG_LOGGER_INFO(mSLog, "\n--- Initialization from powerflow finished ---");
426}

428void EMT::Ph3::VSIVoltageControlVCO::mnaParentInitialize(
429 Real omega, Real timeStep, Attribute<Matrix>::Ptr leftVector) {
430 mTimeStep = timeStep;
431
432 if (!mUseSubcomponents) {
433 mDirectRf = Matrix::Identity(3, 3) / mRf;
434 mDirectLf = (timeStep / (2. * mLf)) * Matrix::Identity(3, 3);
435 mDirectCf = (2. * mCf / timeStep) * Matrix::Identity(3, 3);
436 mDirectRc = Matrix::Identity(3, 3) / mRc;
437 }
438
439 // initialize state space controller
440 mVoltageControllerVSI->initializeStateSpaceModel(omega, timeStep, leftVector);
441 mVCO->setSimulationParameters(timeStep);
442
443 // TODO: these are actually no MNA tasks
444 mMnaTasks.push_back(std::make_shared<ControlPreStep>(*this));
445 mMnaTasks.push_back(std::make_shared<ControlStep>(*this));
446}

448void EMT::Ph3::VSIVoltageControlVCO::mnaParentApplySystemMatrixStamp(
449 SparseMatrixRow &systemMatrix) {
450 if (mUseSubcomponents)
451 return;
452 updateMatrixNodeIndices();
453 const UInt source = mVirtualNodes[0]->matrixNodeIndex(PhaseType::A);
454 const UInt filter = mVirtualNodes[1]->matrixNodeIndex(PhaseType::A);
455 const UInt capacitor = mVirtualNodes[2]->matrixNodeIndex(PhaseType::A);
456 const UInt sourceCurrent = mVirtualNodes[3]->matrixNodeIndex(PhaseType::A);
457 MNAStampUtils::stampConductanceMatrix(mDirectRf, systemMatrix, source, filter,
458 true, true, mSLog);
459 MNAStampUtils::stampConductanceMatrix(mDirectLf, systemMatrix, filter,
460 capacitor, true, true, mSLog);
461 MNAStampUtils::stampConductanceMatrix(mDirectCf, systemMatrix, capacitor, 0,
462 true, false, mSLog);
463 MNAStampUtils::stampConductanceMatrix(mDirectRc, systemMatrix, capacitor,
464 matrixNodeIndex(0), true,
465 terminalNotGrounded(0), mSLog);
466 for (UInt phase = 0; phase < 3; ++phase) {
467 Math::addToMatrixElement(systemMatrix, source + phase,
468 sourceCurrent + phase, 1.);
469 Math::addToMatrixElement(systemMatrix, sourceCurrent + phase,
470 source + phase, 1.);
471 }
472}

474void EMT::Ph3::VSIVoltageControlVCO::mnaParentApplyRightSideVectorStamp(
475 Matrix &rightVector) {
476 if (mUseSubcomponents)
477 return;
478 const UInt filter = mVirtualNodes[1]->matrixNodeIndex(PhaseType::A);
479 const UInt capacitor = mVirtualNodes[2]->matrixNodeIndex(PhaseType::A);
480 const UInt sourceCurrent = mVirtualNodes[3]->matrixNodeIndex(PhaseType::A);
481 for (UInt phase = 0; phase < 3; ++phase) {
482 Math::setVectorElement(rightVector, sourceCurrent + phase,
483 mDirectSourceVoltage(phase));
484 Math::setVectorElement(rightVector, filter + phase,
485 mDirectHistoryLf(phase));
486 Math::setVectorElement(rightVector, capacitor + phase,
487 mDirectHistoryCf(phase) - mDirectHistoryLf(phase));
488 }
489}

524void EMT::Ph3::VSIVoltageControlVCO::controlStep(Real time, Int timeStepCount) {
525 // Transformation interface forward
526 const Real theta = mVCO->mOutputPrev->get();
527 const ParkTransform park = makePowerInvariantParkTransform(theta);
528 const PhaseVector capacitorVoltage = **mVirtualNodes[2]->mVoltage;
529 PhaseVector filterCurrent;
530 if (mUseSubcomponents)
531 filterCurrent = -**mSubResistorF->mIntfCurrent;
532 else
533 filterCurrent = -mDirectIf;
534 const DqVector vcdq = park * capacitorVoltage;
535 const DqVector ircdq = park * filterCurrent;
536
537 const ParkTransform networkPark = makePowerInvariantParkTransform(mThetaN);
538 const PhaseVector interfaceVoltage = **mIntfVoltage;
539 const PhaseVector interfaceCurrent = **mIntfCurrent;
540 const DqVector intfVoltageDQ = networkPark * interfaceVoltage;
541 const DqVector intfCurrentDQ = networkPark * interfaceCurrent;
542 **mElecActivePower = -(intfVoltageDQ(0, 0) * intfCurrentDQ(0, 0) +
543 intfVoltageDQ(1, 0) * intfCurrentDQ(1, 0));
544 **mElecReactivePower = -(intfVoltageDQ(1, 0) * intfCurrentDQ(0, 0) -
545 intfVoltageDQ(0, 0) * intfCurrentDQ(1, 0));
546 //vector of voltages
547 **mVcd = vcdq(0, 0);
548 **mVcq = vcdq(1, 0);
549
550 //vector of currents
551 **mIrcd = ircdq(0, 0);
552 **mIrcq = ircdq(1, 0);
553
554 // add step of subcomponents
555 mVCO->signalStep(time, timeStepCount);
556 mVoltageControllerVSI->signalStep(time, timeStepCount);
557
558 // Transformation interface backward
559 DqVector voltageControllerOutput;
560 voltageControllerOutput = mVoltageControllerVSI->mOutputCurr->get();
561 **mVsref =
562 makePowerInvariantInverseParkTransform(theta) * voltageControllerOutput;
563 mThetaN = std::fmod(mThetaN + mTimeStep * **mOmegaN, 2 * PI);
564}

613//Current update
614void EMT::Ph3::VSIVoltageControlVCO::mnaCompUpdateCurrent(
615 const Matrix &leftvector) {
616 if (mWithConnectionTransformer)
617 **mIntfCurrent = mConnectionTransformer->mIntfCurrent->get();
618 else if (mUseSubcomponents)
619 **mIntfCurrent = mSubResistorC->mIntfCurrent->get();
620 else {
621 PhaseVector terminalVoltage = PhaseVector::Zero();
622 if (terminalNotGrounded(0))
623 for (UInt phase = 0; phase < 3; ++phase)
624 terminalVoltage(phase) =
625 Math::realFromVectorElement(leftvector, matrixNodeIndex(0, phase));
626 const UInt capacitor = mVirtualNodes[2]->matrixNodeIndex(PhaseType::A);
627 PhaseVector capacitorVoltage = PhaseVector::Zero();
628 for (UInt phase = 0; phase < 3; ++phase)
629 capacitorVoltage(phase) =
630 Math::realFromVectorElement(leftvector, capacitor + phase);
631 const PhaseVector voltageDifference = terminalVoltage - capacitorVoltage;
632 PhaseVector current;
633 current.noalias() = mDirectRc * voltageDifference;
634 **mIntfCurrent = current;
635 }
636}

638//Voltage update
639void EMT::Ph3::VSIVoltageControlVCO::mnaCompUpdateVoltage(
640 const Matrix &leftVector) {
641 if (mUseSubcomponents)
642 for (auto virtualNode : mVirtualNodes)
643 virtualNode->mnaUpdateVoltage(leftVector);
644 else {
645 for (UInt node = 0; node < 3; ++node)
646 mVirtualNodes[node]->mnaUpdateVoltage(leftVector);
647 const UInt filter = mVirtualNodes[1]->matrixNodeIndex(PhaseType::A);
648 const UInt capacitor = mVirtualNodes[2]->matrixNodeIndex(PhaseType::A);
649 for (UInt phase = 0; phase < 3; ++phase) {
650 mDirectVlf(phase) =
651 Math::realFromVectorElement(leftVector, capacitor + phase) -
652 Math::realFromVectorElement(leftVector, filter + phase);
653 mDirectVcf(phase) =
654 Math::realFromVectorElement(leftVector, capacitor + phase);
655 mDirectIf(phase) =
656 mDirectLf(phase, phase) * mDirectVlf(phase) + mDirectHistoryLf(phase);
657 mDirectIcf(phase) = -mDirectCf(phase, phase) * mDirectVcf(phase) +
658 mDirectHistoryCf(phase);
659 }
660 }
661 (**mIntfVoltage)(0, 0) =
662 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 0));
663 (**mIntfVoltage)(1, 0) =
664 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 1));
665 (**mIntfVoltage)(2, 0) =
666 Math::realFromVectorElement(leftVector, matrixNodeIndex(0, 2));
667}
