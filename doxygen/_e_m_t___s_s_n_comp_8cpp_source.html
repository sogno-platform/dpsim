39void EMT::SSNComp::setParameters(const Matrix &A, const Matrix &B,
40 const Matrix &C, const Matrix &D) {
41 mParametersSet = false;
42
43 if (A.rows() != A.cols())
44 throw std::invalid_argument("A must be square.");
45
46 if (B.rows() != A.rows() || B.cols() != mInputSize)
47 throw std::invalid_argument("B has invalid dimensions.");
48
49 if (C.rows() != mOutputSize || C.cols() != A.rows())
50 throw std::invalid_argument("C has invalid dimensions.");
51
52 if (D.rows() != mOutputSize || D.cols() != mInputSize)
53 throw std::invalid_argument("D has invalid dimensions.");
54
55 mA = A;
56 mB = B;
57 mC = C;
58 mD = D;
59
60 **mX = Matrix::Zero(mA.rows(), 1);
61
62 mdA = Matrix::Zero(mA.rows(), mA.cols());
63 mdB = Matrix::Zero(mB.rows(), mB.cols());
64
65 mW = Matrix::Zero(mOutputSize, mInputSize);
66 mYHist = Matrix::Zero(mOutputSize, 1);
67
68 mParametersSet = true;
69}

71void EMT::SSNComp::calculateHistoryVector(Matrix &history) const {
72 // Algebraic SSN components have no dynamic state. Their discrete history is
73 // identically zero; feedthrough remains represented by mW = D and is
74 // handled by the component's MNA stamp/update path.
75 if (mA.rows() == 0) {
76 history.setZero();
77 return;
78 }
79
80 mHistoryState.noalias() = mdA * stateValue();
81 mHistoryInput.noalias() = mdB * inputValue();
82 mHistoryState += mHistoryInput;
83 history.noalias() = mC * mHistoryState;
84}

86MatrixComp
87EMT::SSNComp::calculateSteadyStateStateFromInput(const MatrixComp &u,
88 Real frequency) const {
89 const Real omega = 2.0 * PI * frequency;
90 MatrixComp h =
91 Complex(0.0, omega) * MatrixComp::Identity(mA.rows(), mA.cols()) -
92 mA.cast<Complex>();
93
94 return h.inverse() * mB.cast<Complex>() * u;
95}

103void EMT::SSNComp::updateState(const Matrix &uOld, const Matrix &uNew) {
104 if (mA.rows() == 0)

123void EMT::SSNComp::mnaCompInitialize(Real, Real timeStep,
124 Attribute<Matrix>::Ptr leftVector) {
125 if (!mParametersSet)
126 throw std::logic_error(
127 "setParameters() must be called before initialization.");
128
129 mTimeStep = timeStep;
130 mXValue = &mX->get();
131 mInputValue = &inputAttribute()->get();
132 updateMatrixNodeIndices();
133
134 mRightVectorAffectedIndices.clear();
135 for (UInt terminal = 0; terminal < terminalNumberConnected(); ++terminal) {
136 if (terminalNotGrounded(terminal)) {
137 for (UInt index : matrixNodeIndices(terminal))
138 if (index < static_cast<UInt>(leftVector->get().rows()))
139 mRightVectorAffectedIndices.push_back(index);
140 }
141 }
142 for (UInt virtualNode = 0; virtualNode < virtualNodesNumber();
143 ++virtualNode) {
144 for (UInt index : virtualMatrixNodeIndices(virtualNode))
145 if (index < static_cast<UInt>(leftVector->get().rows()))
146 mRightVectorAffectedIndices.push_back(index);
147 }
148 std::sort(mRightVectorAffectedIndices.begin(),
149 mRightVectorAffectedIndices.end());
150 mRightVectorAffectedIndices.erase(
151 std::unique(mRightVectorAffectedIndices.begin(),
152 mRightVectorAffectedIndices.end()),
153 mRightVectorAffectedIndices.end());
154
155 recomputeDiscreteModel();
156 mHistoryState.resize(mA.rows(), 1);
157 mHistoryInput.resize(mA.rows(), 1);
158 calculateHistoryVector(mYHist);
159}

161void EMT::SSNComp::mnaAddCompactRightSideVectorStamp(Matrix &rightVector) {
