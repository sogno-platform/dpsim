46void EMT::VTypeSplitSSNComp::setSplitParameters(
47 const Matrix &plantA, const Matrix &plantB, const Matrix &plantC,
48 const Matrix &plantD, const Matrix &controllerOutputB,
49 const Matrix &measurementState, const Matrix &measurementTerminal) {
50 const Int plantStateSize = plantA.rows();
51
52 if (controllerOutputB.rows() != plantStateSize ||
53 controllerOutputB.cols() != mControllerOutputSize)
54 throw std::invalid_argument(
55 "Controller-output input matrix has invalid dimensions.");
56
57 if (measurementState.rows() != mControllerInputSize ||
58 measurementState.cols() != plantStateSize)
59 throw std::invalid_argument(
60 "Measurement state matrix has invalid dimensions.");
61
62 if (measurementTerminal.rows() != mControllerInputSize ||
63 measurementTerminal.cols() != plantB.cols())
64 throw std::invalid_argument(
65 "Measurement terminal matrix has invalid dimensions.");
66
67 SSNComp::setParameters(plantA, plantB, plantC, plantD);
68
69 mBControllerOutput = controllerOutputB;
70 mdBControllerOutput = Matrix::Zero(plantStateSize, mControllerOutputSize);
71 mMeasurementState = measurementState;
72 mMeasurementTerminal = measurementTerminal;
73
74 mControllerState.setZero();
75 mControllerMeasurementOld.setZero();
76 mControllerOutput.setZero();
77 mDelayedControllerOutput.setZero();
78
79 buildControllerStateSpaceModel(mControllerState, mControllerMeasurementOld,
80 mControllerA, mControllerB, mControllerC,
81 mControllerD, mControllerE, mControllerF);
82
83 const Int splitStateSize =
84 mControllerStateSize + plantStateSize + mControllerOutputSize;
85 mSplitDiscreteA = Matrix::Zero(splitStateSize, splitStateSize);
86 mSplitDiscreteB = Matrix::Zero(splitStateSize, plantB.cols());
87 mSplitHistoryC = Matrix::Zero(plantC.rows(), splitStateSize);
88}

90Matrix EMT::VTypeSplitSSNComp::buildControllerMeasurement(
91 const Matrix &plantState, const Matrix &terminalVoltage) const {
92 if (plantState.rows() != mA.rows() || plantState.cols() != 1)
93 throw std::invalid_argument(
94 "Split SSN plant state has invalid dimensions.");
95
96 if (terminalVoltage.rows() != mB.cols() || terminalVoltage.cols() != 1)
97 throw std::invalid_argument(
98 "Split SSN terminal voltage has invalid dimensions.");
99
100 return mMeasurementState * plantState +
101 mMeasurementTerminal * terminalVoltage;
102}

113void EMT::VTypeSplitSSNComp::rebuildSplitDiscreteModel(
114 const Matrix &controllerBdUsed) {
115 mControllerBdUsed = controllerBdUsed;
116
117 const UInt controllerOffset = 0;
118 const UInt plantOffset = static_cast<UInt>(mControllerStateSize);
119 const UInt delayOffset = plantOffset + static_cast<UInt>(mA.rows());
120
121 mSplitDiscreteA.setZero();
122 mSplitDiscreteB.setZero();
123 mSplitHistoryC.setZero();
124
125 const Matrix controllerInputUpdate =
126 mControllerAd * controllerBdUsed + mControllerBd;
127
128 const Matrix measurementFromPlantHistory = mMeasurementState;
129 const Matrix measurementFromDelay =
130 2.0 * mMeasurementState * mdBControllerOutput;
131 const Matrix measurementFromTerminal =
132 mMeasurementState * mdB + mMeasurementTerminal;
133
134 const Matrix delayedOutputFromMeasurement =
135 mControllerC * controllerBdUsed + mControllerD;
136
137 mSplitDiscreteA.block(controllerOffset, controllerOffset,
138 mControllerStateSize, mControllerStateSize) =
139 mControllerAd;
140 mSplitDiscreteA.block(controllerOffset, plantOffset, mControllerStateSize,
141 mA.rows()) =
142 controllerInputUpdate * measurementFromPlantHistory;
143 mSplitDiscreteA.block(controllerOffset, delayOffset, mControllerStateSize,
144 mControllerOutputSize) =
145 controllerInputUpdate * measurementFromDelay;
146 mSplitDiscreteB.block(controllerOffset, 0, mControllerStateSize, mB.cols()) =
147 controllerInputUpdate * measurementFromTerminal;
148
149 mSplitDiscreteA.block(plantOffset, plantOffset, mA.rows(), mA.cols()) = mdA;
150 mSplitDiscreteA.block(plantOffset, delayOffset, mA.rows(),
151 mControllerOutputSize) =
152 2.0 * mdA * mdBControllerOutput;
153 mSplitDiscreteB.block(plantOffset, 0, mA.rows(), mB.cols()) =
154 (mdA + Matrix::Identity(mA.rows(), mA.cols())) * mdB;
155
156 mSplitDiscreteA.block(delayOffset, controllerOffset, mControllerOutputSize,
157 mControllerStateSize) = mControllerC;
158 mSplitDiscreteA.block(delayOffset, plantOffset, mControllerOutputSize,
159 mA.rows()) =
160 delayedOutputFromMeasurement * measurementFromPlantHistory;
161 mSplitDiscreteA.block(delayOffset, delayOffset, mControllerOutputSize,
162 mControllerOutputSize) =
163 delayedOutputFromMeasurement * measurementFromDelay;
164 mSplitDiscreteB.block(delayOffset, 0, mControllerOutputSize, mB.cols()) =
165 delayedOutputFromMeasurement * measurementFromTerminal;
166
167 mSplitHistoryC.block(0, plantOffset, mC.rows(), mC.cols()) = mC;
168 mSplitHistoryC.block(0, delayOffset, mC.rows(), mControllerOutputSize) =
169 2.0 * mC * mdBControllerOutput;
170}

172void EMT::VTypeSplitSSNComp::recomputeDiscreteModel() {
173 SSNComp::recomputeDiscreteModel();
174
175 const Matrix identity = Matrix::Identity(mA.rows(), mA.cols());
176 const Matrix lhs = identity - 0.5 * mTimeStep * mA;
177 mdBControllerOutput =
178 lhs.fullPivLu().solve(0.5 * mTimeStep * mBControllerOutput);
179
180 recomputeControllerDiscreteModel();
181 rebuildSplitDiscreteModel(mControllerBd);
182}

184Matrix EMT::VTypeSplitSSNComp::calculateHistoryVector() const {
185 return mC * (mdA * (**mX) + mdB * (**mIntfVoltage) +
186 2.0 * mdBControllerOutput * mDelayedControllerOutput);

189void EMT::VTypeSplitSSNComp::updateState(const Matrix &uOld,
190 const Matrix &uNew) {
191 const Matrix delayedControllerOutputUsedThisStep = mDelayedControllerOutput;
192 const Matrix controllerBdUsedThisStep = mControllerBd;
193
194 **mX = mdA * (**mX) + mdB * (uNew + uOld) +
195 2.0 * mdBControllerOutput * delayedControllerOutputUsedThisStep;
196
197 const Matrix measurementNew = buildControllerMeasurement(**mX, uNew);
198
199 mControllerState =
200 mControllerAd * mControllerState +
201 mControllerBd * (measurementNew + mControllerMeasurementOld) +
202 mControllerEd;
203 mControllerMeasurementOld = measurementNew;
204
205 buildControllerStateSpaceModel(mControllerState, measurementNew, mControllerA,
206 mControllerB, mControllerC, mControllerD,
207 mControllerE, mControllerF);
208 recomputeControllerDiscreteModel();
209
210 rebuildSplitDiscreteModel(controllerBdUsedThisStep);
211
212 evaluateControllerOutput(mControllerState, measurementNew, mControllerOutput);
213 mDelayedControllerOutput = mControllerOutput;
214}
