19MatrixComp EMT::VTypeVariableSSNComp::calculateSteadyStateStateFromInput(
20 const MatrixComp &u, Real frequency) const {
21 const Real omega = 2.0 * PI * frequency;
22 MatrixComp h =
23 Complex(0.0, omega) * MatrixComp::Identity(mA.rows(), mA.cols()) -
24 mA.cast<Complex>();
25
26 return h.inverse() * (mB.cast<Complex>() * u + mE.cast<Complex>());
27}

29MatrixComp EMT::VTypeVariableSSNComp::calculateSteadyStateOutputFromInput(
30 const MatrixComp &x, const MatrixComp &u) const {
31 return mC.cast<Complex>() * x + mD.cast<Complex>() * u + mF.cast<Complex>();
32}

97void EMT::VTypeVariableSSNComp::initializeFromNodesAndTerminals(
98 Real frequency) {
99 if (!mParametersSet)
100 throw std::logic_error("setParameters() must be called before "
101 "initializeFromNodesAndTerminals().");
102
103 const MatrixComp uInit = buildInitialInputFromNodes(frequency);
104
105 **mIntfVoltage = uInit.real();
106 **mX = Matrix::Zero((**mX).rows(), 1);
107
108 SPDLOG_LOGGER_INFO(mSLog,
109 "\n--- Initialization from nodes and terminals ---");
110
111 // Fixed-point initialization for operating-point-dependent SSN models.
112 // The initial input is reconstructed from terminal quantities, then the
113 // component parameters are updated from the current state estimate and the
114 // steady-state state is recomputed. This is repeated until the state no
115 // longer changes noticeably or until the iteration limit is reached.
116 Bool converged = false;
117
118 for (Int iter = 0; iter < mInitializationMaxIterations; ++iter) {
119 const Matrix xPrev = **mX;
120
121 updateComponentParameters();
122
123 const MatrixComp xInit =
124 calculateSteadyStateStateFromInput(uInit, frequency);
125 **mX = xInit.real();
126
127 if ((**mX).isApprox(xPrev, mInitializationTolerance)) {
128 converged = true;
129 break;
130 }
131 }
132
133 if (!converged) {
134 SPDLOG_LOGGER_WARN(
135 mSLog,
136 "Fixed-point initialization did not converge within {} iterations.",
137 mInitializationMaxIterations);
138 }
139
140 // Ensure the local model corresponds to the final initialized state.
141 updateComponentParameters();
142
143 const MatrixComp xInit = (**mX).cast<Complex>();
144 const MatrixComp yInit = calculateSteadyStateOutputFromInput(xInit, uInit);
145
146 **mIntfCurrent = yInit.real();
147 updateLogAttributes(**mIntfVoltage);
148 mParameterChanged = false;
149
150 SPDLOG_LOGGER_INFO(
151 mSLog,
152 "\nInput u: {:s}"
153 "\nOutput y: {:s}"
154 "\nState x: {:s}"
155 "\n--- Initialization from nodes and terminals finished ---",
156 Logger::matrixToString(**mIntfVoltage),
157 Logger::matrixToString(**mIntfCurrent), Logger::matrixToString(**mX));
158}
