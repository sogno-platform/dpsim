251 // #### MNA Solver Tasks ####
253 class SolveTask : public CPS::Task {
254 public:
255 SolveTask(MnaSolverDirect<VarType> &solver)
256 : Task(solver.mName + ".Solve"), mSolver(solver) {
257 setFastExecutor<SolveTask, &SolveTask::executeFastImpl>(this);
258
259 for (auto it : solver.mMNAComponents) {
260 if (it->getRightVector()->get().size() != 0)
261 mAttributeDependencies.push_back(it->getRightVector());
262 }
263 // Switch interfaces are solved using the precomputed switch matrix
264 // bank, but their pre-step RHS stamps still feed the global solve.
265 for (auto it : solver.mMNAIntfSwitches) {
266 if (it->getRightVector()->get().size() != 0)
267 mAttributeDependencies.push_back(it->getRightVector());
268 }
269 for (auto node : solver.mNodes) {
270 mModifiedAttributes.push_back(node->mVoltage);
271 }
272 mModifiedAttributes.push_back(solver.mLeftSideVector);
273 }
274
275 void execute(Real time, Int timeStepCount) {
276 mSolver.solve(time, timeStepCount);
277 }
278 void executeFastImpl(Real time, Int timeStepCount) {
279 mSolver.solve(time, timeStepCount);
280 }
281
282 private:
283 MnaSolverDirect<VarType> &mSolver;
284 };

356 class StateSpaceExtractionTask : public CPS::Task {
357 public:
358 StateSpaceExtractionTask(MnaSolverDirect<VarType> &solver)
359 : Task(solver.mName + ".StateSpaceExtraction"), mSolver(solver) {
360 setFastExecutor<StateSpaceExtractionTask,
361 &StateSpaceExtractionTask::executeFastImpl>(this);
362 mAttributeDependencies.push_back(solver.mLeftSideVector);
363 if (solver.mStateSpaceExtractor) {
364 // Some contributors update their matrices in post-step;
365 // these dependencies prevent extraction from reading not updated matrices.
366 for (const auto &dependency :
367 solver.mStateSpaceExtractor->getAttributeDependencies())
368 mAttributeDependencies.push_back(dependency);
369 }
370 mModifiedAttributes.push_back(Scheduler::external);
371 }
372
373 void execute(Real time, Int timeStepCount) {
374 mSolver.extractStateSpace(time);
375 }
376 void executeFastImpl(Real time, Int timeStepCount) {
377 mSolver.extractStateSpace(time);
