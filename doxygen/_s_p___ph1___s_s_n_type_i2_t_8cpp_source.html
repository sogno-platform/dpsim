16void SP::Ph1::SSNTypeI2T::setParameters(const MatrixComp &A,
17 const MatrixComp &B,
18 const MatrixComp &C,
19 const MatrixComp &D) {
20
21 if (A.cols() != A.rows())
22 throw std::invalid_argument("A needs to be square.");
23
24 if (B.rows() != A.cols())
25 throw std::invalid_argument(
26 "Rows of B do not match columns and rows of A.");
27
28 if (B.cols() != 1)
29 throw std::invalid_argument(
30 "Two terminal I-type SSN component only has one "
31 "external current, so column number of B must be one.");
32
33 if (C.rows() != 1)
34 throw std::invalid_argument(
35 "A two terminal I type SSN component only has"
36 "one output voltage. C must have exactly one row.");
37
38 if (C.cols() != A.rows())
39 throw std::invalid_argument("Columns of C do not match rows of A.");
40
41 if (D.rows() != 1)
42 throw std::invalid_argument(
43 "A two terminal I type SSN component only has"
44 "one output voltage. D must have exactly one row.");
45
46 if (D.cols() != B.cols())
47 throw std::invalid_argument(
48 "Columns of D do not match columns of B and rows"
49 "of input vector u, which need to be one.");
50
51 mA = A;
52 mB = B;
53 mC = C;
54 mD = D;
55
56 mX = Matrix::Zero(mA.rows(), 1);
57 mU = Matrix::Zero(mB.cols(), 1);
58 mUPrev = Matrix::Zero(mB.cols(), 1);
59
60 mParametersSet = true;
61}

69void SP::Ph1::SSNTypeI2T::calculateAdmittance(Real omega) {
70
71 MatrixComp H_inv =
72 omega * Complex(0, 1.) * Matrix::Identity(mA.rows(), mA.cols()) - mA;
73
74 MatrixComp H = MatrixComp(H_inv.rows(), H_inv.cols());
75
76 H = H_inv.inverse().eval();
77
78 Complex denom = ((mC.eval() * H * mB.eval() + mD.eval()).value());
79
80 if (!Math::isFinite(denom) || std::abs(denom) < DOUBLE_EPSILON)
81 throw std::invalid_argument(
82 "calculateAdmittance: denominator is near zero or infinite.");
83
84 mAdmittance =
85 1. /
86 denom; //I-type: y=V, x=I ->Matrix factor is Impedance, not Admittance
87}

89void SP::Ph1::SSNTypeI2T::initializeFromNodesAndTerminals(Real frequency) {
90
91 SPDLOG_LOGGER_INFO(
92 mSLog, "\n--- Initialization from node voltages and terminals ---");
93 if (!mParametersSet)
94 throw std::logic_error("Not initialized.");
95
96 Real omega = 2 * PI * frequency;
97
98 calculateAdmittance(omega);
99
100 (**mIntfVoltage)(0, 0) = initialSingleVoltage(1) - initialSingleVoltage(0);
101 **mIntfCurrent = mAdmittance * **mIntfVoltage;
102
103 SPDLOG_LOGGER_INFO(mSLog,
104 "\nImpedance [Ohm]: {:s}"
105 "\nAdmittance [S]: {:s}",
106 Logger::complexToString(1. / mAdmittance),
107 Logger::complexToString(mAdmittance));
108 SPDLOG_LOGGER_INFO(mSLog,
109 "\n--- Initialization from powerflow ---"
110 "\nVoltage across: {:s}"
111 "\nCurrent: {:s}"
112 "\nTerminal 0 voltage: {:s}"
113 "\nTerminal 1 voltage: {:s}"
114 "\n--- Initialization from powerflow finished ---",
115 Logger::phasorToString((**mIntfVoltage)(0, 0)),
116 Logger::phasorToString((**mIntfCurrent)(0, 0)),
117 Logger::phasorToString(initialSingleVoltage(0)),
118 Logger::phasorToString(initialSingleVoltage(1)));
119}

168void SP::Ph1::SSNTypeI2T::mnaCompUpdateVoltage(const Matrix &leftVector) {
169 // v1 - v0
