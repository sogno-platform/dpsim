33void SP::Ph1::SynchronGenerator6bOrderVBR::specificInitialization() {
34
35 // initial voltage behind the transient reactance in the dq reference frame
36 (**mEdq_t)(0, 0) = (mLq - mLq_t) * (**mIdq)(1, 0);
37 (**mEdq_t)(1, 0) = **mEf - (mLd - mLd_t) * (**mIdq)(0, 0);
38
39 // initial dq behind the subtransient reactance in the dq reference frame
40 (**mEdq_s)(0, 0) = (**mVdq)(0, 0) - mLq_s * (**mIdq)(1, 0);
41 (**mEdq_s)(1, 0) = (**mVdq)(1, 0) + mLd_s * (**mIdq)(0, 0);
42
43 // initialize history term behind the transient reactance
44 mEh_t(0, 0) = mAd_t * (**mIdq)(1, 0) + mBd_t * (**mEdq_t)(0, 0);
45 mEh_t(1, 0) = mAq_t * (**mIdq)(0, 0) + mBq_t * (**mEdq_t)(1, 0) +
46 mDq_t * (**mEf) + mDq_t * mEf_prev;
47
48 SPDLOG_LOGGER_INFO(mSLog,
49 "\n--- Model specific initialization ---"
50 "\nSG model: 6th order type b (Anderson-Fouad's model)"
51 "\nInitial Ed_t (per unit): {:f}"
52 "\nInitial Eq_t (per unit): {:f}"
53 "\nInitial Ed_s (per unit): {:f}"
54 "\nInitial Eq_s (per unit): {:f}"
55 "\n--- Model specific initialization finished ---",
56
57 (**mEdq_t)(0, 0), (**mEdq_t)(1, 0), (**mEdq_s)(0, 0),
58 (**mEdq_s)(1, 0));
59 mSLog->flush();
60}

62void SP::Ph1::SynchronGenerator6bOrderVBR::stepInPerUnit() {
63
64 // calculate Edq_t at t=k
65 (**mEdq_t)(0, 0) = mAd_t * (**mIdq)(1, 0) + mEh_t(0, 0);
66 (**mEdq_t)(1, 0) = mAq_t * (**mIdq)(0, 0) + mEh_t(1, 0);
67
68 // calculate Edq_s at t=k
69 (**mEdq_s)(0, 0) = -(**mIdq)(1, 0) * mLq_s + (**mVdq)(0, 0);
70 (**mEdq_s)(1, 0) = (**mIdq)(0, 0) * mLd_s + (**mVdq)(1, 0);
71
72 mDqToComplexA = get_DqToComplexATransformMatrix();
73 mComplexAToDq = mDqToComplexA.transpose();
74
75 // calculate resistance matrix at t=k+1
76 calculateResistanceMatrix();
77
78 // calculate history term behind the transient reactance
79 mEh_t(0, 0) = mAd_t * (**mIdq)(1, 0) + mBd_t * (**mEdq_t)(0, 0);
80 mEh_t(1, 0) = mAq_t * (**mIdq)(0, 0) + mBq_t * (**mEdq_t)(1, 0) +
81 mDq_t * (**mEf) + mDq_t * mEf_prev;
82
83 // calculate history term behind the subtransient reactance
84 mEh_s(0, 0) = mAd_s * (**mIdq)(1, 0) + mBd_s * (**mEdq_t)(0, 0) +
85 mCd_s * (**mEdq_s)(0, 0);
86 mEh_s(1, 0) = mAq_s * (**mIdq)(0, 0) + mBq_s * (**mEdq_t)(1, 0) +
87 mCq_s * (**mEdq_s)(1, 0) + mDq_s * (**mEf) + mDq_s * mEf_prev;
88
89 // convert Edq_t into the abc reference frame
90 mEh_s = mDqToComplexA * mEh_s;
91 mEvbr = Complex(mEh_s(0, 0), mEh_s(1, 0)) * mBase_V_RMS;
92}
