127class Barrier {
128public:
130 Barrier() = delete;
133 Barrier(Int limit, Bool useCondition = false)
134 : mLimit(limit), mCount(0), mGeneration(0), mUseCondition(useCondition) {}
135
139 void wait() {
140 if (mUseCondition) {
141 std::unique_lock<std::mutex> lk(mMutex);
142 Int gen = mGeneration;
143 mCount++;
144 if (mCount == mLimit) {
145 mCount = 0;
146 mGeneration++;
147 lk.unlock();
148 mCondition.notify_all();
149 } else {
150 // necessary because of spurious wakeups
151 while (gen == mGeneration)
152 mCondition.wait(lk);
153 }
154 } else {
155 Int gen = mGeneration.load(std::memory_order_acquire);
156 // We need at least one release from each thread to ensure that
157 // every write from before the wait() is visible in every other thread,
158 // and the fetch needs to be an acquire anyway, so use acq_rel instead of acquire.
159 // (This generates the same code on x86.)
160 if (mCount.fetch_add(1, std::memory_order_acq_rel) == mLimit - 1) {
161 mCount.store(0, std::memory_order_relaxed);
162 mGeneration.fetch_add(1, std::memory_order_release);
163 } else {
164 while (mGeneration.load(std::memory_order_acquire) == gen)
165 ;
166 }
167 }
168 }
169
174 void signal() {
175 if (mUseCondition) {
176 std::unique_lock<std::mutex> lk(mMutex);
177 mCount++;
178 if (mCount == mLimit) {
179 mCount = 0;
180 mGeneration++;
181 lk.unlock();
182 mCondition.notify_all();
183 }
184 } else {
185 // No release here, as this call does not provide any synchronization anyway.
186 if (mCount.fetch_add(1, std::memory_order_acquire) == mLimit - 1) {
187 mCount.store(0, std::memory_order_relaxed);
188 mGeneration.fetch_add(1, std::memory_order_release);
189 }
190 }
191 }
192
193private:
195 Int mLimit;
197 std::atomic<Int> mCount;
199 std::atomic<Int> mGeneration;
200 Bool mUseCondition;
201
202 std::mutex mMutex;
203 std::condition_variable mCondition;
204};
