49Complex TopologicalTerminal::singlePower() {

58MatrixComp TopologicalTerminal::initialVoltage() {
59 if (mPhaseType == PhaseType::Single || mPhaseType == PhaseType::DC ||
60 mPhaseType == PhaseType::A)
61 return topologicalNodes()->initialVoltage().block(0, 0, 1, 1);
62 else if (mPhaseType == PhaseType::B)
63 return topologicalNodes()->initialVoltage().block(1, 0, 1, 1);
64 else if (mPhaseType == PhaseType::C)
65 return topologicalNodes()->initialVoltage().block(2, 0, 1, 1);
66 else
67 return topologicalNodes()->initialVoltage();
